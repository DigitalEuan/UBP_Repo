# toggle_history_decoder.py
from coherence_substrate import CoherenceState, Y_CONSTANT
from geometric_error_correction import restore_coherence

def toggle(state: CoherenceState, label: str) -> CoherenceState:
    """Apply a toggle (+1/-1 phase shift) and record it in history."""
    toggled = state * CoherenceState(-1.0, nrci=1.0, history=f"Toggle({label})")
    return toggled

def survive_restore(state: CoherenceState) -> (CoherenceState, bool):
    """Attempt coherence restoration. Return (state, survived?)"""
    restored, absorbed = restore_coherence(state, Y_CONSTANT)
    survived = not absorbed and restored.nrci > 0.999
    return restored, survived

# === Level 0: OffBit — Pure potential ===
offbit = CoherenceState(0.0, nrci=1.0, history="OffBit()")

# === Simulate A+ lifecycle ===
history = [("START", offbit.history, offbit.nrci)]

# Toggle A
state = toggle(offbit, "A")
state, survived = survive_restore(state)
history.append(("A-toggle", state.history, state.nrci))

# Toggle RhD
if survived:
    state = toggle(state, "RhD")
    state, survived = survive_restore(state)
    history.append(("RhD-toggle", state.history, state.nrci))

# Observer binding
if survived:
    O_OBSERVER = CoherenceState(3.778, nrci=1.0, history="O_observer")
    state = state * O_OBSERVER
    history.append(("Bind(Observer)", state.history, state.nrci))

# === Output: The fossilized sequence ===
print("\n🩸 Blood Type A+ as Fossilized Toggle Sequence:\n")
for step, h, nrci in history:
    print(f"{step:15} | NRCI={nrci:.6f} | History: {h}")

# Now, compute Jaccard similarity between A+ and B+ histories:
A_hist_set = {"Toggle(A)", "Toggle(RhD)", "O_observer"}
B_hist_set = {"Toggle(B)", "Toggle(RhD)", "O_observer"}
intersection = A_hist_set & B_hist_set
union = A_hist_set | B_hist_set
jaccard = 1 - len(intersection)/len(union) if union else 1.0
print(f"\nJaccard Distance (A+ ↔ B+): {jaccard:.2f} (Shared: {intersection})")
