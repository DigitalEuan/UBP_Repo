PI = math.pi
Y = PI / (PI**2 + 2)                    # 0.264675430404527 (geometric resonance)
Y_INVERSE = PI + 2/PI                   # 3.778212425957375 (observer cost)
O_OBSERVER = Y_INVERSE                  # Observer emerges from geometry
NRCI_TARGET = 0.999997                  # Supercoherent regime
GLR_EPS = 1e-4                          # GLR completion threshold

# Verify involutory property (within floating-point limits)

T = TypeVar('T', float, complex, bool)


@dataclass
class CoherenceMetrics:
    """Container for coherence metrics of any operation."""
    nrci: float
    closure_error: float
    refinements: int
    operation_complexity: float
    is_coherent: bool
    
    def __repr__(self):
        coherence_status = "✓ COHERENT" if self.is_coherent else "✗ INCOHERENT"
        return (
            f"CoherenceMetrics({coherence_status}, nrci={self.nrci:.8f}, "
            f"closure_err={self.closure_error:.2e}, refinements={self.refinements})"
        )


class CoherenceState(Generic[T]):
    """
    The fundamental unit of the UBP substrate.
    
    Every value in this system is not just a number or boolean—it's a coherence state
    that knows its own trustworthiness (NRCI), its refinement history, and can verify
    its own structural integrity through bidirectional closure tests.
    
    This is information-first computation: the value and its quality are inseparable.
    """
    
    __slots__ = ('value', 'log_nrci_error', 'net_refinements', 'metadata')
    
    def __init__(self, 
                 value: T, 
                 log_nrci_error: Optional[float] = None,
                 net_refinements: int = 0,
                 metadata: Optional[Dict] = None):
        """
        Initialize a coherence state.
        
        Args:
            value: The primary value (float, complex, bool)
            log_nrci_error: Log-space representation of 1-NRCI (smaller = better)
            net_refinements: Net count of Y-refinements applied (+forward, -backward)
            metadata: Optional dictionary for additional context
        """
        self.value = value
        # Default to target coherence if not specified
        self.log_nrci_error = log_nrci_error if log_nrci_error is not None else math.log(1 - NRCI_TARGET)
        self.net_refinements = net_refinements
        self.metadata = metadata or {}
    
    @property
    def nrci(self) -> float:
        """Compute NRCI (Non-Random Coherence Index) from log-error space."""
        # NRCI = 1 - exp(log_nrci_error), bounded between 0 and 1
        return max(0.0, min(1.0, 1.0 - math.exp(self.log_nrci_error)))
    
    @property
    def is_coherent(self) -> bool:
        """Check if this state meets the minimum coherence threshold."""
        return self.nrci > NRCI_TARGET and abs(self.closure_error) < 1e-12
    
    @property
    def closure_error(self) -> float:
        """Compute bidirectional closure error for this state."""
        if isinstance(self.value, (int, float)):
            return self._numeric_closure_error()
        elif isinstance(self.value, bool):
            return self._logical_closure_error()
        elif isinstance(self.value, complex):
            return self._complex_closure_error()
        return 0.0
    
    def _numeric_closure_error(self) -> float:
        """Compute closure error for numeric values."""
        original = float(self.value)
        if original == 0:
            return 0.0
        
        # Forward refinement (geometry space)
        forward = original * Y
        # Backward refinement (observer space)
        backward = forward * Y_INVERSE
        
        # Relative error in round trip
        return abs(backward - original) / abs(original)
    
    def _logical_closure_error(self) -> float:
        """Compute closure error for logical values."""
        # For logical values, closure is about stability under refinement
        # This is a simplified version - real implementation would track
        # logical structure through refinements
        stability_factor = math.exp(-abs(self.log_nrci_error))
        return 1.0 - stability_factor
    
    def _complex_closure_error(self) -> float:
        """Compute closure error for complex values."""
        if not isinstance(self.value, complex):
            return 0.0
        
        original = self.value
        if original == 0:
            return 0.0
        
        # Apply Y-refinement to both real and imaginary parts
        forward = complex(original.real * Y, original.imag * Y)
        backward = complex(forward.real * Y_INVERSE, forward.imag * Y_INVERSE)
        
        # Relative error magnitude
        error = abs(backward - original) / abs(original)
        return error
    
    def degrade_by(self, delta_log_error: float) -> 'CoherenceState[T]':
        """
        Apply controlled coherence degradation.
        
        This models the cost of operations, noise, or uncertainty.
        Uses additive accumulation in log-space for numerical stability.
        """
        return CoherenceState(
            self.value,
            self.log_nrci_error + delta_log_error,
            self.net_refinements,
            self.metadata.copy()
        )
    
    def refine_forward(self) -> 'CoherenceState[T]':
        """
        Apply Y-refinement: transform to geometry space.
        
        This compresses information into a more stable representation.
        Used for integration, noise reduction, and structural analysis.
        """
        if isinstance(self.value, (int, float)):
            new_value = float(self.value) * Y
        elif isinstance(self.value, complex):
            new_value = complex(self.value.real * Y, self.value.imag * Y)
        elif isinstance(self.value, bool):
            # For logical states, refinement represents abstraction
            new_value = self.value
        else:
            new_value = self.value
        
        # Coherence typically improves with forward refinement (for stable systems)
        # But add a small operational cost
        improvement = -0.5 * abs(math.log(Y))  # Negative log error = improvement
        operational_cost = 0.1 * abs(math.log(Y))
        
        return CoherenceState(
            new_value,
            self.log_nrci_error + improvement + operational_cost,
            self.net_refinements + 1,
            self.metadata.copy()
        )
    
    def refine_backward(self) -> 'CoherenceState[T]':
        """
        Apply inverse refinement: transform to observer space.
        
        This expands information back to the original representation.
        Used for interpretation, measurement, and application.
        """
        if isinstance(self.value, (int, float)):
            new_value = float(self.value) * Y_INVERSE
        elif isinstance(self.value, complex):
            new_value = complex(self.value.real * Y_INVERSE, self.value.imag * Y_INVERSE)
        elif isinstance(self.value, bool):
            # For logical states, backward refinement represents instantiation
            new_value = self.value
        else:
            new_value = self.value
        
        # Backward refinement typically has higher operational cost
        operational_cost = 0.3 * abs(math.log(Y))
        
        return CoherenceState(
            new_value,
            self.log_nrci_error + operational_cost,
            self.net_refinements - 1,
            self.metadata.copy()
        )
    
    def bidirectional_closure_test(self) -> Tuple[float, bool]:
        """
        Test if this state maintains perfect round-trip integrity.
        
        Returns:
            (closure_error, is_perfect): The relative error and whether it's within tolerance
        """
        error = self.closure_error
        return error, error < 1e-12
    
    def metrics(self) -> CoherenceMetrics:
        """Get comprehensive coherence metrics for this state."""
        closure_err, is_perfect = self.bidirectional_closure_test()
        complexity = abs(self.log_nrci_error) / 10.0  # Simplified complexity metric
        
        return CoherenceMetrics(
            nrci=self.nrci,
            closure_error=closure_err,
            refinements=self.net_refinements,
            operation_complexity=complexity,
            is_coherent=self.is_coherent
        )
    
    def __repr__(self):
        return f"CoherenceState(value={self.value}, nrci={self.nrci:.8f}, refinements={self.net_refinements})"

def _estimate_operation_complexity(a: CoherenceState, b: Optional[CoherenceState] = None) -> float:
    """
    Estimate the coherence cost of an operation based on input states.
    
    This models how different operations affect coherence based on:
    - Magnitude of values
    - Current NRCI of inputs
    - Operation type complexity
    """
    complexity = abs(math.log(a.nrci + 1e-20))  # Base complexity from input coherence
    
    if b is not None:
        complexity += abs(math.log(b.nrci + 1e-20))
        # Scale by relative magnitude difference
        if isinstance(a.value, (int, float)) and isinstance(b.value, (int, float)):
            mag_ratio = max(abs(a.value), abs(b.value)) / (min(abs(a.value), abs(b.value)) + 1e-20)
            complexity *= math.log1p(mag_ratio)
    
    return complexity * COMPLEXITY_SCALE


def _coherence_transform(
    state: CoherenceState, 
    operation: Callable[[Any], Any], 
    name: str = "transform",
    preserve_closure: bool = True
) -> CoherenceState:
    """
    Apply an operation while maintaining coherence tracking.
    
    This is the core computational primitive of the UBP substrate.
    Every numerical operation is wrapped in coherence preservation.
    """
    try:
        # Apply the operation to the raw value
        result_value = operation(state.value)
        
        # Estimate coherence degradation based on operation complexity
        complexity = _estimate_operation_complexity(state)
        delta_log_error = complexity * abs(math.log(Y))
        
        # Create new state with degraded coherence
        new_state = CoherenceState(result_value).degrade_by(delta_log_error)
        
        # Apply Y-refinement if needed for stability
        if preserve_closure:
            # Forward then backward refinement to stabilize
            new_state = new_state.refine_forward().refine_backward()
        
        # Update metadata
        metadata = new_state.metadata.copy()
        metadata.update({
            'operation': name,
            'input_nrci': state.nrci,
            'complexity': complexity
        })
        new_state.metadata = metadata
        
        return new_state
    
    except Exception as e:
        # On error, return a low-coherence state with error metadata
        error_state = CoherenceState(0.0, log_nrci_error=0.0)  # NRCI = 0
        error_state.metadata['error'] = str(e)
        error_state.metadata['operation'] = name
        return error_state

def integrate(
    f: Callable[[float], float], 
    a: float, 
    b: float,
    exact: Optional[float] = None,
    max_refinements: int = 5
) -> Tuple[float, CoherenceMetrics]:
    """
    Numerical integration as coherence accumulation.
    
    Unlike traditional quadrature methods, this approach:
    1. Tracks coherence (NRCI) throughout the integration
    2. Uses adaptive sampling based on local coherence
    3. Applies Y-refinement for stability
    4. Self-certifies results via closure tests
    
    Args:
        f: Function to integrate
        a, b: Integration limits
        exact: Known exact value (for validation)
        max_refinements: Maximum Y-refinements to apply
    
    Returns:
        (result, metrics): The integral value and its coherence metrics
    """
    if a > b:
        result, metrics = integrate(f, b, a, exact, max_refinements)
        return -result, metrics
    
    # Initial state with perfect coherence
    state = CoherenceState(0.0, log_nrci_error=-15.0)  # Very high initial NRCI
    
    # Adaptive sampling based on interval size and function behavior
    n_samples = max(10, int(abs(b - a) * 10))
    h = (b - a) / n_samples
    
    # Accumulate with coherence tracking
    for i in range(n_samples + 1):
        x = a + i * h
        weight = 1.0 if (i == 0 or i == n_samples) else 2.0
        
        try:
            fx = f(x)
        except:
            fx = 0.0
        
        # Trapezoidal rule contribution
        contribution = fx * weight * h / 2.0
        
        # Update state with contribution
        state.value += contribution
        
        # Estimate local coherence degradation from function curvature
        if i > 0:
            prev_x = a + (i-1) * h
            try:
                prev_fx = f(prev_x)
                # Estimate second derivative (curvature)
                curvature = abs(fx - 2*prev_fx + (f(prev_x - h) if i > 1 else prev_fx)) / (h**2 + 1e-20)
                # Convert curvature to coherence degradation
                delta_log_error = 0.1 * curvature * h * abs(math.log(Y))
                state = state.degrade_by(delta_log_error)
            except:
                pass
    
    # Apply Y-refinement for stability
    for _ in range(max_refinements):
        state = state.refine_forward()
    for _ in range(max_refinements):
        state = state.refine_backward()
    
    # Compute metrics
    metrics = state.metrics()
    
    # If exact value is known, update NRCI based on actual error
    if exact is not None:
        actual_error = abs(state.value - exact) / (abs(exact) + 1e-20)
        # Update log_nrci_error based on actual performance
        state.log_nrci_error = math.log(min(0.999999, max(1e-15, actual_error)))
        metrics = state.metrics()
        metrics.nrci = state.nrci
    
    return state.value, metrics
