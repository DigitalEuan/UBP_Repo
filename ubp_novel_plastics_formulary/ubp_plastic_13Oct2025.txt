# Simple BitTab encoding simulation for periodic table
# Assume: atomic_number (7 bits), period (3 bits), group (5 bits), block (2 bits: s=0,p=1,d=2,f=3 but 2 bits), valence (3 bits)

def encode_element(atomic_num, period, group, block, valence):
    # Block map: s=00, p=01, d=10, f=11
    block_code = {'s': 0b00, 'p': 0b01, 'd': 0b10, 'f': 0b11}[block]
    bits = np.binary_repr(atomic_num, 7) + np.binary_repr(period, 3) + np.binary_repr(group, 5) + np.binary_repr(block_code, 2) + np.binary_repr(valence, 3)
    # Pad to 24 bits if needed
    bits = bits.zfill(24)
    return bits

        for element, percentage in elements_with_percentages.items():
            if percentage > 0:
                freq = self.ubp_elemental_frequencies.get(element)
                if freq is None:
                    continue
                weight = percentage / 100.0
                freq_ratio = min(freq, avg_frequency) / max(freq, avg_frequency) if max(freq, avg_frequency) > 0 else 0.0
                freq_diff = abs(freq - avg_frequency)
                element_coherence = freq_ratio * math.exp(-0.0002 * (freq_diff / max(avg_frequency, 1e-15))**2)
                coherence_sum += element_coherence * weight
        return min(1.0, coherence_sum / total_weight) if total_weight > 0 else 0.0

    def compute_structure_coherence(self, composition: MaterialComposition, structure: Union[CrystalStructure, PolymerStructure, str], temperature: float = 20.0) -> float:
        if self.material_category == MaterialCategory.METALLIC:
            base_coherence = 0.5
            if structure in self.crystal_parameters:
                base_coherence = self.crystal_parameters[structure]["ubp_coherence_base"]
            temp_kelvin = temperature + 273.15
            temp_factor = math.exp(-0.0001 * (temp_kelvin - 293.15)**2)
            elemental_coherence = self.compute_ubp_elemental_coherence(composition)
            carbon_content = composition.elements.get("C", 0.0)
            carbon_effect = 1.0
            if structure == CrystalStructure.FERRITE:
                if carbon_content < 0.02:
                    carbon_effect = 1.0 + (0.02 - carbon_content) * 5.0
                else:
                    carbon_effect = math.exp(-carbon_content * 5.0)
            elif structure == CrystalStructure.AUSTENITE:
                carbon_effect = 1.0 + carbon_content * 0.2
            elif structure == CrystalStructure.MARTENSITE:
                if carbon_content > 0.2 and carbon_content < 1.0:
                    carbon_effect = 1.0 + (carbon_content - 0.2) * 0.8
                else:
                    carbon_effect = 0.5
            elif structure == CrystalStructure.PEARLITE:
                carbon_effect = 1.0 - abs(carbon_content - 0.76) * 0.8
            alloy_effect = 1.0
            if structure == CrystalStructure.FERRITE:
                alloy_effect *= (1.0 + composition.elements.get("Cr", 0.0) * 0.005)
            if structure == CrystalStructure.AUSTENITE:
                alloy_effect *= (1.0 + composition.elements.get("Ni", 0.0) * 0.01)
            total_coherence = (base_coherence * temp_factor * elemental_coherence * carbon_effect * alloy_effect)
            total_coherence *= 0.98
            return min(1.0, max(0.0, total_coherence))
        elif self.material_category == MaterialCategory.POLYMER:
            return 0.6
        return 0.5

    def predict_material_structure(self, composition: MaterialComposition, temperature: float = 20.0, processing: ProcessingMethod = ProcessingMethod.NORMALIZING) -> Union[CrystalStructure, PolymerStructure, str]:
        if self.material_category == MaterialCategory.METALLIC:
            structure_scores = {}
            possible_structures = [CrystalStructure.FERRITE, CrystalStructure.AUSTENITE, CrystalStructure.MARTENSITE, CrystalStructure.PEARLITE]
            A1_TEMP_C = 727
            A3_TEMP_C = 912
            carbon_content = composition.elements.get("C", 0.0)
            nickel_content = composition.elements.get("Ni", 0.0)
            chromium_content = composition.elements.get("Cr", 0.0)
            manganese_content = composition.elements.get("Mn", 0.0)
            for structure in possible_structures:
                score = 0.5
                if structure == CrystalStructure.FERRITE:
                    score += max(0, 1.0 - carbon_content * 5.0)
                    score += chromium_content * 0.05
                    score += composition.elements.get("Si", 0.0) * 0.03
                elif structure == CrystalStructure.AUSTENITE:
                    score += carbon_content * 0.2
                    score += nickel_content * 0.1
                    score += manganese_content * 0.08
                elif structure == CrystalStructure.MARTENSITE:
                    score += carbon_content * 0.3
                    score += (chromium_content + manganese_content) * 0.02
                elif structure == CrystalStructure.PEARLITE:
                    score += (1.0 - abs(carbon_content - 0.76) * 1.5)
                    score += manganese_content * 0.05
                if temperature > A3_TEMP_C:
                    if structure == CrystalStructure.AUSTENITE:
                        score *= 2.5
                    else:
                        score *= 0.5
                elif temperature > A1_TEMP_C:
                    if structure == CrystalStructure.AUSTENITE:
                        score *= 1.5
                    elif structure == CrystalStructure.FERRITE:
                        score *= 1.2
                elif temperature <= 20.0:
                    if structure == CrystalStructure.AUSTENITE:
                        if nickel_content > 8.0 or manganese_content > 10.0:
                            score *= 1.5
                        else:
                            score *= 0.2
                    elif structure == CrystalStructure.MARTENSITE:
                        score *= 1.5
                    elif structure == CrystalStructure.PEARLITE:
                        score *= 1.2
                    elif structure == CrystalStructure.FERRITE:
                        score *= 1.0
                if processing == ProcessingMethod.ANNEALING:
                    if structure in [CrystalStructure.FERRITE, CrystalStructure.PEARLITE]:
                        score *= 1.8
                    elif structure == CrystalStructure.AUSTENITE and temperature <= A1_TEMP_C:
                        score *= 0.5
                    else:
                        score *= 0.7
                elif processing == ProcessingMethod.NORMALIZING:
                    if structure == CrystalStructure.PEARLITE:
                        score *= 1.5
                    elif structure == CrystalStructure.FERRITE:
                        score *= 1.2
                    else:
                        score *= 0.8
                elif processing == ProcessingMethod.QUENCHING:
                    if structure == CrystalStructure.MARTENSITE:
                        if carbon_content > 0.15:
                            score *= 3.0
                        else:
                            score *= 0.8
                    else:
                        score *= 0.3
                elif processing == ProcessingMethod.TEMPERING:
                    if structure == CrystalStructure.MARTENSITE:
                        score *= 1.1
                structure_coherence_factor = self.compute_structure_coherence(composition, structure, temperature)
                score *= structure_coherence_factor
                structure_scores[structure] = score
            if not structure_scores:
                return "unknown"
            return max(structure_scores, key=structure_scores.get)
        elif self.material_category == MaterialCategory.POLYMER:
            return PolymerStructure.AMORPHOUS
        return "unknown"

        properties = {}
        properties[MaterialProperty.TENSILE_STRENGTH] = self.predict_tensile_strength(composition, structure, processing)
        properties[MaterialProperty.HARDNESS] = self.predict_hardness(composition, structure, processing)
        properties[MaterialProperty.DUCTILITY] = self.predict_ductility(composition, structure, processing)
        if self.material_category == MaterialCategory.METALLIC:
            yield_factor = 0.7 if structure == CrystalStructure.AUSTENITE else 0.75
            properties[MaterialProperty.YIELD_STRENGTH] = properties[MaterialProperty.TENSILE_STRENGTH] * yield_factor
            base_modulus = 200.0
            modulus_variation = (
                composition.elements.get("Cr", 0.0) * 0.5 +
                composition.elements.get("Ni", 0.0) * 0.3 +
                composition.elements.get("Mo", 0.0) * 1.0 +
                composition.elements.get("W", 0.0) * 2.0
            )
            properties[MaterialProperty.ELASTIC_MODULUS] = base_modulus + modulus_variation
        elif self.material_category == MaterialCategory.POLYMER:
            properties[MaterialProperty.GLASS_TRANSITION_TEMP] = 80.0
            properties[MaterialProperty.MELTING_POINT] = 180.0
        elemental_coherence = self.compute_ubp_elemental_coherence(composition)
        structure_coherence = self.compute_structure_coherence(composition, structure, temperature)
        ubp_metrics = {
            "elemental_coherence": elemental_coherence,
            "structure_coherence": structure_coherence,
            "overall_coherence": (elemental_coherence + structure_coherence) / 2,
            "composition_balance": composition.get_total_composition() / 100.0,
            "processing_compatibility": self.processing_effects.get(processing, {}).get("coherence_factor", 1.0)
        }
        confidence = min(1.0, ubp_metrics["overall_coherence"] * ubp_metrics["composition_balance"])
        prediction = MaterialPrediction(
            composition=composition,
            material_category=self.material_category,
            structure=structure,
            processing_method=processing,
            temperature=temperature,
            properties=properties,
            ubp_metrics=ubp_metrics,
            confidence=confidence,
            metadata={"prediction_timestamp": time.time(), "ubp_version": "2.0.0"}
        )
        return prediction

    def optimize_composition(self, target_properties: Dict[MaterialProperty, float], processing: ProcessingMethod = ProcessingMethod.NORMALIZING, max_iterations: int = 10000, learning_rate: float = 0.002) -> Tuple[MaterialComposition, MaterialPrediction]:
        if self.material_category == MaterialCategory.METALLIC:
            element_ranges = {
                "C": (0.01, 1.5), "Mn": (0.1, 1.5), "Si": (0.1, 1.0), "P": (0.0, 0.04), "S": (0.0, 0.04),
                "Cr": (0.0, 10.0), "Ni": (0.0, 8.0), "Mo": (0.0, 2.0), "V": (0.0, 1.0), "W": (0.0, 2.0),
                "Co": (0.0, 2.0), "Al": (0.0, 0.5), "Cu": (0.0, 1.0), "Ti": (0.0, 0.5), "Nb": (0.0, 0.5)
            }
            base_element_type = "Fe"
        elif self.material_category == MaterialCategory.POLYMER:
            element_ranges = {"C": (40.0, 80.0), "H": (5.0, 15.0), "O": (0.0, 10.0), "N": (0.0, 5.0), "Cl": (0.0, 20.0)}
            base_element_type = "C"
        else:
            raise ValueError(f"Optimization not implemented for material category: {self.material_category}")
        initial_elements = {}
        for elem, (min_val, max_val) in element_ranges.items():
            if elem == base_element_type: continue
            initial_elements[elem] = min_val + (max_val - min_val) * 0.15
        best_composition = MaterialComposition(base_element=base_element_type, elements=initial_elements)
        best_score = float('inf')
        best_prediction = None
        for iteration in range(max_iterations):
            test_elements = {}
            for element, (min_val, max_val) in element_ranges.items():
                if element == base_element_type: continue
                current_val = best_composition.elements.get(element, 0.0)
                variation_amplitude = learning_rate * (1.0 - (iteration / max_iterations) * 0.5) * (max_val - min_val)
                variation = np.random.normal(0, variation_amplitude)
                value = np.clip(current_val + variation, min_val, max_val)
                test_elements[element] = value
            test_composition = MaterialComposition(base_element=base_element_type, elements=test_elements)
            if test_composition.get_total_composition() > 100.01:
                continue
            prediction = self.predict_all_properties(test_composition, processing)
            score = 0.0
            for prop, target_value in target_properties.items():
                if prop in prediction.properties:
                    predicted_value = prediction.properties[prop]
                    deviation = abs(predicted_value - target_value)
                    if target_value > 1e-6:
                        score += (deviation / target_value)**2 * 100.0
                    else:
                        score += deviation**2 * 100.0
            if score < best_score:
                best_score = score
                best_composition = test_composition
                best_prediction = prediction
        return best_composition, best_prediction

    def validate_materials_analysis(self) -> Dict[str, Any]:

            else:

base_freq_scale = 1e9
freqs = {el: base_freq_scale * (atomic_nums[el] / 100.0)**2 + (atomic_nums[el] * 1e6) for el in elements}

# Ranges for alloy % (wt%, sum to 100)
ranges = {'Fe':(50,95), 'Cr':(0,20), 'Ni':(0,20), 'Mn':(0,2), 'C':(0,2), 'Mo':(0,5), 'V':(0,2)}

def compute_coherence(comp):
    total_freq = 0.0
    total_weight = 0.0
    for el, pct in comp.items():
        if pct > 0:
            w = pct / 100.0
            f = freqs.get(el, 1e9)
            total_freq += f * w
            total_weight += w
    if total_weight == 0: return 0.0
    avg_freq = total_freq / total_weight
    coh_sum = 0.0
    for el, pct in comp.items():
        if pct > 0:
            w = pct / 100.0
            f = freqs.get(el, 1e9)
            ratio = min(f, avg_freq) / max(f, avg_freq) if max(f, avg_freq) > 0 else 0.0
            diff = abs(f - avg_freq)
            el_coh = ratio * np.exp(-0.0002 * (diff / max(avg_freq, 1e-15))**2)
            coh_sum += el_coh * w
    return min(1.0, coh_sum / total_weight) if total_weight > 0 else 0.0

            o_wt = ox_wt * (48 / 225.81)
