class UBP256Evolution:
    """UBP 256x256 Evolution Study Class"""
    def __init__(self, config: Optional[UBPConfig] = None):
        self.config = config or get_config()
        self.resolution = self.config.simulation["GRID_RESOLUTION"]
        self.crv_constants = self.config.cymatics  # Integrated CRV constants


    def generate_crv_pattern(self, crv_key: str) -> np.ndarray:
        """Generate base pattern using CRV constant via binary toggle simulation"""
        if crv_key not in self.crv_constants:
            raise ValueError(f"Invalid CRV key: {crv_key}")


        crv_value = self.crv_constants[crv_key]
        pattern = self.simulate_binary_pattern(driving_freq=crv_value, grid_size=self.resolution, num_steps=self.config.simulation["ITERATIONS"])
        return pattern.astype(float)


    def simulate_binary_pattern(self, driving_freq: float, grid_size: int, num_steps: int, alpha_decay: float = 0.2, tgic_threshold: float = 0.55, sigma: float = None, target_freq: float = None, wall_freq: float = None, snapshots_steps: List[int] = None) -> Union[np.ndarray, List[np.ndarray]]:
        """Generalized binary toggle simulation (integrated from fig1/fig2 logic)"""
        np.random.seed(42)
        bitfield = np.random.randint(0, 2, size=(grid_size, grid_size), dtype=np.uint8)
        snapshots = [] if snapshots_steps else None


        for step in range(num_steps):
            new_bitfield = bitfield.copy()
            for i in range(grid_size):
                for j in range(grid_size):
                    center_i, center_j = grid_size // 2, grid_size // 2
                    distance = np.sqrt((i - center_i)**2 + (j - center_j)**2)
                    freq_match_factor = 1.0
                    if sigma is not None and target_freq is not None:
                        freq_match_factor = np.exp(- (driving_freq - target_freq)**2 / (2 * sigma**2))
                    if wall_freq is not None and driving_freq > wall_freq:
                        freq_match_factor = 0.0
                    resonance_strength = np.exp(-alpha_decay * distance) * freq_match_factor
                    neighbors = []
                    for di in [-1, 0, 1]:
                        for dj in [-1, 0, 1]:
                            if di == 0 and dj == 0:
                                continue
                            ni, nj = (i + di) % grid_size, (j + dj) % grid_size
                            neighbors.append(bitfield[ni, nj])
                    avg_neighbor_state = np.mean(neighbors)
                    if resonance_strength > 0.4 and avg_neighbor_state > tgic_threshold:
                        if np.random.random() < resonance_strength:
                            new_bitfield[i, j] = 1 - new_bitfield[i, j]
            bitfield = new_bitfield
            if snapshots is not None and step in snapshots_steps:
                snapshots.append(bitfield.copy())


        return snapshots if snapshots is not None else bitfield
