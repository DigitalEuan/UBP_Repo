/**
 * LINGUIST PANEL — Transparency & Persistence UI
 *
 * Shows what the MoE Linguist has learned, why, and allows
 * download/upload of the learned manifold JSON for session continuity.
 */
import React, { useState, useRef } from 'react';
import { motion } from 'framer-motion';
import { Download, Upload, Brain, BookOpen, TrendingUp, AlertTriangle, CheckCircle2, RefreshCw } from 'lucide-react';
import { Button } from '@/components/ui/button';

function Section({ title, icon: Icon, color = 'cyan', children }) {
  return (
    <div className="p-3 bg-slate-800/30 rounded-xl border border-slate-700/50">
      <h4 className={`text-xs font-medium mb-3 flex items-center gap-2 text-${color}-400`}>
        <Icon className="w-3.5 h-3.5" />
        {title}
      </h4>
      {children}
    </div>
  );
}

export function LinguistPanel({ engine }) {
  const [importing, setImporting] = useState(false);
  const [importStatus, setImportStatus] = useState(null);
  const fileRef = useRef(null);

  if (!engine) return <p className="text-xs text-slate-500">Engine not loaded.</p>;

  const linguist = engine.linguist;
  const s = linguist.stats();
  const log = linguist.getLog ? linguist.getLog() : [];
  const topContexts = linguist.getTopContexts ? linguist.getTopContexts(20) : [];
  const gapReport = linguist.getGapReport ? linguist.getGapReport() : null;

  // ── Download ──────────────────────────────────────────────────────────────
  const handleDownload = () => {
    const blob = new Blob([JSON.stringify(linguist.exportManifold(), null, 2)], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `ubp_linguist_${Date.now()}.json`;
    a.click();
    URL.revokeObjectURL(url);
  };

  // ── Upload ────────────────────────────────────────────────────────────────
  const handleFileChange = async (e) => {
    const file = e.target.files?.[0];
    if (!file) return;
    setImporting(true);
    setImportStatus(null);
    try {
      const text = await file.text();
      const data = JSON.parse(text);
      linguist.importManifold(data);
      setImportStatus({ ok: true, msg: `Loaded ${data.contextCount ?? '?'} contexts · ${data.totalLearned?.toLocaleString() ?? '?'} samples` });
    } catch (err) {
      setImportStatus({ ok: false, msg: `Import failed: ${err.message}` });
    }
    setImporting(false);
    e.target.value = '';
  };

  return (
    <div className="space-y-4">
      <div className="text-center pb-3 border-b border-slate-700/50">
        <h2 className="text-lg font-semibold text-slate-100">MoE Linguist</h2>
        <p className="text-xs text-slate-400 mt-0.5">Geometric N-gram Cortex · session persistence</p>
      </div>

      {/* Status */}
      <div className="flex items-center gap-3 p-3 bg-slate-800/30 rounded-xl border border-slate-700/50">
        <div className={`w-2.5 h-2.5 rounded-full flex-shrink-0 ${s.trained ? 'bg-green-400' : 'bg-amber-400 animate-pulse'}`} />
        <div className="flex-1 min-w-0">
          <p className="text-xs text-slate-200">{s.trained ? 'Active & trained' : 'Training in background…'}</p>
          <p className="text-xs text-slate-500">{s.contexts?.toLocaleString()} contexts · vocab {s.vocab} · {s.totalLearned?.toLocaleString()} samples learned</p>
        </div>
      </div>

      {/* Save / Load */}
      <Section title="Save & Load Manifold" icon={Brain} color="cyan">
        <p className="text-xs text-slate-400 mb-3">
          Download the current manifold JSON and re-upload it next session to continue learning where you left off.
        </p>
        <div className="flex gap-2 flex-wrap">
          <Button size="sm" onClick={handleDownload} disabled={!s.trained}
            className="bg-cyan-600 hover:bg-cyan-500 text-white text-xs h-8 px-3">
            <Download className="w-3 h-3 mr-1" /> Download Manifold
          </Button>
          <Button size="sm" variant="outline" onClick={() => fileRef.current?.click()} disabled={importing}
            className="border-slate-600 text-slate-300 hover:bg-slate-700 text-xs h-8 px-3">
            {importing ? <RefreshCw className="w-3 h-3 mr-1 animate-spin" /> : <Upload className="w-3 h-3 mr-1" />}
            {importing ? 'Loading…' : 'Upload Manifold'}
          </Button>
          <input ref={fileRef} type="file" accept=".json" className="hidden" onChange={handleFileChange} />
        </div>
        {importStatus && (
          <div className={`mt-2 flex items-center gap-2 text-xs ${importStatus.ok ? 'text-green-400' : 'text-red-400'}`}>
            {importStatus.ok ? <CheckCircle2 className="w-3 h-3" /> : <AlertTriangle className="w-3 h-3" />}
            {importStatus.msg}
          </div>
        )}
      </Section>

      {/* What has it learned */}
      {topContexts.length > 0 && (
        <Section title="Top Learned Contexts (5-grams)" icon={TrendingUp} color="violet">
          <p className="text-xs text-slate-500 mb-2">
            The linguist stores character-level 5-gram → next-character frequency tables.
            High-count entries = strongly reinforced patterns from KB text + your queries.
          </p>
          <div className="space-y-1 max-h-52 overflow-y-auto pr-1">
            {topContexts.map(({ ctx, total, topChar, topCount }, i) => (
              <div key={i} className="flex items-center gap-2 text-xs">
                <span className="font-mono text-violet-300 w-14 flex-shrink-0">"{ctx}"</span>
                <span className="text-slate-500">→</span>
                <span className="font-mono text-cyan-300 w-4">{topChar === ' ' ? '␣' : topChar}</span>
                <div className="flex-1 h-1 bg-slate-700 rounded-full overflow-hidden">
                  <div
                    className="h-full bg-violet-500/60 rounded-full"
                    style={{ width: `${Math.min(100, (topCount / (topContexts[0]?.topCount || 1)) * 100)}%` }}
                  />
                </div>
                <span className="text-slate-600 w-8 text-right">{total}</span>
              </div>
            ))}
          </div>
        </Section>
      )}

      {/* Learning log */}
      {log.length > 0 && (
        <Section title="Learning Log" icon={BookOpen} color="green">
          <p className="text-xs text-slate-500 mb-2">
            Each entry = a text source the linguist ingested and how many n-gram samples it extracted.
          </p>
          <div className="space-y-1 max-h-48 overflow-y-auto pr-1">
            {log.slice().reverse().map((entry, i) => (
              <div key={i} className="flex items-start gap-2 text-xs border-b border-slate-700/30 pb-1 last:border-0">
                <span className={`mt-0.5 w-2 h-2 rounded-full flex-shrink-0 ${
                  entry.type === 'kb' ? 'bg-cyan-400' :
                  entry.type === 'query' ? 'bg-green-400' :
                  entry.type === 'feedback' ? 'bg-amber-400' : 'bg-slate-400'
                }`} />
                <div className="flex-1 min-w-0">
                  <p className="text-slate-300 truncate">{entry.summary}</p>
                  <p className="text-slate-600">{entry.samples.toLocaleString()} samples · {new Date(entry.t).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' })}</p>
                </div>
              </div>
            ))}
          </div>
        </Section>
      )}

      {/* Gap report */}
      {gapReport && gapReport.gaps.length > 0 && (
        <Section title="Semantic Gaps" icon={AlertTriangle} color="amber">
          <p className="text-xs text-slate-500 mb-2">
            Words from your queries that had no KB vector anchor — the linguist had to use geometric hash fallback for these.
            Adding these to the KB would improve deterministic accuracy.
          </p>
          <div className="flex flex-wrap gap-1">
            {gapReport.gaps.slice(0, 30).map((w, i) => (
              <span key={i} className="px-1.5 py-0.5 bg-amber-500/10 border border-amber-500/20 rounded text-xs text-amber-300 font-mono">{w}</span>
            ))}
          </div>
          {gapReport.gaps.length > 30 && (
            <p className="text-xs text-slate-600 mt-1">+{gapReport.gaps.length - 30} more</p>
          )}
        </Section>
      )}

      {/* Why / How explanation */}
      <Section title="Why & How It Learns" icon={Brain} color="slate">
        <div className="space-y-2 text-xs text-slate-400">
          <p><span className="text-cyan-300 font-medium">Initial training:</span> On engine load, all KB lexicon strings (~150k chars) are fed into a 5-gram character manifold — mapping 5-char contexts to the most likely next character.</p>
          <p><span className="text-green-300 font-medium">Incremental learning:</span> Every query + response pair adds ~150 n-gram samples. The manifold grows denser with each session.</p>
          <p><span className="text-violet-300 font-medium">Feedback reinforcement:</span> Thumbs Up boosts the lexicon of matched KB entries (+200 samples). Thumbs Down adds nothing.</p>
          <p><span className="text-amber-300 font-medium">Elaboration scoring:</span> Candidate words are scored by Golay XOR bridge stability — low Hamming weight between a candidate's KB vector and the answer vector = high geometric stability = preferred word.</p>
          <p><span className="text-pink-300 font-medium">Fluency limit:</span> The linguist cannot invent new knowledge — it can only recombine patterns from text it has seen. To become more accurate, the underlying KB needs richer entries.</p>
        </div>
      </Section>
    </div>
  );
}
