/**
 * UBP FRAME OF MIND (FOM) SYSTEM — JS Port of ubp_fom_system.py v4.3.1
 *
 * Manages cognitive bias frames that weight different geometric domains.
 * Each FOM has a base NRCI and per-category/per-entry weight overrides.
 * The active FOM multiplies the NRCI component of every scored entry.
 */

const FOM_INDEX_URL = 'https://raw.githubusercontent.com/DigitalEuan/UBP_Repo/main/core_studio_v4.0/system_kb/ubp_fom_index.json';

// ── Default FOM frames (used if GitHub fetch fails) ───────────────────────────
// FIXED v7.2: base_nrci=1.0 (neutral multiplier) for DEFAULT to prevent "Zombie States"
// where unclassified entries drop to 0.25 effective NRCI (below Noise Floor).
// Category weights are now multiplicative boosts (>1 = boost, <1 = suppress).
const DEFAULT_FRAMES = [
  {
    frame_id: 'FOM_DEFAULT',
    description: 'Balanced Standard Bias — neutral multiplier, no domain preference',
    base_nrci: 1.0, // FIXED: neutral, not 0.5 noise-floor
    weights: {},
    category_weights: {},
  },
  {
    frame_id: 'FOM_SCIENCE',
    description: 'Physical Science Bias — boosts SUBSTANCE, MECHANISM, QUANTITY',
    base_nrci: 1.0,
    weights: {},
    category_weights: { SUBSTANCE: 1.2, MECHANISM: 1.2, QUANTITY: 1.1, ALGORITHM: 1.0 },
  },
  {
    frame_id: 'FOM_PHILOSOPHY',
    description: 'Philosophical Bias — boosts MEANING & IMPERATIVE entries',
    base_nrci: 1.0,
    weights: {},
    category_weights: { MEANING: 1.2, IMPERATIVE: 1.1, ORGANISM: 1.05 },
  },
];

class FrameOfMind {
  constructor({ frame_id, description = '', base_nrci = 1.0, weights = {}, category_weights = {} }) {
    this.frameId = frame_id;
    this.description = description;
    this.baseNrci = parseFloat(base_nrci) || 1.0;
    // weights: ubp_id → nrci override
    this.weights = {};
    for (const [k, v] of Object.entries(weights)) {
      this.weights[k] = parseRational(v);
    }
    // category_weights: domain → nrci override
    this.categoryWeights = {};
    for (const [k, v] of Object.entries(category_weights)) {
      this.categoryWeights[k] = parseRational(v);
    }
  }

  /**
   * Get effective NRCI multiplier for a given entry.
   * FIXED v7.2: return 1.0 (neutral) when no specific bias is found,
   * preventing accidental "Zombie" transitions below 0.70 threshold.
   */
  getWeight(ubpId, domain) {
    if (ubpId && this.weights[ubpId] !== undefined) return this.weights[ubpId];
    if (domain && this.categoryWeights[domain] !== undefined) return this.categoryWeights[domain];
    return 1.0; // neutral — do not suppress unclassified entries
  }
}

function parseRational(v) {
  if (typeof v === 'number') return v;
  if (typeof v === 'string' && v.includes('/')) {
    const [n, d] = v.split('/').map(Number);
    return d > 0 ? n / d : 0.5;
  }
  return parseFloat(v) || 0.5;
}

export class FOMManager {
  constructor() {
    this.frames = {};
    this.activeFrameId = 'FOM_DEFAULT';
    this._loadDefaults();
  }

  _loadDefaults() {
    for (const f of DEFAULT_FRAMES) {
      this.frames[f.frame_id] = new FrameOfMind(f);
    }
  }

  async loadFromGitHub() {
    try {
      const res = await fetch(FOM_INDEX_URL);
      if (!res.ok) return; // silently use defaults
      const data = await res.json();
      const list = Array.isArray(data) ? data : Object.values(data);
      for (const f of list) {
        if (f.frame_id) this.frames[f.frame_id] = new FrameOfMind(f);
      }
      console.log(`[FOM] Loaded ${Object.keys(this.frames).length} frames from GitHub`);
    } catch {
      // use defaults silently
    }
  }

  switchFrame(frameId) {
    if (this.frames[frameId]) { this.activeFrameId = frameId; return true; }
    return false;
  }

  getActiveFrame() {
    return this.frames[this.activeFrameId] || this.frames['FOM_DEFAULT'];
  }

  /** NRCI multiplier for a KB entry — called during scoring */
  getWeight(ubpId, domain) {
    return this.getActiveFrame().getWeight(ubpId, domain);
  }

  listFrames() {
    return Object.values(this.frames).map(f => ({
      id: f.frameId,
      description: f.description,
      active: f.frameId === this.activeFrameId,
    }));
  }
}
