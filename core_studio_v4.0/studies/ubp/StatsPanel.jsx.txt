import React from 'react';
import { motion } from 'framer-motion';
import { Database, Layers, MessageSquare, ThumbsUp, Cpu, Binary, Zap } from 'lucide-react';
import { DOMAIN_COLORS } from './UBPLexicon';

function StatCard({ icon: Icon, label, value, color = 'cyan' }) {
  const colorClasses = {
    cyan: 'from-cyan-500/20 to-blue-600/20 border-cyan-500/30 text-cyan-400',
    violet: 'from-violet-500/20 to-purple-600/20 border-violet-500/30 text-violet-400',
    green: 'from-green-500/20 to-emerald-600/20 border-green-500/30 text-green-400',
    amber: 'from-amber-500/20 to-orange-600/20 border-amber-500/30 text-amber-400',
  };
  
  return (
    <div className={`p-3 rounded-xl bg-gradient-to-br ${colorClasses[color]} border`}>
      <div className="flex items-center gap-2 mb-1">
        <Icon className="w-4 h-4 opacity-70" />
        <span className="text-xs opacity-70">{label}</span>
      </div>
      <div className="text-xl font-bold">{value}</div>
    </div>
  );
}

export function StatsPanel({ stats, lexiconStats }) {
  if (!stats) return null;
  
  return (
    <div className="space-y-4">
      {/* Main stats grid */}
      <div className="grid grid-cols-2 gap-3">
        <StatCard 
          icon={Database} 
          label="Memories" 
          value={stats.totalMemories || 0}
          color="cyan"
        />
        <StatCard 
          icon={Binary} 
          label="Name Anchors" 
          value={stats.lexiconClusters?.toLocaleString() || 0}
          color="violet"
        />
        <StatCard 
          icon={MessageSquare} 
          label="Conversations" 
          value={stats.contextTurns || 0}
          color="green"
        />
        <StatCard 
          icon={ThumbsUp} 
          label="Feedback" 
          value={stats.feedbackCount || 0}
          color="amber"
        />
      </div>
      
      {/* Domain breakdown */}
      {stats.domains && Object.keys(stats.domains).length > 0 && (
        <div className="p-3 bg-slate-800/30 rounded-xl border border-slate-700/50">
          <h4 className="text-xs font-medium text-slate-400 mb-3 flex items-center gap-2">
            <Layers className="w-3 h-3" />
            Memory Domains
          </h4>
          <div className="space-y-2">
            {Object.entries(stats.domains).map(([domain, count]) => {
              const colors = DOMAIN_COLORS[domain] || DOMAIN_COLORS.MEANING;
              const maxCount = Math.max(...Object.values(stats.domains));

                <div key={domain} className="space-y-1">
                  <div className="flex items-center justify-between text-xs">
                    <span className={colors.text}>{domain}</span>
                    <span className="text-slate-500">{count}</span>
                  </div>
                  <div className="h-1.5 bg-slate-700/50 rounded-full overflow-hidden">
                    <motion.div
                      initial={{ width: 0 }}
                      animate={{ width: `${percentage}%` }}
                      transition={{ duration: 0.5, delay: 0.1 }}
                      className={`h-full rounded-full ${colors.bg.replace('/20', '/60')}`}
                    />
                  </div>
                </div>
              );
            })}
          </div>
        </div>
      )}
      
      {/* Lexicon stats */}

            </div>
            <div className="flex justify-between">
              <span className="text-slate-500">Max Cluster</span>
              <span className="text-slate-300">{lexiconStats.maxClusterSize}</span>
            </div>
            <div className="flex justify-between">
              <span className="text-slate-500">Clusters</span>
              <span className="text-slate-300">{lexiconStats.clusterCount?.toLocaleString()}</span>
            </div>
          </div>
        </div>
      )}
      
      {/* Linguist (MoE Cortex) status */}
      {stats.linguist && (
        <div className="p-3 bg-slate-800/30 rounded-xl border border-slate-700/50">

            <MessageSquare className="w-3 h-3 text-pink-400" />
            MoE Linguist

            <span className="text-xs text-slate-500">{stats.linguist.trained ? 'Active' : 'Training…'}</span>
          </h4>
          <div className="grid grid-cols-2 gap-2 text-xs">
            <div className="flex justify-between">
              <span className="text-slate-500">Contexts</span>
              <span className="text-pink-300">{stats.linguist.contexts?.toLocaleString()}</span>
            </div>
            <div className="flex justify-between">
              <span className="text-slate-500">Vocab</span>
              <span className="text-pink-300">{stats.linguist.vocab}</span>
            </div>
            <div className="flex justify-between col-span-2">
              <span className="text-slate-500">Samples Learned</span>

        <h4 className="text-xs font-medium text-slate-400 mb-2 flex items-center gap-2">
          <Zap className="w-3 h-3 text-amber-400" />
          NoiseCore v4.0 + Swarm v1.0 · Phase-Lock Edition

          <span className="text-xs text-slate-500">SV mode</span>
        </h4>
        <div className="text-xs text-slate-500 space-y-1">
          <p className="text-slate-400">Golay [24,12,8] + Leech Λ₂₄ + BW-256 + Monster (26 sporadics) + NativeMath (v28 Oracle port)</p>
          <p>ALU: <span className="text-amber-300 font-mono">factorial · fibonacci · gcd · modpow · isPrime · lcm · extGcd</span></p>
          <p>Swarm: <span className="text-amber-300 font-mono">catalan · euler phi · factorise · CRT · divisors</span></p>
          <p>Swarm stats: <span className="text-amber-300 font-mono">median · stddev · cross product · quadratic · modinv</span></p>
