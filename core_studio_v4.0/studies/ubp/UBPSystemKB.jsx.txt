/**
 * UBP SYSTEM KNOWLEDGE BASE LOADER
 * Loads system KB (812 entries) + language KB v3 (169 semantic operators)
 * from GitHub. Falls back gracefully on any fetch failure.
 */

const KB_URLS = [
  // Primary system KB — physical entities, particles, laws, constants
  'https://raw.githubusercontent.com/DigitalEuan/UBP_Repo/main/core_studio_v4.0/system_kb/ubp_system_kb.json',
  // Beliefs KB — laws, axioms, imperatives
  'https://raw.githubusercontent.com/DigitalEuan/UBP_Repo/main/core_studio_v4.0/system_kb/ubp_beliefs_kb.json',
  // Hash memory KB
  'https://raw.githubusercontent.com/DigitalEuan/UBP_Repo/main/core_studio_v4.0/system_kb/hash_memory_kb.json',
  // Combined Language KB v4 — all semantic operators (replaces v3 + local extension)
  'https://raw.githubusercontent.com/DigitalEuan/UBP_Repo/main/core_studio_v4.0/core/ubp_lang_kb_combined_v4.json',
];

// Note: lang KB now loaded from GitHub (ubp_lang_kb_combined_v4.json) — no local extension needed

/**
 * Hydrate v9.9 columnar format { _fields, _null_token, entries: { fp: [v0,v1,...] } }
 * into the flat object format: { fp: { ubp_id, lexicon, tags, vector, nrci_val, ... } }
 * Falls back to identity for legacy object-format KBs.
 */
function hydrateKB(data) {
  if (!data || typeof data !== 'object') return {};
  // v9.9 columnar format
  if (data._fields && data.entries) {
    const fields = data._fields;
    const result = {};
    for (const [fp, arr] of Object.entries(data.entries)) {
      const entry = {};
      for (let i = 0; i < fields.length; i++) entry[fields[i]] = arr[i];
      // Ensure vector is a clean array of 0/1 integers
      if (Array.isArray(entry.vector)) {
        entry.vector = entry.vector.map(b => Number(b));
      }
      result[fp] = entry;
    }
    return result;
  }
  // Legacy format — already a flat object map
  return data;
}

export async function loadSystemKB() {
  try {
    console.log('[SystemKB] Fetching KB files...');
    const responses = await Promise.allSettled(KB_URLS.map(url => fetch(url)));
    const merged = {};
    for (let i = 0; i < responses.length; i++) {
      const r = responses[i];
      if (r.status === 'rejected' || !r.value?.ok) {
        console.warn(`[SystemKB] Skipping ${KB_URLS[i]}: ${r.reason || r.value?.status}`);
        continue;
      }
      const data = await r.value.json();
      Object.assign(merged, hydrateKB(data));
    }
    const total = Object.keys(merged).length;
    console.log(`[SystemKB] Loaded ${total} KB entries total`);
    return merged;
  } catch (error) {
    console.error('[SystemKB] Load error:', error);
    throw error;
  }
}
