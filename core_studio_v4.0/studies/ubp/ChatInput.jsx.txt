import React, { useState, useRef, useEffect } from 'react';
import { motion } from 'framer-motion';
import { Send, Loader2, Sparkles } from 'lucide-react';
import { Button } from '@/components/ui/button';
import { Textarea } from '@/components/ui/textarea';

export function ChatInput({ onSend, isLoading, placeholder, hideSuggestions }) {
  const [input, setInput] = useState('');
  const textareaRef = useRef(null);
  
  const handleSubmit = (e) => {
    e.preventDefault();
    if (input.trim() && !isLoading) {
      onSend(input.trim());
      setInput('');
    }
  };
  
  const handleKeyDown = (e) => {
    if (e.key === 'Enter' && !e.shiftKey) {
      e.preventDefault();
      handleSubmit(e);
    }
  };
  
  // Auto-resize textarea
  useEffect(() => {
    if (textareaRef.current) {
      textareaRef.current.style.height = 'auto';
      textareaRef.current.style.height = Math.min(textareaRef.current.scrollHeight, 120) + 'px';
    }
  }, [input]);
  
  const suggestions = [
    "What is the Higgs boson?",
    "euler phi 12",
    "factorial 12",
    "factorise 360",
    "catalan 5",
    "median 3 1 4 1 5 9 2 6",
    "cross product <1,0,0> <0,1,0>",
    "lattice attraction",
  ];
  
  return (
    <div className="space-y-3">
      {/* Quick suggestions when empty — hidden on mobile via prop */}
      {!input && !hideSuggestions && (
        <motion.div 
          initial={{ opacity: 0 }}
          animate={{ opacity: 1 }}
          className="flex flex-wrap gap-2"
        >
          {suggestions.map((suggestion, i) => (
            <motion.button
              key={i}
              initial={{ opacity: 0, y: 10 }}
              animate={{ opacity: 1, y: 0 }}
              transition={{ delay: i * 0.05 }}
              onClick={() => setInput(suggestion)}
              className="px-3 py-1.5 text-xs bg-slate-800/50 hover:bg-slate-700/50 
                         border border-slate-700/50 hover:border-slate-600/50
                         rounded-full text-slate-400 hover:text-slate-300
                         transition-all duration-200"
            >
              <Sparkles className="w-3 h-3 inline mr-1.5 text-cyan-500" />
              {suggestion}
            </motion.button>
          ))}
        </motion.div>
      )}

      {/* Input form */}
      <form onSubmit={handleSubmit} className="relative">
        <div className="relative flex items-end gap-2 p-2 bg-slate-800/50 rounded-2xl border border-slate-700/50 focus-within:border-cyan-500/50 transition-colors">
          <Textarea
            ref={textareaRef}
            value={input}
            onChange={(e) => setInput(e.target.value)}
            onKeyDown={handleKeyDown}
            placeholder={placeholder || "Ask about UBP, Golay codes, geometric reasoning..."}
            disabled={isLoading}
            className="flex-1 min-h-[44px] max-h-[120px] resize-none border-0 bg-transparent 
                       focus:ring-0 focus-visible:ring-0 focus-visible:ring-offset-0
                       text-slate-100 placeholder:text-slate-500 text-sm p-2"
            rows={1}
          />
          
          <Button
            type="submit"
            size="icon"
            disabled={!input.trim() || isLoading}
            className="flex-shrink-0 w-10 h-10 rounded-xl bg-gradient-to-r from-cyan-500 to-blue-600 
                       hover:from-cyan-400 hover:to-blue-500 
                       disabled:from-slate-600 disabled:to-slate-700 disabled:opacity-50
                       transition-all duration-200"
          >
            {isLoading ? (
              <Loader2 className="w-4 h-4 animate-spin" />
            ) : (
              <Send className="w-4 h-4" />
            )}
          </Button>
        </div>
        
        <div className="flex items-center justify-between mt-2 px-2 text-xs text-slate-600">
          <span>Press Enter to send, Shift+Enter for new line</span>
          <span>{input.length} chars</span>
        </div>
      </form>
    </div>
  );
}
