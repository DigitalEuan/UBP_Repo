/**
 * UBP GEOMETRIC VISUALISER
 * Spatial map of all KB entries as nodes in 3D Golay space.
 */
import { useEffect, useRef, useMemo } from 'react';
import * as THREE from 'three';
import { Sparkles } from 'lucide-react';

const DOMAIN_HEX = {
  SUBSTANCE:  0x22d3ee,
  MECHANISM:  0x60a5fa,
  ORGANISM:   0x34d399,
  ALGORITHM:  0xa78bfa,
  QUANTITY:   0xfbbf24,
  IMPERATIVE: 0xf87171,
  ENTROPY:    0xfb923c,
  MEANING:    0xe879f9,
};

function domainColor(domain) {
  return DOMAIN_HEX[domain] ?? 0x94a3b8;
}

function uidJitter(uid) {
  let h = 5381;
  for (let i = 0; i < uid.length; i++) { h = ((h << 5) + h) ^ uid.charCodeAt(i); h = h >>> 0; }
  return [((h & 0xFF) / 255 - 0.5) * 1.2, (((h >> 8) & 0xFF) / 255 - 0.5) * 1.2, (((h >> 16) & 0xFF) / 255 - 0.5) * 1.2];
}

function vecToXYZ(vec, uid) {
  const x = (vec.slice(0, 8).reduce((s, b) => s + b, 0) / 8) * 6 - 3;
  const y = (vec.slice(8, 16).reduce((s, b) => s + b, 0) / 8) * 6 - 3;
  const z = (vec.slice(16, 24).reduce((s, b) => s + b, 0) / 8) * 6 - 3;
  const j = uidJitter(uid || '');
  return [x + j[0], y + j[1], z + j[2]];
}

export default function GeometricVisualizer({ entries = [], activatedUids = [], compact = false }) {
  const mountRef = useRef(null);
  const frameRef = useRef(null);
  const activatedSet = useMemo(() => new Set(activatedUids), [activatedUids]);

  useEffect(() => {
    if (!mountRef.current || !entries.length) return;
    const W = mountRef.current.clientWidth || 300;
    const H = mountRef.current.clientHeight || 200;

    const scene = new THREE.Scene();
    scene.background = new THREE.Color(0x0a0f1a);

    const camera = new THREE.PerspectiveCamera(60, W / H, 0.1, 100);
    camera.position.set(0, 0, 12);

    const renderer = new THREE.WebGLRenderer({ antialias: true });
    renderer.setSize(W, H);
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    mountRef.current.appendChild(renderer.domElement);

    scene.add(new THREE.AmbientLight(0xffffff, 0.6));
    const pLight = new THREE.PointLight(0x22d3ee, 2, 20);
    pLight.position.set(0, 5, 5);
    scene.add(pLight);

    const positions = [], colors = [];
    const nodeData = [];

    for (const entry of entries) {
      const vec = entry.vector;
      if (!vec || vec.length !== 24) continue;
      const [x, y, z] = vecToXYZ(vec, entry.uid || '');
      positions.push(x, y, z);
      const col = new THREE.Color(domainColor(entry.domain));
      colors.push(col.r, col.g, col.b);
      nodeData.push({ uid: entry.uid, x, y, z, nrci: entry.nrci || 0.5 });
    }

    const geo = new THREE.BufferGeometry();
    geo.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3));
    geo.setAttribute('color', new THREE.Float32BufferAttribute(colors, 3));
    const mat = new THREE.PointsMaterial({ size: 0.07, vertexColors: true, transparent: true, opacity: 0.65, sizeAttenuation: true });
    const pivot = new THREE.Group();
    pivot.add(new THREE.Points(geo, mat));
    scene.add(pivot);

    const sphereGeo = new THREE.SphereGeometry(0.1, 8, 8);
    const activatedNodes = nodeData.filter(n => activatedSet.has(n.uid));
    activatedNodes.slice(0, 10).forEach((n, i) => {
      const t = i / Math.max(activatedNodes.length - 1, 1);
      const col = new THREE.Color().lerpColors(new THREE.Color(0x22d3ee), new THREE.Color(0xe879f9), t);
      const sphere = new THREE.Mesh(sphereGeo, new THREE.MeshBasicMaterial({ color: col, transparent: true, opacity: 0.9 }));
      sphere.position.set(n.x, n.y, n.z);
      sphere.scale.setScalar(0.5 + n.nrci * 1.5);
      pivot.add(sphere);
    });

    // Lines only between activated nodes
    if (activatedNodes.length > 1) {
      const actLinePos = [];
      for (let i = 0; i < activatedNodes.length - 1; i++) {
        const a = activatedNodes[i], b = activatedNodes[i + 1];
        actLinePos.push(a.x, a.y, a.z, b.x, b.y, b.z);
      }
      const actLineGeo = new THREE.BufferGeometry();
      actLineGeo.setAttribute('position', new THREE.Float32BufferAttribute(actLinePos, 3));
      pivot.add(new THREE.LineSegments(actLineGeo, new THREE.LineBasicMaterial({ color: 0xffffff, transparent: true, opacity: 0.5 })));
    }

    scene.add(new THREE.GridHelper(12, 12, 0x1e293b, 0x1e293b));

    let t = 0;
    const animate = () => {
      frameRef.current = requestAnimationFrame(animate);
      t += 0.003;
      pivot.rotation.y = t;
      pivot.rotation.x = Math.sin(t * 0.3) * 0.12;
      renderer.render(scene, camera);
    };
    animate();

    const onResize = () => {
      if (!mountRef.current) return;
      const w = mountRef.current.clientWidth, h = mountRef.current.clientHeight;
      camera.aspect = w / h; camera.updateProjectionMatrix(); renderer.setSize(w, h);
    };
    window.addEventListener('resize', onResize);

    return () => {
      cancelAnimationFrame(frameRef.current);
      window.removeEventListener('resize', onResize);
      renderer.dispose();
      if (mountRef.current?.contains(renderer.domElement)) mountRef.current.removeChild(renderer.domElement);
    };
  }, [entries, activatedSet]);

  return (
    <div className="flex flex-col h-full">
      {!compact && (
        <div className="flex items-center gap-3 mb-2 flex-wrap">
          {Object.entries(DOMAIN_HEX).map(([d, hex]) => (
            <span key={d} className="flex items-center gap-1 text-xs text-slate-500">
              <span className="w-2 h-2 rounded-full inline-block" style={{ background: `#${hex.toString(16).padStart(6,'0')}` }} />
              {d}
            </span>
          ))}
        </div>
      )}
      <div ref={mountRef} className="flex-1 rounded-xl overflow-hidden" style={{ minHeight: compact ? 120 : 280 }} />
      {!compact && activatedUids.length > 0 && (
        <div className="mt-2 text-xs text-cyan-400 text-center">
          ● {activatedUids.slice(0, 5).join(' · ')}
        </div>
      )}
    </div>
  );
}
