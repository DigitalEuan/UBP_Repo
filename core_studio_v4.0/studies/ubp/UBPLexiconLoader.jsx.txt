/**
 * UBP LEXICON V2 LOADER (WITH DEFINITIONS)
 * Fetches enhanced lexicon from GitHub — optional, fails gracefully
 */

const LEXICON_URL = 'https://raw.githubusercontent.com/DigitalEuan/UBP_Repo/main/core_studio_v4.0/core/ubp_lexicon_v2_defs.json';

/**
 * Load lexicon with definitions from GitHub.
 * Returns null if unavailable instead of throwing.
 */
export async function loadLexicon() {
  try {
    console.log('[Lexicon] Fetching from GitHub...');
    const response = await fetch(LEXICON_URL);
    
    if (!response.ok) {
      console.warn(`[Lexicon] Not available (${response.status}) — running without lexicon`);
      return null;
    }
    
    const data = await response.json();
    
    if (data.v !== 2 || !data.c) {
      console.warn('[Lexicon] Invalid format — running without lexicon');
      return null;
    }
    
    console.log(`[Lexicon] Loaded ${Object.keys(data.c).length} word clusters`);
    return data;
  } catch (error) {
    console.warn('[Lexicon] Load failed (non-fatal):', error.message);
    return null;
  }
}

/**
 * Lookup word in lexicon
 */
export function lookupWord(lexicon, word) {
  if (!lexicon || !lexicon.c) return null;
  
  const wordLower = word.toLowerCase().trim();
  
  // Search through all clusters
  for (const [hex, cluster] of Object.entries(lexicon.c)) {
    const [clusterWord, domain, count, definition] = cluster;
    
    if (clusterWord.toLowerCase() === wordLower) {
      return {
        word: clusterWord,
        hex,
        domain,
        count: parseInt(count),
        definition: definition || 'No definition available',
        vector: hexToBits(hex)
      };
    }
  }
  
  return null;
}

/**
 * Get all words in a cluster (same codeword)
 */
export function getCluster(lexicon, hex) {
  if (!lexicon || !lexicon.c || !lexicon.c[hex]) return null;
  
  const [word, domain, count, definition] = lexicon.c[hex];
  
  return {
    hex,
    representative: word,
    domain,
    count: parseInt(count),
    definition: definition || 'No definition available',
    vector: hexToBits(hex)
  };
}

/**
 * Tokenize text into Golay clusters
 */
export function tokenize(lexicon, text) {
  if (!lexicon || !lexicon.c) return [];
  
  const words = text.toLowerCase()
    .replace(/[^a-z0-9\s'-]/g, ' ')
    .split(/\s+/)
    .filter(w => w.length > 0);
  
  const tokens = [];
  
  for (const word of words) {
    const result = lookupWord(lexicon, word);
    if (result) {
      tokens.push(result);
    }
  }
  
  return tokens;
}

/**
 * Get lexicon statistics
 */
export function getLexiconStats(lexicon) {
  if (!lexicon || !lexicon.c) {
    return { loaded: false };
  }
  
  const clusters = Object.values(lexicon.c);
  const totalWords = clusters.reduce((sum, c) => sum + parseInt(c[2] || 0), 0);
  const clusterSizes = clusters.map(c => parseInt(c[2] || 0));
  
  return {
    loaded: true,
    version: lexicon.v,
    clusterCount: clusters.length,
    totalWords,
    avgClusterSize: totalWords / clusters.length,
    maxClusterSize: Math.max(...clusterSizes),
    minClusterSize: Math.min(...clusterSizes)
  };
}

/**
 * Find nearest codewords by Hamming distance
 */
export function findNearest(lexicon, targetVector, maxDistance = 8, limit = 20) {
  if (!lexicon || !lexicon.c) return [];
  
  const results = [];
  
  for (const [hex, cluster] of Object.entries(lexicon.c)) {
    const vector = hexToBits(hex);
    const distance = hammingDistance(targetVector, vector);
    
    if (distance <= maxDistance) {
      results.push({
        hex,
        word: cluster[0],
        domain: cluster[1],
        distance,
        vector
      });
    }
  }
  
  results.sort((a, b) => a.distance - b.distance);
  return results.slice(0, limit);
}

/**
 * Convert hex string to 24-bit array
 */
function hexToBits(hex) {
  const n = parseInt(hex, 16);
  const bits = [];
  for (let i = 23; i >= 0; i--) {
    bits.push((n >> i) & 1);
  }
  return bits;
}

/**
 * Hamming distance helper
 */
function hammingDistance(v1, v2) {
  let distance = 0;
  for (let i = 0; i < Math.min(v1.length, v2.length); i++) {
    if (v1[i] !== v2[i]) distance++;
  }
  return distance;
}
