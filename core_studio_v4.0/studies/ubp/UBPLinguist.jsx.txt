/**
 * UBP LINGUIST — N-Gram MoE Cortex (JS port of ubp_moe_cortex_v2)
 *
 * Trains a 5-gram character manifold on KB lexicon text.
 * Uses Golay XOR bridge stability to score word candidates (MoE research loop).
 * Learns incrementally from every user query + engine response.
 *
 * Gray-code → Hex color: vector[0..7] = R, [8..15] = G, [16..23] = B
 * Not a physical color — a geometric fingerprint color for recall augmentation.
 */

const ORDER = 5;

// ── Gray-code vector → CSS hex color ─────────────────────────────────────────
export function grayCodeToHex(vector24) {
  if (!vector24 || vector24.length !== 24) return '#888888';
  const r = parseInt(vector24.slice(0, 8).join(''), 2);
  const g = parseInt(vector24.slice(8, 16).join(''), 2);
  const b = parseInt(vector24.slice(16, 24).join(''), 2);
  return '#' + [r, g, b].map(v => v.toString(16).padStart(2, '0')).join('');
}

// ── XOR bridge stability score using strict UBP NRCI formula ─────────────────
// FIXED v7.2: Use Y_LEECH NRCI formula instead of linear (24-hw)/24.
// Linear formula treats Octad(hw=8) and hw=7 as nearly identical;
// NRCI formula correctly respects the Law of Ontological Friction.
const _Y_LEECH_LINGUIST = 1 / (Math.PI + 2 / Math.PI); // ≈ 0.2647
function xorBridgeScore(goalVec, wordVec) {
  if (!goalVec || !wordVec || goalVec.length !== 24) return 0;
  let hw = 0;
  for (let i = 0; i < goalVec.length; i++) hw += goalVec[i] ^ wordVec[i];
  const tax = hw * _Y_LEECH_LINGUIST + hw / 8;
  return 10 / (10 + tax); // NRCI of the XOR bridge
}

// ── Linguist class ────────────────────────────────────────────────────────────
export class UBPLinguist {
  constructor() {
    this.manifold = new Map(); // 5-char context -> Map<char, count>
    this.vocab = new Set();
    this.trained = false;
    this._totalLearned = 0;
    this._log = [];          // learning audit log
    this._gaps = new Set();  // words with no KB vector anchor
  }

  // Build initial manifold from all KB lexicon strings (async, batched)
  async trainOnKB(kbData) {
    const docs = [];
    for (const entry of Object.values(kbData)) {
      if (entry.lexicon) docs.push(entry.lexicon.toLowerCase());
    }
    const text = docs.join(' ').replace(/[^a-z0-9 :_\-\.]/g, '').slice(0, 150000);
    console.log(`[Linguist] Training on ${text.length} chars from ${docs.length} KB entries...`);

    for (const c of text) this.vocab.add(c);

    const BATCH = 50000;
    const TOTAL = Math.min(300000, text.length * 2);
    let done = 0;
    while (done < TOTAL) {
      const end = Math.min(done + BATCH, TOTAL);
      for (let i = done; i < end; i++) {
        const idx = Math.floor(Math.random() * (text.length - ORDER - 1));
        const ctx = text.slice(idx, idx + ORDER);
        const tar = text[idx + ORDER];
        if (!tar) continue;
        if (!this.manifold.has(ctx)) this.manifold.set(ctx, new Map());
        const m = this.manifold.get(ctx);
        m.set(tar, (m.get(tar) || 0.01) + 1);
      }
      done = end;
      if (done < TOTAL) await new Promise(r => setTimeout(r, 0));
    }
    this._totalLearned = TOTAL;
    this.trained = true;
    this._log.push({ type: 'kb', summary: `KB initial training: ${docs.length} entries`, samples: TOTAL, t: Date.now() });
    console.log(`[Linguist] Trained: ${this.manifold.size} contexts · vocab ${this.vocab.size}`);
  }

  // Incremental learning from a text string (called after each interaction)
  learnFrom(text, iterations = 400, logSummary = null) {
    if (!text || text.length < ORDER + 1) return;
    const cleaned = text.toLowerCase().replace(/[^a-z0-9 :_\-\.]/g, '').slice(0, 2000);
    for (const c of cleaned) this.vocab.add(c);
    for (let i = 0; i < iterations; i++) {
      const idx = Math.floor(Math.random() * (cleaned.length - ORDER - 1));
      if (idx < 0) break;
      const ctx = cleaned.slice(idx, idx + ORDER);
      const tar = cleaned[idx + ORDER];
      if (!tar) continue;
      if (!this.manifold.has(ctx)) this.manifold.set(ctx, new Map());
      const m = this.manifold.get(ctx);
      m.set(tar, (m.get(tar) || 0) + 1);
    }
    this._totalLearned += iterations;
    if (logSummary) {
      this._log.push({ type: 'query', summary: logSummary, samples: iterations, t: Date.now() });
      if (this._log.length > 200) this._log = this._log.slice(-200);
    }
  }

  // Track lexical gaps (words with no KB vector)
  trackGap(word) {
    this._gaps.add(word.toLowerCase());
  }

  // Reinforce associations from feedback (boost context counts for known UIDs)
  reinforceFromFeedback(uids, uidToEntry, positive) {
    if (!uidToEntry) return;
    for (const uid of uids) {
      const entry = uidToEntry.get(uid);
      if (!entry?.lexicon) continue;
      const samples = positive ? 200 : 0;
      this.learnFrom(entry.lexicon, samples);
      if (positive) this._log.push({ type: 'feedback', summary: `👍 Reinforced: ${uid}`, samples, t: Date.now() });
    }
  }

  // Sample a word from the manifold given a context string
  _sampleWord(seed, maxLen = 14) {
    let word = '';
    let ctx = seed.slice(-ORDER).padStart(ORDER, ' ');
    for (let i = 0; i < maxLen; i++) {
      const m = this.manifold.get(ctx);
      if (!m) break;
      // Temperature sampling (T=0.6 for focused output)
      const entries = [...m.entries()];
      const weights = entries.map(([, c]) => Math.pow(c, 1 / 0.6));
      const total = weights.reduce((s, w) => s + w, 0);
      let r = Math.random() * total;
      let chosen = entries[0][0];
      for (let j = 0; j < entries.length; j++) {
        r -= weights[j];
        if (r <= 0) { chosen = entries[j][0]; break; }
      }
      if (chosen === ' ') break;
      word += chosen;
      ctx = (ctx + chosen).slice(-ORDER);
    }
    return word;
  }

  /**
   * MoE elaboration: given a goal vector + query context, build a short
   * geometrically-scored elaboration sentence (max 10 words).
   * Words are proposed by the linguist and scored via Golay XOR bridge stability.
   */
  elaborate(goalVec, queryText, shortNameIndex, uidToEntry, maxWords = 8) {
    if (!this.trained || !goalVec) return null;

    const STOP = new Set(['is','the','a','an','of','to','and','in','with','for','on','it','as','by']);
    const used = new Set(queryText.toLowerCase().split(/\s+/));
    let sentence = queryText.slice(-ORDER * 3); // seed from end of query
    const words = [];

    for (let step = 0; step < maxWords; step++) {
      // Propose 16 candidate words via n-gram sampling
      const candidates = new Set();
      for (let k = 0; k < 16; k++) {
        const w = this._sampleWord(sentence);
        if (w.length >= 2) candidates.add(w.toLowerCase());
      }

      let bestWord = null, bestScore = -1;

      for (const word of candidates) {
        if (used.has(word)) continue;

        if (STOP.has(word)) {
          // Glue words get a base score — allow them to connect
          if (bestScore < 0.5) { bestWord = word; bestScore = 0.5; }
          continue;
        }

        // Look up word vector in KB
        let wordVec = null;
        const uid = shortNameIndex?.get(word);
        if (uid && uidToEntry) {
          const entry = uidToEntry.get(uid);
          if (entry?.vector?.length === 24) wordVec = entry.vector;
        }
        if (!wordVec) continue; // Skip unknown content words (penalised)

        const score = xorBridgeScore(goalVec, wordVec) * 1.5;
        if (score > bestScore) { bestScore = score; bestWord = word; }
      }

      if (!bestWord || bestScore < 0.3) break;
      words.push(bestWord);
      used.add(bestWord);
      sentence += ' ' + bestWord;
    }

    return words.length >= 3 ? words.join(' ') : null;
  }

  // Export manifold as JSON (for download/persistence)
  exportManifold() {
    const obj = {};
    for (const [ctx, charMap] of this.manifold) {
      obj[ctx] = Object.fromEntries(charMap);
    }
    return {
      version: 1,
      contextCount: this.manifold.size,
      vocabSize: this.vocab.size,
      totalLearned: this._totalLearned,
      exportedAt: new Date().toISOString(),
      manifold: obj,
      log: this._log.slice(-50),
      gaps: [...this._gaps].slice(0, 500),
    };
  }

  // Import manifold from JSON (for upload/reload)
  importManifold(data) {
    if (!data?.manifold) throw new Error('Invalid manifold file: missing manifold key');
    for (const [ctx, charObj] of Object.entries(data.manifold)) {
      if (!this.manifold.has(ctx)) this.manifold.set(ctx, new Map());
      const m = this.manifold.get(ctx);
      for (const [c, count] of Object.entries(charObj)) {
        m.set(c, (m.get(c) || 0) + Number(count));
      }
      for (const c of ctx) this.vocab.add(c);
    }
    if (data.totalLearned) this._totalLearned += data.totalLearned;
    if (data.log) this._log = [...(data.log || []), ...this._log].slice(-200);
    if (data.gaps) data.gaps.forEach(g => this._gaps.add(g));
    this.trained = true;
    this._log.push({ type: 'import', summary: `Imported manifold: ${data.contextCount ?? '?'} contexts from ${data.exportedAt ?? 'unknown'}`, samples: data.totalLearned ?? 0, t: Date.now() });
    console.log(`[Linguist] Imported: ${this.manifold.size} total contexts`);
  }

  // Get top N contexts by total frequency
  getTopContexts(n = 20) {
    const list = [];
    for (const [ctx, charMap] of this.manifold) {
      let total = 0, topChar = '', topCount = 0;
      for (const [c, count] of charMap) {
        total += count;
        if (count > topCount) { topCount = count; topChar = c; }
      }
      list.push({ ctx, total, topChar, topCount });
    }
    list.sort((a, b) => b.total - a.total);
    return list.slice(0, n);
  }

  getLog() { return this._log; }
  getGapReport() { return { gaps: [...this._gaps] }; }

  stats() {
    return {
      trained: this.trained,
      contexts: this.manifold.size,
      vocab: this.vocab.size,
      totalLearned: this._totalLearned,
    };
  }
}
