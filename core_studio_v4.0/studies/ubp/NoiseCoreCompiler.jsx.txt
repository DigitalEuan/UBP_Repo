// ── Query → program selector ──────────────────────────────────────────────────

export function compileQuery(query) {
  const q = query.toLowerCase().trim();

  // Triad
  if (/\btriad\b/.test(q)) return PROGRAMS.triad();

  // Factorial
  const factM = q.match(/factorial\s+(?:of\s+)?(\d+)|(\d+)!/);
  if (factM) {
    const n = parseInt(factM[1] || factM[2]);
    if (n >= 0 && n <= 20) return PROGRAMS.factorial(n);
  }

  // Fibonacci
  const fibM = q.match(/fib(?:onacci)?\s*(?:of\s+)?(?:\()?\s*(\d+)/);
  if (fibM) {
    const n = parseInt(fibM[1]);
    if (n >= 0 && n <= 30) return PROGRAMS.fibonacci(n);
  }

  // GCD
  const gcdM = q.match(/gcd\s*[\(\s,of]*\s*(\d+)\s*[,\s)and]+\s*(\d+)/);
  if (gcdM) return PROGRAMS.gcd(parseInt(gcdM[1]), parseInt(gcdM[2]));

  // LCM
  const lcmM = q.match(/lcm\s*[\(\s,of]*\s*(\d+)\s*[,\s)and]+\s*(\d+)/);
  if (lcmM) return PROGRAMS.lcm(parseInt(lcmM[1]), parseInt(lcmM[2]));

  // Modpow
  const modpowM = q.match(/(\d+)\s*\^\s*(\d+)\s*mod\s*(\d+)/);
  if (modpowM) return PROGRAMS.modpow(parseInt(modpowM[1]), parseInt(modpowM[2]), parseInt(modpowM[3]));

  // Power (no mod)
  const powM = q.match(/(\d+)\s*\^\s*(\d+)(?!\s*mod)/);
  if (powM) return PROGRAMS.power(parseInt(powM[1]), parseInt(powM[2]));

  // Prime test
  const primeM = q.match(/(?:is\s+)?(\d+)\s*prime|prime\s*(?:test|check)?\s*(\d+)/);
  if (primeM) {
    const n = parseInt(primeM[1] || primeM[2]);
    if (n < 10000) return PROGRAMS.prime(n);
  }

  // Sum 1..n
  const sumM = q.match(/sum\s+(?:of\s+)?(?:\d+\s+to\s+)?(?:integers?\s+)?(?:1\s+to\s+)?(\d+)/);
  if (sumM) return PROGRAMS.sum(parseInt(sumM[1]));

  // Combination
  const combM = q.match(/(?:c|choose|comb(?:ination)?)\s*\(?(\d+)[,\s]+(\d+)/);
  if (combM) return PROGRAMS.comb(parseInt(combM[1]), parseInt(combM[2]));

  // Arithmetic: +,-,*,/,%
  const arithM = q.match(/(-?\d+)\s*([\+\-\*×\/÷%]|\bmod\b)\s*(-?\d+)/);
  if (arithM) {
    const [a, op, b] = [parseInt(arithM[1]), arithM[2], parseInt(arithM[3])];
    if (op === '+') return PROGRAMS.add(a, b);
    if (op === '-') return PROGRAMS.sub(a, b);
    if (op === '*' || op === '×') return PROGRAMS.mul(a, b);
    if (op === '/' || op === '÷') return PROGRAMS.divmod(a, b);
    if (op === '%' || op === 'mod') return PROGRAMS.divmod(a, b);
  }

  return null; // no match
}
