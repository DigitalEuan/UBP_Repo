/**
 * UBP VOCAB LEARNER
 * Collects words that fell through to geometric-hash fallback (unknown words).
 * Lets the user anchor them to KB entries or define them manually.
 * Exports a JSON file compatible with ubp_kb_architect.py.
 */
import React, { useState, useEffect } from 'react';
import { Download, Hash, CheckCircle, Trash2, ChevronDown, ChevronUp } from 'lucide-react';
import { Button } from '@/components/ui/button';

function wordToFingerprintHex(word) {
  let h = 5381;
  for (let i = 0; i < word.length; i++) { h = ((h << 5) + h) ^ word.charCodeAt(i); h = h >>> 0; }
  return h.toString(16).padStart(8, '0');
}

function buildExportEntry(word, def, anchor) {
  const fp = wordToFingerprintHex(word);
  return {
    [fp]: {
      ubp_id: `VOCAB_${word.toUpperCase().replace(/\s+/g, '_')}_${fp.slice(0,4).toUpperCase()}`,
      lexicon: `[VOCAB: ${word}] [${def || word}] [user-defined]`,
      math: '',
      tags: ['VOCAB', 'USER_DEFINED'],
      atlas: {
        nrci: '1/2',
        hierarchy: anchor || `user/${word}`,
        vector: null,
      },
    },
  };
}

function WordCard({ word, domain, onDefine, onRemove }) {
  const [open, setOpen] = useState(false);
  const [def, setDef] = useState('');
  const [anchor, setAnchor] = useState('');
  const [saved, setSaved] = useState(false);

  const save = () => {
    if (!def.trim()) return;
    onDefine(word, def.trim(), anchor.trim());
    setSaved(true);
  };

  return (
    <div className={`rounded-xl border p-3 text-sm transition-colors ${saved ? 'border-emerald-500/40 bg-emerald-500/5' : 'border-slate-700/50 bg-slate-800/30'}`}>
      <div className="flex items-center gap-2">
        <Hash className="w-3.5 h-3.5 text-amber-400 flex-shrink-0" />
        <span className="font-mono text-amber-300 flex-1">{word}</span>
        <span className="text-xs text-slate-500 bg-slate-700/40 px-1.5 py-0.5 rounded">{domain || '?'}</span>
        {saved && <CheckCircle className="w-3.5 h-3.5 text-emerald-400" />}
        <button onClick={() => setOpen(o => !o)} className="text-slate-500 hover:text-slate-300">
          {open ? <ChevronUp className="w-3.5 h-3.5" /> : <ChevronDown className="w-3.5 h-3.5" />}
        </button>
        <button onClick={() => onRemove(word)} className="text-slate-600 hover:text-red-400">
          <Trash2 className="w-3.5 h-3.5" />
        </button>
      </div>

      {open && (
        <div className="mt-3 space-y-2">
          <input
            value={def}
            onChange={e => setDef(e.target.value)}
            placeholder="Definition / description…"
            className="w-full bg-slate-900/60 border border-slate-700/50 rounded-lg px-3 py-1.5 text-xs text-slate-200 placeholder:text-slate-600 focus:outline-none focus:border-cyan-500/60"
          />
          <input
            value={anchor}
            onChange={e => setAnchor(e.target.value)}
            placeholder="Anchor to KB entry UID (optional)"
            className="w-full bg-slate-900/60 border border-slate-700/50 rounded-lg px-3 py-1.5 text-xs text-slate-200 placeholder:text-slate-600 focus:outline-none focus:border-cyan-500/60"
          />
          <Button
            size="sm"
            onClick={save}
            disabled={!def.trim()}
            className="h-7 text-xs bg-cyan-600 hover:bg-cyan-500 disabled:opacity-40"
          >
            {saved ? 'Update' : 'Save definition'}
          </Button>
        </div>
      )}
    </div>
  );
}

export function VocabLearner({ vocabWords = [] }) {
  // vocabWords: [{ word, domain }]
  const [words, setWords] = useState(() => {
    const seen = new Set();
    return vocabWords.filter(w => { if (seen.has(w.word)) return false; seen.add(w.word); return true; });
  });
  const [definitions, setDefinitions] = useState({});

  // Merge in new words from prop changes
  React.useEffect(() => {
    setWords(prev => {
      const existing = new Set(prev.map(w => w.word));
      const newOnes = vocabWords.filter(w => !existing.has(w.word));
      return [...prev, ...newOnes];
    });
  }, [vocabWords]);

  const define = (word, def, anchor) => {
    setDefinitions(d => ({ ...d, [word]: { def, anchor } }));
  };
  const remove = (word) => {
    setWords(prev => prev.filter(w => w.word !== word));
    setDefinitions(d => { const c = { ...d }; delete c[word]; return c; });
  };

  const exportJSON = () => {
    const defined = words.filter(w => definitions[w.word]);
    if (!defined.length) return;
    const out = {};
    for (const { word, domain } of defined) {
      const { def, anchor } = definitions[word];
      Object.assign(out, buildExportEntry(word, def, anchor));
    }
    const blob = new Blob([JSON.stringify(out, null, 2)], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a'); a.href = url; a.download = 'ubp_vocab_extension.json'; a.click();
    URL.revokeObjectURL(url);
  };

  const definedCount = Object.keys(definitions).length;

  return (
    <div className="space-y-4">
      <div className="flex items-center justify-between">
        <div>
          <h2 className="text-base font-semibold text-slate-100">Vocab Learner</h2>
          <p className="text-xs text-slate-500 mt-0.5">
            Words that hit geometric-hash fallback (not in KB)
          </p>
        </div>
        {definedCount > 0 && (
          <Button size="sm" onClick={exportJSON} className="h-8 text-xs bg-violet-600 hover:bg-violet-500 gap-1.5">
            <Download className="w-3 h-3" />
            Export JSON ({definedCount})
          </Button>
        )}
      </div>

      {words.length === 0 ? (
        <div className="text-center py-10 text-slate-500 text-sm">
          <Hash className="w-8 h-8 mx-auto mb-3 opacity-30" />
          No unknown words yet.<br />
          <span className="text-xs">Send queries — words not in the KB will appear here.</span>
        </div>
      ) : (
        <div className="space-y-2">
          {words.map(w => (
            <WordCard
              key={w.word}
              word={w.word}
              domain={w.domain}
              onDefine={define}
              onRemove={remove}
            />
          ))}
        </div>
      )}

      {definedCount > 0 && (
        <div className="p-3 rounded-xl border border-violet-500/30 bg-violet-500/5 text-xs text-slate-400">
          <p className="font-medium text-violet-300 mb-1">Export usage:</p>
          <p>Load <code className="text-violet-400">ubp_vocab_extension.json</code> into <code className="text-violet-400">ubp_kb_architect.py</code> to integrate these definitions into your KB pipeline.</p>
        </div>
      )}
    </div>
  );
}
