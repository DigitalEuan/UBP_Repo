/**
 * UBP GRAY-CODE NUMERIC VECTOR ENGINE
 * JS port of workflow/03_numeric_vector_engine.py (Study Report §3.4)
 *
 * KEY INSIGHT: SHA256 has the avalanche property — similar inputs produce random
 * Hamming distances (~12/24). Gray-code quantisation maps similar numeric values
 * to nearby bit patterns (d=0 for same bin, ≤2 for small differences).
 *
 * Pipeline per property:
 *   value → clamp → normalise (linear or log) → quantise → Gray-encode → [present_flag | bits]
 *
 * Unified 166-bit vector = [24-bit Golay semantic | 142-bit Gray-code numeric]
 * Optimal alpha for property queries: 0.0 (numeric only)
 * Optimal alpha for symbolic queries: 1.0 (Golay only)
 */

// Build a lookup map for fast access
const SCHEMA_MAP = {};

// ── Gray code encoding ────────────────────────────────────────────────────────
/** Convert integer k to n-bit binary-reflected Gray code */
function grayEncode(k, n) {
  const gray = k ^ (k >> 1);
  const bits = [];
  for (let i = n - 1; i >= 0; i--) bits.push((gray >> i) & 1);
  return bits;
}

// ── Math string parser ────────────────────────────────────────────────────────
/**
 * Parse a UBP math DNA string "M=18.015|BP=373.15|Z=1" into a dict of floats.
 * Supports: integers, decimals, rational fractions (126/125), scientific notation (1.0546e-34)
 */
export function parseMathString(mathStr) {
  if (!mathStr || typeof mathStr !== 'string') return {};
  const result = {};
  for (const pair of mathStr.split('|')) {
    const eq = pair.indexOf('=');
    if (eq < 0) continue;
    const key = pair.slice(0, eq).trim();
    const val = pair.slice(eq + 1).trim();
    if (!key || !val) continue;
    // Rational fraction
    if (val.includes('/')) {
      const [n, d] = val.split('/').map(Number);
      if (!isNaN(n) && !isNaN(d) && d !== 0) result[key] = n / d;
    } else {
      const f = parseFloat(val);
      if (!isNaN(f)) result[key] = f;
    }
  }
  return result;
}

// ── Per-property encoder ──────────────────────────────────────────────────────
function encodeProperty(value, schema) {
  const { bits, scale, min, max } = schema;
  // Clamp
  const v = Math.max(min, Math.min(max, value));
  let normalized;
  if (scale === 'log') {
    const logMin = Math.log(Math.max(min, 1e-10));
    const logMax = Math.log(Math.max(max, 1e-10));
    normalized = (Math.log(Math.max(v, 1e-10)) - logMin) / (logMax - logMin);
  } else {
    normalized = (v - min) / (max - min);
  }
  normalized = Math.max(0, Math.min(1, normalized));
  const binCount = (1 << bits) - 1;
  // FIXED v7.2: safety clamp to prevent overflow when normalized is exactly 1.0
  const bin = Math.min(Math.round(normalized * binCount), binCount);
  return grayEncode(bin, bits);
}

// ── Full 142-bit numeric vector encoder ──────────────────────────────────────
/**
 * Encode a math dict into a 142-bit Gray-code numeric vector.
 * Each property contributes [present_flag (1 bit) | Gray-coded value (N bits)].
 * Returns { vector: number[], presenceMask: number[] }
 * presenceMask[i] = 1 if bit i belongs to a present property.
 */
export function encodeNumericVector(mathDict) {
  const vector = [];
  const presenceMask = [];

  for (const schema of PROP_SCHEMA) {
    const value = mathDict[schema.key];
    const present = value !== undefined && value !== null && !isNaN(value) ? 1 : 0;
    // Present flag
    vector.push(present);
    presenceMask.push(present);
    // Value bits
    const valueBits = present ? encodeProperty(value, schema) : new Array(schema.bits).fill(0);
    for (const b of valueBits) {
      vector.push(b);
      presenceMask.push(present); // only score these bits if present
    }
  }

  return { vector, presenceMask };
}

// ── Masked Hamming distance ───────────────────────────────────────────────────
/**
 * Compute masked Hamming distance between query and entry vectors.
 * Only bits where queryMask[i]=1 are counted — absent properties are wildcards.
 * Returns normalised distance in [0, 1].
 */
export function maskedHammingDistance(queryVec, entryVec, queryMask) {
  if (!queryVec || !entryVec) return 1;
  let diff = 0;
  let total = 0;
  const len = Math.min(queryVec.length, entryVec.length);
  for (let i = 0; i < len; i++) {
    if (queryMask[i]) {
      total++;
      if (queryVec[i] !== entryVec[i]) diff++;
    }
  }
  return total > 0 ? diff / total : 1;
}

// ── Cache for KB numeric vectors ──────────────────────────────────────────────
const _numVecCache = new Map();

/** Get or compute numeric vector for a KB entry (cached).
 * v9.9: entries no longer have an `entry.math` string.
 * Properties sit in the mog_tensor (complex sparse structure).
 * For the numeric channel we only score query-side properties;
 * entry-side we return an empty (zero) vector so masked Hamming
 * simply falls back to Gray-code query matching. */
export function getEntryNumericVector(entry) {
  const uid = entry.ubp_id;
  if (_numVecCache.has(uid)) return _numVecCache.get(uid);
  // v9.9 has no flat math string — use empty dict (all absent = all wildcards on entry side)
  const mathStr = entry.math || '';
  const mathDict = parseMathString(mathStr);
  const result = encodeNumericVector(mathDict);
  _numVecCache.set(uid, result);
  return result;
}

/** Clear the cache (call when KB changes) */
export function clearNumericVectorCache() {
  _numVecCache.clear();
}
