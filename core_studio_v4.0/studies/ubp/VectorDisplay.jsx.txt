import React from 'react';
import { motion } from 'framer-motion';
import { DOMAIN_COLORS } from './UBPLexicon';

/**
 * Displays a 24-bit vector visually with domain coloring
 */
export function VectorDisplay({ bits, domain, size = 'md', showHex = true, animate = false }) {
  if (!bits || !Array.isArray(bits) || bits.length !== 24) return null;
  const colors = DOMAIN_COLORS[domain] || DOMAIN_COLORS.MEANING;
  
  const sizeClasses = {
    sm: 'w-1.5 h-3',
    md: 'w-2 h-4',
    lg: 'w-3 h-6'
  };
  
  const gapClasses = {
    sm: 'gap-0.5',
    md: 'gap-0.5',
    lg: 'gap-1'
  };

  const vectorInt = parseInt(bits.join(''), 2);
  const hexStr = vectorInt.toString(16).padStart(6, '0').toUpperCase();

  return (
    <div className="flex flex-col gap-1">
      <div className={`flex items-center ${gapClasses[size]}`}>
        {/* Domain bits (0-2) */}
        <div className="flex gap-px">
          {bits.slice(0, 3).map((bit, i) => (
            <motion.div
              key={`d${i}`}
              initial={animate ? { scale: 0 } : false}
              animate={{ scale: 1 }}
              transition={{ delay: i * 0.02 }}
              className={`${sizeClasses[size]} rounded-sm ${
                bit === 1 
                  ? `${colors.bg} border ${colors.border}` 
                  : 'bg-slate-700/50 border border-slate-600/30'
              }`}
            />
          ))}
        </div>
        
        <div className="w-1" />
        
        {/* Message bits (3-11) */}
        <div className="flex gap-px">
          {bits.slice(3, 12).map((bit, i) => (
            <motion.div
              key={`m${i}`}
              initial={animate ? { scale: 0 } : false}
              animate={{ scale: 1 }}
              transition={{ delay: (3 + i) * 0.02 }}
              className={`${sizeClasses[size]} rounded-sm ${
                bit === 1 
                  ? 'bg-cyan-500/30 border border-cyan-500/40' 
                  : 'bg-slate-700/50 border border-slate-600/30'
              }`}
            />
          ))}
        </div>
        
        <div className="w-1" />
        
        {/* Parity bits (12-23) */}
        <div className="flex gap-px">
          {bits.slice(12, 24).map((bit, i) => (
            <motion.div
              key={`p${i}`}
              initial={animate ? { scale: 0 } : false}
              animate={{ scale: 1 }}
              transition={{ delay: (12 + i) * 0.02 }}
              className={`${sizeClasses[size]} rounded-sm ${
                bit === 1 
                  ? 'bg-violet-500/30 border border-violet-500/40' 
                  : 'bg-slate-700/50 border border-slate-600/30'
              }`}
            />
          ))}
        </div>
      </div>
      
      {showHex && (
        <div className="flex items-center gap-2 text-xs">
          <span className="font-mono text-slate-500">0x</span>
          <span className={`font-mono ${colors.text}`}>{hexStr}</span>
        </div>
      )}
    </div>
  );
}

/**
 * Compact badge showing domain
 */
export function DomainBadge({ domain, size = 'md' }) {
  const colors = DOMAIN_COLORS[domain] || DOMAIN_COLORS.MEANING;
  
  const sizeClasses = {
    sm: 'text-xs px-1.5 py-0.5',
    md: 'text-xs px-2 py-1',
    lg: 'text-sm px-3 py-1.5'
  };
  
  return (
    <span className={`
      ${sizeClasses[size]} 
      ${colors.bg} ${colors.text} ${colors.border}
      border rounded-full font-medium
    `}>
      {domain}
    </span>
  );
}

/**
 * Token visualization with vector popup
 */
export function TokenChip({ token, showVector = false }) {
  const colors = DOMAIN_COLORS[token.domain] || DOMAIN_COLORS.MEANING;
  const [isHovered, setIsHovered] = React.useState(false);
  
  return (
    <div 
      className="relative inline-block"
      onMouseEnter={() => setIsHovered(true)}
      onMouseLeave={() => setIsHovered(false)}
    >
      <span className={`
        inline-flex items-center gap-1 px-2 py-0.5 rounded-md
        ${colors.bg} ${colors.border} border
        text-sm font-mono cursor-default
        transition-all duration-200
        ${isHovered ? 'ring-1 ring-white/20' : ''}
      `}>
        <span className={colors.text}>{token.word}</span>
        {token.found && (
          <span className="text-xs text-slate-500">→{token.representative}</span>
        )}
      </span>
      
      {isHovered && showVector && (
        <motion.div
          initial={{ opacity: 0, y: 4 }}
          animate={{ opacity: 1, y: 0 }}
          className="absolute z-10 top-full left-0 mt-1 p-2 bg-slate-800 rounded-lg border border-slate-700 shadow-xl"
        >
          <VectorDisplay 
            bits={token.vectorBits} 
            domain={token.domain} 
            size="sm"
            showHex={true}
          />
          <div className="mt-1 text-xs text-slate-400">
            Cluster: {token.clusterSize} words
          </div>
        </motion.div>
      )}
    </div>
  );
}

/**
 * Hamming distance indicator
 */
export function HammingIndicator({ distance, maxDistance = 12 }) {
  const percentage = Math.max(0, 100 - (distance / maxDistance) * 100);
  
  let color = 'bg-green-500';
  if (distance > 8) color = 'bg-red-500';
  else if (distance > 4) color = 'bg-yellow-500';
  else if (distance > 2) color = 'bg-cyan-500';
  
  return (
    <div className="flex items-center gap-2">
      <div className="w-16 h-1.5 bg-slate-700 rounded-full overflow-hidden">
        <motion.div
          initial={{ width: 0 }}
          animate={{ width: `${percentage}%` }}
          transition={{ duration: 0.3 }}
          className={`h-full ${color} rounded-full`}
        />
      </div>
      <span className="text-xs font-mono text-slate-400">d={distance}</span>
    </div>
  );
}
