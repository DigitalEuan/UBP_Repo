/**
 * NoiseCore VM — JavaScript port of the NoiseCore VM v1.3.0
 * Full 46-instruction ISA + SubstrateALU (base-12) + SYSCALLs 1–15
 * Two-pass symbolic assembler with labels, .equ, .data directives
 *
 * Architecture: 8 registers (R0–R7, R7=SP), 256-cell memory,
 *               5 flags {Z N C V S}, 16-bit word, max_cycles guard
 */

// ── SubstrateALU: base-12 digit array arithmetic ─────────────────────────────
// LAW_SUBSTRATE_ALU_001/002/003 — addition/subtraction/multiplication via
// carry/borrow propagation on base-12 cell arrays

const SUBSTRATE_BASE = 12;
const SHADOW_THRESHOLD = 4; // digit > 4 → Shadow regime

function toBase12(n, cells = 24) {
  const sign = n < 0 ? -1 : 1;
  let abs = Math.abs(Math.trunc(n));
  const digits = new Array(cells).fill(0);
  for (let i = 0; i < cells && abs > 0; i++) {
    digits[i] = abs % SUBSTRATE_BASE;
    abs = Math.floor(abs / SUBSTRATE_BASE);
  }
  return { sign, digits, shadow: digits.some(d => d > SHADOW_THRESHOLD) };
}

function fromBase12({ sign, digits }) {
  let val = 0;
  for (let i = digits.length - 1; i >= 0; i--) val = val * SUBSTRATE_BASE + digits[i];
  return sign * val;
}

function substrateAdd(a, b) {
  const da = toBase12(Math.abs(a));
  const db = toBase12(Math.abs(b));
  // Determine effective signs
  const sA = a < 0 ? -1 : 1, sB = b < 0 ? -1 : 1;
  let carry = 0, carryWork = 0;
  const result = new Array(da.digits.length).fill(0);
  if (sA === sB) {
    for (let i = 0; i < result.length; i++) {
      const s = da.digits[i] + db.digits[i] + carry;
      carry = Math.floor(s / SUBSTRATE_BASE);
      result[i] = s % SUBSTRATE_BASE;
      if (carry > 0) carryWork++;
    }
    const val = fromBase12({ sign: sA, digits: result });
    return { value: val, carry, carryWork, shadow: result.some(d => d > SHADOW_THRESHOLD) };
  } else {
    // Subtraction path: subtract smaller from larger
    const absA = Math.abs(a), absB = Math.abs(b);
    const [larger, smaller, sign] = absA >= absB ? [da.digits, db.digits, sA] : [db.digits, da.digits, sB];
    let borrow = 0;
    for (let i = 0; i < result.length; i++) {
      let d = larger[i] - smaller[i] - borrow;
      if (d < 0) { d += SUBSTRATE_BASE; borrow = 1; carryWork++; } else borrow = 0;
      result[i] = d;
    }
    const val = fromBase12({ sign, digits: result });
    return { value: val, carry: 0, carryWork, shadow: result.some(d => d > SHADOW_THRESHOLD) };
  }
}

function substrateMul(a, b) {
  // LAW_SUBSTRATE_ALU_002: base-12 long multiplication
  const da = toBase12(Math.abs(a), 12);
  const db = toBase12(Math.abs(b), 12);
  const result = new Array(24).fill(0);
  let carryWork = 0;
  for (let i = 0; i < 12; i++) {
    for (let j = 0; j < 12 && i + j < 24; j++) {
      result[i + j] += da.digits[i] * db.digits[j];
    }
  }
  for (let i = 0; i < result.length - 1; i++) {
    if (result[i] >= SUBSTRATE_BASE) {
      const carry = Math.floor(result[i] / SUBSTRATE_BASE);
      result[i] %= SUBSTRATE_BASE;
      result[i + 1] += carry;
      carryWork++;
    }
  }
  const sign = (a < 0) !== (b < 0) ? -1 : 1;
  const val = fromBase12({ sign, digits: result });
  return { value: val, carryWork, shadow: result.some(d => d > SHADOW_THRESHOLD) };
}

// ── Assembler ─────────────────────────────────────────────────────────────────

const MNEMONICS = new Set([
  'LOAD','MOV','ADD','SUB','AND','OR','XOR','NOT','LSL','LSR','MUL','DIV','MOD','CMP','HALT',
  'JMP','JZ','JNZ','JN','JNN','JC','JE','JNE','JG','JL','JGE','JLE',
  'ADDI','SUBI','MULI','ANDI','ORI','CMPI',
  'STORE','LOAD_MEM','STOREI','LOADI','STOREX','LOADX',
  'PUSH','POP','CALL','RET','SYSCALL','NOP','CLRS',
]);

function parseImmediate(tok, equs) {
  if (tok === undefined || tok === null) return null;
  const s = String(tok).trim();
  // Register ref is not an immediate
  if (/^R[0-7]$/i.test(s)) return null;
  // Strip leading #
  const t = s.startsWith('#') ? s.slice(1) : s;
  // Char literal
  if (t.startsWith("'") && t.endsWith("'")) return t.charCodeAt(1);
  // Hex
  if (t.startsWith('0x') || t.startsWith('0X')) return parseInt(t, 16);
  // Binary
  if (t.startsWith('0b') || t.startsWith('0B')) return parseInt(t.slice(2), 2);
  // Decimal
  if (/^-?\d+$/.test(t)) return parseInt(t, 10);
  // Named constant
  if (equs && equs.has(t)) return equs.get(t);
  return null;
}

function parseRegister(tok) {
  if (!tok) return null;
  const m = String(tok).trim().match(/^[Rr]([0-7])$/);
  return m ? parseInt(m[1]) : null;
}

export function assemble(source) {
  const lines = source.split('\n');
  const equs = new Map();   // name → value
  const labels = new Map(); // name → instruction index
  const dataMap = new Map(); // name → memory address
  const memory = new Array(256).fill(0);
  let dataPtr = 200; // .data starts near end of memory

  // ── Strip comment, trim
  const clean = (line) => line.replace(/;.*$/, '').trim();

  // ── Pass 1: collect labels, .equ, .data ───────────────────────────────────
  const codeLines = []; // only instruction lines
  for (const raw of lines) {
    const line = clean(raw);
    if (!line) continue;

    // .equ NAME VALUE
    const equM = line.match(/^\.equ\s+(\w+)\s+(.+)$/i);
    if (equM) {
      const val = parseImmediate(equM[2].trim(), equs);
      if (val !== null) equs.set(equM[1], val);
      continue;
    }

    // .org ADDR
    const orgM = line.match(/^\.org\s+(\d+)$/i);
    if (orgM) { dataPtr = parseInt(orgM[1]); continue; }

    // .data NAME v1 v2 ...
    const dataM = line.match(/^\.data\s+(\w+)\s+(.+)$/i);
    if (dataM) {
      const name = dataM[1];
      dataMap.set(name, dataPtr);
      equs.set(name, dataPtr);
      const parts = dataM[2].trim().split(/\s+/);
      for (const p of parts) {
        const v = parseImmediate(p, equs);
        if (v !== null && dataPtr < 256) memory[dataPtr++] = v;
      }
      continue;
    }

    // LABEL: (possibly followed by instruction on same line)
    const labelM = line.match(/^(\w+):\s*(.*)?$/);
    if (labelM && !MNEMONICS.has(labelM[1].toUpperCase())) {
      labels.set(labelM[1], codeLines.length);
      const rest = labelM[2]?.trim();
      if (rest) codeLines.push(rest);
      continue;
    }

    // Regular instruction line
    codeLines.push(line);
  }

  // ── Pass 2: emit instructions ─────────────────────────────────────────────
  const instructions = [];

  const resolveAddr = (tok) => {
    const t = String(tok).trim();
    if (labels.has(t)) return labels.get(t);
    if (equs.has(t)) return equs.get(t);
    const n = parseInt(t, 10);
    return isNaN(n) ? 0 : n;
  };

  for (const line of codeLines) {
    // tokenise: split on spaces and commas
    const tokens = line.split(/[\s,]+/).filter(Boolean);
    const mnem = tokens[0]?.toUpperCase();
    const args = tokens.slice(1);

    const R = (i) => parseRegister(args[i]);
    const I = (i) => { const v = parseImmediate(args[i], equs); return v !== null ? v : resolveAddr(args[i]); };
    const A = (i) => resolveAddr(args[i]);

    switch (mnem) {
      case 'LOAD':     instructions.push({ op: 'LOAD',     rd: R(0), imm: I(1) }); break;
      case 'MOV':      instructions.push({ op: 'MOV',      rd: R(0), rs: R(1) }); break;
      case 'ADD':      instructions.push({ op: 'ADD',      rd: R(0), rs1: R(1), rs2: R(2) }); break;
      case 'SUB':      instructions.push({ op: 'SUB',      rd: R(0), rs1: R(1), rs2: R(2) }); break;
      case 'AND':      instructions.push({ op: 'AND',      rd: R(0), rs1: R(1), rs2: R(2) }); break;
      case 'OR':       instructions.push({ op: 'OR',       rd: R(0), rs1: R(1), rs2: R(2) }); break;
      case 'XOR':      instructions.push({ op: 'XOR',      rd: R(0), rs1: R(1), rs2: R(2) }); break;
      case 'NOT':      instructions.push({ op: 'NOT',      rd: R(0), rs: R(1) }); break;
      case 'LSL':      instructions.push({ op: 'LSL',      rd: R(0), rs: R(1), imm: I(2) }); break;
      case 'LSR':      instructions.push({ op: 'LSR',      rd: R(0), rs: R(1), imm: I(2) }); break;
      case 'MUL':      instructions.push({ op: 'MUL',      rd: R(0), rs1: R(1), rs2: R(2) }); break;
      case 'DIV':      instructions.push({ op: 'DIV',      rd: R(0), rs1: R(1), rs2: R(2) }); break;
      case 'MOD':      instructions.push({ op: 'MOD',      rd: R(0), rs1: R(1), rs2: R(2) }); break;
      case 'CMP':      instructions.push({ op: 'CMP',      rs1: R(0), rs2: R(1) }); break;
      case 'HALT':     instructions.push({ op: 'HALT' }); break;
      case 'JMP':      instructions.push({ op: 'JMP',      addr: A(0) }); break;
      case 'JZ':       instructions.push({ op: 'JZ',       addr: A(0) }); break;
      case 'JNZ':      instructions.push({ op: 'JNZ',      addr: A(0) }); break;
      case 'JN':       instructions.push({ op: 'JN',       addr: A(0) }); break;
      case 'JNN':      instructions.push({ op: 'JNN',      addr: A(0) }); break;
      case 'JC':       instructions.push({ op: 'JC',       addr: A(0) }); break;
      case 'JE':       instructions.push({ op: 'JE',       addr: A(0) }); break;
      case 'JNE':      instructions.push({ op: 'JNE',      addr: A(0) }); break;
      case 'JG':       instructions.push({ op: 'JG',       addr: A(0) }); break;
      case 'JL':       instructions.push({ op: 'JL',       addr: A(0) }); break;
      case 'JGE':      instructions.push({ op: 'JGE',      addr: A(0) }); break;
      case 'JLE':      instructions.push({ op: 'JLE',      addr: A(0) }); break;
      case 'ADDI':     instructions.push({ op: 'ADDI',     rd: R(0), rs: R(1), imm: I(2) }); break;
      case 'SUBI':     instructions.push({ op: 'SUBI',     rd: R(0), rs: R(1), imm: I(2) }); break;
      case 'MULI':     instructions.push({ op: 'MULI',     rd: R(0), rs: R(1), imm: I(2) }); break;
      case 'ANDI':     instructions.push({ op: 'ANDI',     rd: R(0), rs: R(1), imm: I(2) }); break;
      case 'ORI':      instructions.push({ op: 'ORI',      rd: R(0), rs: R(1), imm: I(2) }); break;
      case 'CMPI':     instructions.push({ op: 'CMPI',     rs: R(0), imm: I(1) }); break;
      case 'STORE':    instructions.push({ op: 'STORE',    rs: R(0), addr: A(1) }); break;
      case 'LOAD_MEM': instructions.push({ op: 'LOAD_MEM', rd: R(0), addr: A(1) }); break;
      case 'STOREI':   instructions.push({ op: 'STOREI',   rs: R(0), rb: R(1) }); break;
      case 'LOADI':    instructions.push({ op: 'LOADI',    rd: R(0), rb: R(1) }); break;
      case 'STOREX':   instructions.push({ op: 'STOREX',   rs: R(0), rb: R(1), imm: I(2) }); break;
      case 'LOADX':    instructions.push({ op: 'LOADX',    rd: R(0), rb: R(1), imm: I(2) }); break;
      case 'PUSH':     instructions.push({ op: 'PUSH',     rs: R(0) }); break;
      case 'POP':      instructions.push({ op: 'POP',      rd: R(0) }); break;
      case 'CALL':     instructions.push({ op: 'CALL',     addr: A(0) }); break;
      case 'RET':      instructions.push({ op: 'RET' }); break;
      case 'SYSCALL':  instructions.push({ op: 'SYSCALL',  num: I(0) }); break;
      case 'NOP':      instructions.push({ op: 'NOP' }); break;
      case 'CLRS':     instructions.push({ op: 'CLRS' }); break;
      default:
        // unknown mnemonic — skip (could be label-only line residual)
        if (mnem) instructions.push({ op: 'NOP', _comment: `unknown: ${mnem}` });
    }
  }

  return { instructions, memory, labels, equs };
}

// ── CPU ───────────────────────────────────────────────────────────────────────

const WORD_BITS = 16;
const WORD_MASK = (1 << WORD_BITS) - 1;
const SIGN_BIT  = 1 << (WORD_BITS - 1);
const MAX_CYCLES = 100_000;

function signExtend(v) {
  const u = v & WORD_MASK;
  return u & SIGN_BIT ? u - (1 << WORD_BITS) : u;
}

function setFlags(regs, val, carry = 0, overflow = 0) {
  regs.flags.Z = val === 0 ? 1 : 0;
  regs.flags.N = val < 0 ? 1 : 0;
  regs.flags.C = carry ? 1 : 0;
  regs.flags.V = overflow ? 1 : 0;
}

export function executeCPU(program, opts = {}) {
  const { substrate = false, maxCycles = MAX_CYCLES, inputQueue = [] } = opts;
  const { instructions, memory: initMem } = typeof program === 'string'
    ? assemble(program)
    : program;

  const regs = {
    r: new Array(8).fill(0),
    flags: { Z: 0, N: 0, C: 0, V: 0, S: 0 },
    pc: 0,
  };
  regs.r[7] = 255; // SP starts at top of memory

  const memory = [...initMem];
  const output = [];
  const trace = [];
  const callStack = [];
  let carryWork = 0;
  let cycles = 0;
  const inputBuf = [...inputQueue];

  const printLine = (s) => output.push(String(s));
  const addTrace = (s) => { trace.push(s); if (trace.length > 200) trace.shift(); };

  const doArith = (op, a, b) => {
    if (substrate) {
      if (op === 'ADD' || op === 'ADDI') {
        const r = substrateAdd(a, b);
        carryWork += r.carryWork;
        if (r.shadow) regs.flags.S = 1;
        return r.value;
      }
      if (op === 'SUB' || op === 'SUBI') {
        const r = substrateAdd(a, -b);
        carryWork += r.carryWork;
        if (r.shadow) regs.flags.S = 1;
        return r.value;
      }
      if (op === 'MUL' || op === 'MULI') {
        const r = substrateMul(a, b);
        carryWork += r.carryWork;
        if (r.shadow) regs.flags.S = 1;
        return r.value;
      }
    }
    switch (op) {
      case 'ADD': case 'ADDI': return a + b;
      case 'SUB': case 'SUBI': return a - b;
      case 'MUL': case 'MULI': return a * b;
      case 'DIV': return b === 0 ? 0 : Math.trunc(a / b);
      case 'MOD': return b === 0 ? 0 : a - Math.trunc(a / b) * b;
      default: return 0;
    }
  };

  while (cycles < maxCycles) {
    if (regs.pc < 0 || regs.pc >= instructions.length) break;
    const inst = instructions[regs.pc];
    cycles++;

    const r = regs.r;
    const f = regs.flags;
    let nextPc = regs.pc + 1;

    addTrace(`PC=${String(regs.pc).padStart(3,'0')} | ${inst.op.padEnd(8)} | R0=${r[0]} R1=${r[1]} R2=${r[2]} R3=${r[3]}`);

    switch (inst.op) {
      case 'LOAD': {
        r[inst.rd] = inst.imm ?? 0;
        setFlags(regs, r[inst.rd]);
        break;
      }
      case 'MOV': {
        r[inst.rd] = r[inst.rs];
        setFlags(regs, r[inst.rd]);
        break;
      }
      case 'ADD': case 'SUB': case 'MUL': case 'DIV': case 'MOD': {
        const a = r[inst.rs1], b = r[inst.rs2];
        const res = doArith(inst.op, a, b);
        r[inst.rd] = res;
        const carry = inst.op === 'ADD' ? (res > 65535 ? 1 : 0) : (inst.op === 'SUB' ? (a < b ? 1 : 0) : 0);
        setFlags(regs, res, carry);
        break;
      }
      case 'AND': r[inst.rd] = r[inst.rs1] & r[inst.rs2]; setFlags(regs, r[inst.rd]); break;
      case 'OR':  r[inst.rd] = r[inst.rs1] | r[inst.rs2]; setFlags(regs, r[inst.rd]); break;
      case 'XOR': r[inst.rd] = r[inst.rs1] ^ r[inst.rs2]; setFlags(regs, r[inst.rd]); break;
      case 'NOT': r[inst.rd] = ~r[inst.rs] & WORD_MASK;   setFlags(regs, r[inst.rd]); break;
      case 'LSL': r[inst.rd] = (r[inst.rs] << inst.imm) & WORD_MASK; setFlags(regs, r[inst.rd]); break;
      case 'LSR': r[inst.rd] = (r[inst.rs] >>> inst.imm) & WORD_MASK; setFlags(regs, r[inst.rd]); break;
      case 'CMP': {
        const diff = r[inst.rs1] - r[inst.rs2];
        setFlags(regs, diff, r[inst.rs1] < r[inst.rs2] ? 1 : 0);
        break;
      }
      case 'HALT': nextPc = regs.pc; goto_halt: { break goto_halt; } // fall through
        // eslint-disable-next-line no-fallthrough
        cycles = maxCycles; break;
      case 'JMP': nextPc = inst.addr; break;
      case 'JZ': case 'JE': if (f.Z) nextPc = inst.addr; break;
      case 'JNZ': case 'JNE': if (!f.Z) nextPc = inst.addr; break;
      case 'JN':  if (f.N) nextPc = inst.addr; break;
      case 'JNN': if (!f.N) nextPc = inst.addr; break;
      case 'JC':  if (f.C) nextPc = inst.addr; break;
      case 'JG':  if (!f.Z && f.N === f.V) nextPc = inst.addr; break;
      case 'JL':  if (f.N !== f.V) nextPc = inst.addr; break;
      case 'JGE': if (f.N === f.V) nextPc = inst.addr; break;
      case 'JLE': if (f.Z || f.N !== f.V) nextPc = inst.addr; break;
      case 'ADDI': { const res = doArith('ADDI', r[inst.rs], inst.imm); r[inst.rd] = res; setFlags(regs, res); break; }
      case 'SUBI': { const res = doArith('SUBI', r[inst.rs], inst.imm); r[inst.rd] = res; setFlags(regs, res); break; }
      case 'MULI': { const res = doArith('MULI', r[inst.rs], inst.imm); r[inst.rd] = res; setFlags(regs, res); break; }
      case 'ANDI': r[inst.rd] = r[inst.rs] & inst.imm; setFlags(regs, r[inst.rd]); break;
      case 'ORI':  r[inst.rd] = r[inst.rs] | inst.imm; setFlags(regs, r[inst.rd]); break;
      case 'CMPI': { const diff = r[inst.rs] - inst.imm; setFlags(regs, diff, r[inst.rs] < inst.imm ? 1 : 0); break; }
      case 'STORE':    if (inst.addr >= 0 && inst.addr < 256) memory[inst.addr] = r[inst.rs]; break;
      case 'LOAD_MEM': if (inst.addr >= 0 && inst.addr < 256) { r[inst.rd] = memory[inst.addr]; setFlags(regs, r[inst.rd]); } break;
      case 'STOREI':   { const addr = r[inst.rb]; if (addr >= 0 && addr < 256) memory[addr] = r[inst.rs]; break; }
      case 'LOADI':    { const addr = r[inst.rb]; if (addr >= 0 && addr < 256) { r[inst.rd] = memory[addr]; setFlags(regs, r[inst.rd]); } break; }
      case 'STOREX':   { const addr = r[inst.rb] + inst.imm; if (addr >= 0 && addr < 256) memory[addr] = r[inst.rs]; break; }
      case 'LOADX':    { const addr = r[inst.rb] + inst.imm; if (addr >= 0 && addr < 256) { r[inst.rd] = memory[addr]; setFlags(regs, r[inst.rd]); } break; }
      case 'PUSH': { const sp = r[7]; if (sp >= 0 && sp < 256) memory[sp] = r[inst.rs]; r[7]--; break; }
      case 'POP':  { r[7]++; const sp = r[7]; if (sp >= 0 && sp < 256) { r[inst.rd] = memory[sp]; setFlags(regs, r[inst.rd]); } break; }
      case 'CALL': callStack.push(regs.pc + 1); nextPc = inst.addr; break;
      case 'RET':  nextPc = callStack.length ? callStack.pop() : regs.pc + 1; break;
      case 'SYSCALL': {
        switch (inst.num) {
          case 1: printLine(r[0]); break;                             // PRINT_INT
          case 2: printLine(String.fromCharCode(r[0])); break;        // PRINT_CHAR
          case 3: {                                                    // PRINT_STR
            let addr = r[0]; let s = '';
            while (addr < 256 && memory[addr] !== 0) s += String.fromCharCode(memory[addr++]);
            printLine(s); break;
          }
          case 4: r[0] = inputBuf.length ? inputBuf.shift() : 0; break; // READ_INT
          case 5: cycles = maxCycles; break;                          // EXIT
          case 6: printLine(''); break;                               // PRINT_NL
          case 7: {                                                    // DUMP_REGS
            const dump = r.map((v,i) => `R${i}=${v}`).join(' ');
            const flags = `Z=${f.Z} N=${f.N} C=${f.C} V=${f.V} S=${f.S}`;
            printLine(`[REGS] ${dump} PC=${regs.pc} ${flags}`);
            break;
          }
          // Triad SYSCALLs 8-15 (lightweight stubs — full substrate via BarnesWallEngine)
          case 8: printLine(`[LEECH] OCTAD LIFT R0=${r[0]} → 128 lattice points`); break;
          case 9: printLine(`[LEECH] TAX R0=${r[0]} → hw×Y_LEECH + |p|²/8`); break;
          case 10: printLine(`[LEECH] HEALTH R0=${r[0]} → Reality·Info·Activation·Potential`); break;
          case 11: printLine(`[BW256] AUDIT → macroNRCI computed`); break;
          case 12: printLine(`[TRIAD] STATUS → Golay✓ Leech✓ Monster✓ Level=3`); break;
          case 13: printLine(`[MONSTER] GROUP INFO R0=${r[0]}`); break;
          case 14: printLine(`[PARTICLE] PREDICTION R0=${r[0]}`); break;
          case 15: printLine(`[LATTICE] PROBE R0=${r[0]} → nearest octad`); break;
          default: printLine(`[SYSCALL ${inst.num}]`);
        }
        break;
      }
      case 'CLRS': f.S = 0; break;
      case 'NOP': break;
      default: break;
    }

    if (inst.op === 'HALT') break;
    regs.pc = nextPc;
  }

  // Compute NRCI from final R0 value
  const finalVal = regs.r[0];
  const absV = Math.abs(Math.trunc(finalVal)) & 0xFFFFFF;
  const gray = absV ^ (absV >> 1);
  let sw = 0;
  for (let i = 0; i < 24; i++) sw += (gray >> i) & 1;
  const nrci = 10 / (10 + sw * 0.2647 + sw / 8);

  return {
    registers: [...regs.r],
    flags: { ...regs.flags },
    pc: regs.pc,
    memory: memory.slice(0, 64), // first 64 cells for display
    output,
    trace: trace.slice(-50),
    cycles,
    carryWork,
    nrci: parseFloat(nrci.toFixed(4)),
    sw,
  };
}
