import React from 'react';
import { motion } from 'framer-motion';
import ReactMarkdown from 'react-markdown';
import { User, Cpu, ChevronDown, ChevronUp } from 'lucide-react';
import { VectorDisplay, DomainBadge, TokenChip, HammingIndicator } from './VectorDisplay';
import { DOMAIN_COLORS } from './UBPLexicon';
import { grayCodeToHex } from './UBPLinguist';

export function ChatMessage({ message, isLast }) {
  const [showDetails, setShowDetails] = React.useState(false);
  const isUser = message.role === 'user';
  
  return (
    <motion.div
      initial={{ opacity: 0, y: 20 }}
      animate={{ opacity: 1, y: 0 }}
      transition={{ duration: 0.3 }}
      className={`flex gap-3 ${isUser ? 'flex-row-reverse' : ''}`}
    >
      {/* Avatar */}
      <div className={`
        flex-shrink-0 w-8 h-8 rounded-lg flex items-center justify-center
        ${isUser 
          ? 'bg-gradient-to-br from-violet-500 to-purple-600' 
          : 'bg-gradient-to-br from-cyan-500 to-blue-600'
        }
      `}>
        {isUser ? (
          <User className="w-4 h-4 text-white" />
        ) : (
          <Cpu className="w-4 h-4 text-white" />
        )}
      </div>
      
      {/* Content */}
      <div className={`flex-1 max-w-[85%] ${isUser ? 'flex flex-col items-end' : ''}`}>
        {/* Main bubble */}
        <div className={`
          rounded-2xl px-4 py-3
          ${isUser 
            ? 'bg-gradient-to-br from-violet-500/20 to-purple-600/20 border border-violet-500/30' 
            : 'bg-slate-800/80 border border-slate-700/50'
          }
        `}>
          {isUser ? (
            <p className="text-slate-100 text-sm leading-relaxed">{message.content}</p>
          ) : (
            <div className="text-slate-100 text-sm leading-relaxed prose prose-sm prose-invert max-w-none">
              <ReactMarkdown
                components={{
                  p: ({ children }) => <p className="mb-2 last:mb-0">{children}</p>,
                  code: ({ inline, children }) => (
                    inline 
                      ? <code className="px-1 py-0.5 bg-slate-700/50 rounded text-cyan-400 text-xs">{children}</code>
                      : <pre className="p-2 bg-slate-900/50 rounded-lg overflow-x-auto"><code className="text-xs">{children}</code></pre>
                  ),
                  strong: ({ children }) => <strong className="text-cyan-400 font-semibold">{children}</strong>,
                }}
              >
                {message.content}
              </ReactMarkdown>
            </div>
          )}
        </div>
        
        {/* UBP Details (only for assistant messages) */}
        {!isUser && message.ubpData && (
          <div className="mt-2 w-full">
            {/* Quick stats bar */}
            <div className="flex items-center gap-3 text-xs text-slate-500 flex-wrap">
              <DomainBadge domain={message.ubpData.domain} size="sm" />
              {message.ubpData.queryType && (
                <span className={`px-1.5 py-0.5 rounded text-xs font-mono ${
                  message.ubpData.queryType === 'noisecore'  ? 'bg-amber-500/10 text-amber-400' :
                  message.ubpData.queryType === 'property'   ? 'bg-emerald-500/10 text-emerald-400' :
                  message.ubpData.queryType === 'symbolic'   ? 'bg-blue-500/10 text-blue-400' :
                  'bg-violet-500/10 text-violet-400'
                }`}>
                  {message.ubpData.queryType === 'noisecore'  ? '⚡ NoiseCore ALU' :
                   message.ubpData.queryType === 'property'   ? 'α=0 numeric' :
                   message.ubpData.queryType === 'symbolic'   ? 'α=1 symbolic + BW-256' : 'α=0.25 mixed + BW-256'}
                </span>
              )}
              {message.ubpData.stats && (
                <>
                  <span>{message.ubpData.stats.chainLength} steps</span>
                  <span>•</span>
                  <span>{message.ubpData.stats.spikesFound} spikes</span>
                  {message.ubpData.stats.propertiesExtracted > 0 && (
                    <><span>•</span><span>{message.ubpData.stats.propertiesExtracted} props</span></>
                  )}
                  {message.ubpData.stats.identityLock && (
                    <><span>•</span><span className="text-emerald-400 font-mono text-xs" title={message.ubpData.stats.identityLock}>🔒 locked</span></>
                  )}
                  {message.ubpData.stats.domainIntent && message.ubpData.stats.domainIntent !== 'general' && (
                    <><span>•</span><span className="text-amber-400 text-xs">{message.ubpData.stats.domainIntent}</span></>
                  )}
                </>
              )}
              <button
                onClick={() => setShowDetails(!showDetails)}
                className="flex items-center gap-1 text-slate-400 hover:text-slate-300 transition-colors ml-auto"
              >
                {showDetails ? <ChevronUp className="w-3 h-3" /> : <ChevronDown className="w-3 h-3" />}
                {showDetails ? 'Hide' : 'Details'}
              </button>
            </div>
            
            {/* Expanded details */}
            {showDetails && (
              <motion.div
                initial={{ opacity: 0, height: 0 }}
                animate={{ opacity: 1, height: 'auto' }}
                exit={{ opacity: 0, height: 0 }}
                className="mt-3 p-3 bg-slate-900/50 rounded-xl border border-slate-700/50 space-y-4"
              >
                {/* Query Vector */}
                {message.ubpData.queryVector && (
                  <div>
                    <h4 className="text-xs font-medium text-slate-400 mb-2">Query Vector</h4>
                    <VectorDisplay 
                      bits={message.ubpData.queryVector.bits} 
                      domain={message.ubpData.domain}
                      size="sm"
                      showHex={true}
                      animate={true}
                    />
                  </div>
                )}
                
                {/* Tokens */}
                {message.ubpData.tokens && message.ubpData.tokens.length > 0 && (
                  <div>
                    <h4 className="text-xs font-medium text-slate-400 mb-2">Geometric Tokens</h4>
                    <div className="flex flex-wrap gap-1">
                      {message.ubpData.tokens.slice(0, 12).map((token, i) => (
                        <TokenChip key={i} token={token} showVector={true} />
                      ))}
                    </div>
                  </div>
                )}
                
                {/* Reasoning Chain */}
                {message.ubpData.steps && message.ubpData.steps.length > 0 && (
                  <div>
                    <h4 className="text-xs font-medium text-slate-400 mb-2">Reasoning Chain</h4>
                    <div className="space-y-2">
                      {message.ubpData.steps.slice(0, 4).map((step, i) => (
                        <div 
                          key={i}
                          className="p-2 bg-slate-800/50 rounded-lg border border-slate-700/30"
                        >
                          <div className="flex items-center gap-2 mb-1">
                            <span className="text-xs font-mono text-slate-500">#{i + 1}</span>
                            <DomainBadge domain={step.domain} size="sm" />
                            <span className="text-xs text-slate-500">{step.source}</span>
                            {step.coherence && (
                              <span className="text-xs text-cyan-400 ml-auto">
                                {(step.coherence * 100).toFixed(0)}% coherence
                              </span>
                            )}
                          </div>
                          <p className="text-xs text-slate-300 line-clamp-2">{step.content}</p>
                          <div className="flex items-center gap-2 mt-1 flex-wrap">
                            {step.vector && (
                              <span
                                className="inline-block w-3 h-3 rounded-sm border border-slate-600 flex-shrink-0"
                                style={{ background: grayCodeToHex(step.vector) }}
                                title={`Gray-code color: ${grayCodeToHex(step.vector)}`}
                              />
                            )}
                            {step.vectorHex && (
                              <span className="text-xs font-mono text-slate-500">0x{step.vectorHex}</span>
                            )}
                            {step.lattice && (
                              <span className={`text-xs px-1 py-0.5 rounded font-mono ${
                                step.lattice === 'Dodecad' ? 'bg-cyan-500/10 text-cyan-400' :
                                step.lattice === 'Octad' ? 'bg-violet-500/10 text-violet-400' :
                                step.lattice === 'Hexadecad' ? 'bg-amber-500/10 text-amber-400' :
                                step.lattice === 'Universe' ? 'bg-red-500/10 text-red-400' :
                                'bg-slate-700/50 text-slate-400'
                              }`}>{step.lattice}</span>
                            )}
                            {step.macroNrci !== null && step.macroNrci !== undefined && (
                              <span className="text-xs text-slate-500">BW:{step.macroNrci.toFixed(3)}</span>
                            )}
                            {step.clarity && (
                              <span className={`text-xs ${step.clarity === 'HIGH' ? 'text-emerald-400' : 'text-slate-500'}`}>
                                {step.clarity}
                              </span>
                            )}
                            {step.shadow && (
                              <span className="text-xs text-slate-500" title="Shadow Processor: Noumenal/Phenomenal split">
                                N:{step.shadow.noumenal_hw}/P:{step.shadow.phenomenal_hw}
                              </span>
                            )}
                            {step.tgicOrtho !== null && step.tgicOrtho !== undefined && (
                              <span className={`text-xs ${step.tgicOrtho > 0.7 ? 'text-cyan-400' : step.tgicOrtho > 0.5 ? 'text-amber-400' : 'text-slate-500'}`} title="TGIC 3-axis orthogonality">
                                3ax:{step.tgicOrtho.toFixed(2)}
                              </span>
                            )}
                          </div>
                        </div>
                      ))}
                    </div>
                  </div>
                )}
              </motion.div>
            )}
          </div>
        )}
        
        {/* Timestamp */}
        <span className="text-xs text-slate-600 mt-1">
          {new Date(message.timestamp).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })}
        </span>
      </div>
    </motion.div>
  );
}
