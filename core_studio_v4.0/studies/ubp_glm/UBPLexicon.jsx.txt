// ── Internal: domain bits (must match UBPEngineV3.inferDomainBits exactly) ────
function _inferDomainBits(word) {
  const w = word.toLowerCase();
  if (/force$|energy$|wave$|field$|gravity|quantum|photon|boson|kinetic/.test(w)) return [0,0,1];
  if (/sis$|biology|cell|organism|life|neural|protein/.test(w)) return [0,1,0];
  if (/algorithm|code|compute|function|logic|binary/.test(w)) return [0,1,1];
  if (/math|number|ratio|constant|sqrt/.test(w)) return [1,0,0];
  if (/law$|rule$|axiom|principle/.test(w)) return [1,0,1];
  if (/entropy|chaos|decay|random/.test(w)) return [1,1,0];
  if (/ium$|ogen$|ine$/.test(w) && w.length <= 8) return [0,0,0];
  return [1,1,1]; // MEANING (default)
}

function _buildMessage12(word) {
  const w = word.toLowerCase();
  const dom = _inferDomainBits(w); // [d0,d1,d2]

  // Magnitude: 5 bits — Gray-encoded word length (0..31)
  const lenVal = Math.min(w.length, 31);
  const grayLen = lenVal ^ (lenVal >> 1);
  const mag = [];
  for (let i = 4; i >= 0; i--) mag.push((grayLen >> i) & 1);

  // State: 4 bits — Gray-encoded char-density sum
  let charSum = 0;
  for (let i = 0; i < Math.min(w.length, 8); i++) {
    const c = w.charCodeAt(i);
    if (c >= 97 && c <= 122) charSum += (c - 96); // a=1..z=26 continuous
  }
  const stateVal = charSum % 16;
  const grayState = stateVal ^ (stateVal >> 1);
  const state = [];
  for (let i = 3; i >= 0; i--) state.push((grayState >> i) & 1);

  return [...dom, ...mag, ...state]; // length 12
}

// ── Word → 24-bit Golay codeword ─────────────────────────────────────────────
/**
 * Synchronous. Deterministic. No SHA-256, no DJB2 — pure Gray-code UMS.
 * Returns a 24-bit unsigned integer codeword.
 */
export function hashWordSync(word) {
  const msg12 = _buildMessage12(word);
  const cw = _encodeGolayArr(msg12);
  let v = 0;
  for (let i = 0; i < 24; i++) v = (v << 1) | cw[i];
  return v >>> 0;
}

/** Async wrapper (backward-compatible API — no network call needed). */
export async function hashWord(word) { return hashWordSync(word); }

/** Returns the 24-bit codeword as a length-24 array of 0/1. */
export function hashWordBits(word) {
  return _encodeGolayArr(_buildMessage12(word));
}

// ── Bit utilities ─────────────────────────────────────────────────────────────
export function vectorToBits(vInt) {
  const bits = new Array(24);
  for (let i = 0; i < 24; i++) bits[i] = (vInt >> (23 - i)) & 1;
  return bits;
}

/** Extract the ontological domain from a 24-bit codeword integer (top 3 bits). */
export function getDomain(vectorInt) {
  const domainBits = (vectorInt >> 21) & 7;
  return DOMAINS[domainBits] || 'MEANING';
}

/** Hamming distance between two 24-bit codeword integers. */
export function hammingDistance(v1, v2) {
  let xor = (v1 ^ v2) >>> 0;
  let count = 0;
  while (xor) { count += xor & 1; xor >>>= 1; }
  return count;
}

/** Hamming weight (popcount) of a 24-bit codeword integer. */
export function hammingWeight(v) {
  let count = 0, val = v >>> 0;
  while (val) { count += val & 1; val >>>= 1; }
  return count;
}

export function vectorToHex(vInt) { return (vInt >>> 0).toString(16).padStart(6, '0'); }

export function vectorToBinaryString(vInt) {
  const bits = vectorToBits(vInt);
  return [
    bits.slice(0, 3).join(''),   // Domain  (3)
    bits.slice(3, 12).join(''),  // Message (9)
    bits.slice(12, 24).join(''), // Parity  (12)
  ].join(' ');
}

// ── Lexicon factory ───────────────────────────────────────────────────────────
/**
 * createLexicon() returns an object with the same shape as the v2 lexicon,
 * but the codeword pipeline is now canonical Golay (Conway-Sloane) + Gray-code.
 * Can ingest a v2-format cluster file from UBPLexiconLoader, or operate
 * KB-free (returns Gray-code synthesis for unknown words).
 */
export function createLexicon() {
  let clusters = {};
  let loaded = false;

  const _lookupVector = (word, vectorInt) => {
    const hexKey = vectorToHex(vectorInt);
    const cluster = clusters[hexKey];
    if (cluster) {
      return {
        word: word.toLowerCase().trim(),
        vector: vectorInt,
        vectorHex: hexKey,
        vectorBits: vectorToBits(vectorInt),
        domain: DOMAIN_CHARS[cluster[1]] || getDomain(vectorInt),
        representative: cluster[0],
        clusterSize: cluster[2],
        definition: cluster[3] || '',
        found: true,
      };
    }
    return {
      word: word.toLowerCase().trim(),
      vector: vectorInt,
      vectorHex: hexKey,
      vectorBits: vectorToBits(vectorInt),
      domain: getDomain(vectorInt),
      representative: word.toLowerCase().trim(),
      clusterSize: 1,
      definition: '',
      found: false,
    };
  };

  return {
    async load(source) {
      const data = typeof source === 'string'
        ? await (await fetch(source)).json()
        : source;
      clusters = data?.c || {};
      loaded = true;
      console.log(`[UBPLexicon] Loaded ${Object.keys(clusters).length} clusters`);
      return this;
    },
    loadSync(data) {
      clusters = data?.c || {};
      loaded = true;
      return this;
    },
    isLoaded()      { return loaded; },
    getClusterCount() { return Object.keys(clusters).length; },

    lookup(word)     { return _lookupVector(word, hashWordSync(word)); },
    lookupSync(word) { return _lookupVector(word, hashWordSync(word)); },

    tokenize(text) {
      return text.toLowerCase().split(/\s+/).filter(w => w.length > 0)
        .map(w => _lookupVector(w, hashWordSync(w)));
    },

    nearest(vectorInt, threshold = 3) {
      let best = null, minDist = threshold + 1;
      for (const [hexKey, cluster] of Object.entries(clusters)) {
        const vInt = parseInt(hexKey, 16);
        const d = hammingDistance(vectorInt, vInt);
        if (d < minDist) {
          minDist = d;
          best = {
            vector: vInt, vectorHex: hexKey, distance: d,
            representative: cluster[0],
            domain: DOMAIN_CHARS[cluster[1]] || 'MEANING',
            clusterSize: cluster[2],
          };
        }
      }
      return best;
    },

    findNearby(vectorInt, threshold = 3, limit = 10) {
      const results = [];
      for (const [hexKey, cluster] of Object.entries(clusters)) {
        const vInt = parseInt(hexKey, 16);
        const d = hammingDistance(vectorInt, vInt);
        if (d <= threshold) {
          results.push({
            vector: vInt, vectorHex: hexKey, distance: d,
            representative: cluster[0],
            domain: DOMAIN_CHARS[cluster[1]] || 'MEANING',
            clusterSize: cluster[2],
          });
        }
      }
      results.sort((a, b) => a.distance - b.distance || b.clusterSize - a.clusterSize);
      return results.slice(0, limit);
    },

    stats() {
      const entries = Object.values(clusters);
      const sizes = entries.map(c => c[2] || 1);
      const total = sizes.reduce((a, b) => a + b, 0);
      const domainCounts = {};
      for (const cluster of entries) {
        const d = DOMAIN_CHARS[cluster[1]] || 'MEANING';
        domainCounts[d] = (domainCounts[d] || 0) + 1;
      }
      return {
        clusterCount: entries.length,
        totalWords: total,
        avgClusterSize: entries.length ? total / entries.length : 0,
        maxClusterSize: sizes.length ? Math.max(...sizes) : 0,
        domainCounts,
        loaded,
      };
    },

    getClusters() { return clusters; },
  };
}
