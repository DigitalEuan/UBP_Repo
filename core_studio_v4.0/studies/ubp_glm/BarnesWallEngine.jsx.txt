// ═══════════════════════════════════════════════════════════════════════════════
//  GOLAY SELF  [24,12,8] — exact syndrome-table decoder
// ═══════════════════════════════════════════════════════════════════════════════

class GolaySelf {
  constructor() {
    // H columns: H[:,k] = B[k] for k<12, identity for k≥12 (B is symmetric)
    this._Hcols = [];
    for (let k = 0; k < 24; k++) {
      if (k < 12) {
        this._Hcols.push(B[k].slice());
      } else {
        const col = new Array(12).fill(0);
        col[k - 12] = 1;
        this._Hcols.push(col);
      }
    }
    this._synTable = null; // lazy
    this._octads   = null; // lazy
  }

  // ── Encoding ────────────────────────────────────────────────────────────────
  encode(msg12) {
    const cw = msg12.slice(0, 12);
    for (let j = 0; j < 12; j++) {
      let p = 0;
      for (let i = 0; i < 12; i++) p ^= (msg12[i] & B[j][i]);
      cw.push(p);
    }
    return cw;
  }

  // ── Syndrome ────────────────────────────────────────────────────────────────
  syndrome(v24) {
    const s = new Array(12).fill(0);
    for (let k = 0; k < 24; k++) {
      if (v24[k]) {
        const col = this._Hcols[k];
        for (let j = 0; j < 12; j++) s[j] ^= col[j];
      }
    }
    return s;
  }

  syndromeWeight(v24) {
    return this.syndrome(v24).reduce((a, b) => a + b, 0);
  }

  // ── Syndrome table (lazy) — all weight ≤ 3 error patterns (2325 entries) ──
  _buildSynTable() {
    const table = new Map();
    const cols = this._Hcols;
    const xorCols = (idxs) => {
      const s = new Array(12).fill(0);
      for (const i of idxs) { const c = cols[i]; for (let j = 0; j < 12; j++) s[j] ^= c[j]; }
      return s.join(',');
    };

    // Weight 0
    const zero = new Array(12).fill(0);
    table.set(zero.join(','), new Array(24).fill(0));

    // Weight 1
    for (let i = 0; i < 24; i++) {
      const e = new Array(24).fill(0); e[i] = 1;
      table.set(xorCols([i]), e);
    }

    // Weight 2
    for (let i = 0; i < 24; i++) {
      for (let j = i + 1; j < 24; j++) {
        const e = new Array(24).fill(0); e[i] = 1; e[j] = 1;
        table.set(xorCols([i, j]), e);
      }
    }

    // Weight 3
    for (let i = 0; i < 24; i++) {
      for (let j = i + 1; j < 24; j++) {
        for (let k = j + 1; k < 24; k++) {
          const e = new Array(24).fill(0); e[i] = 1; e[j] = 1; e[k] = 1;
          table.set(xorCols([i, j, k]), e);
        }
      }
    }
    return table;
  }

  // ── Snap to nearest Golay codeword (exact, t ≤ 3) ──────────────────────────
  snapToCodeword(v24) {
    const s = this.syndrome(v24);
    const sw = s.reduce((a, b) => a + b, 0);
    if (sw === 0) return { cw: v24.slice(), sw: 0, corrected: false, distance: 0 };

    if (!this._synTable) this._synTable = this._buildSynTable();
    const key = s.join(',');
    const e = this._synTable.get(key);
    if (e) {
      const corrected = v24.map((b, i) => b ^ e[i]);
      return { cw: corrected, sw, corrected: true, distance: e.reduce((a, b) => a + b, 0) };
    }
    // Beyond correction radius — return as-is
    return { cw: v24.slice(), sw, corrected: false, distance: -1 };
  }

  // ── All 759 octads (lazy) ────────────────────────────────────────────────────
  getOctads() {
    if (!this._octads) {
      this._octads = [];
      for (let i = 0; i < 4096; i++) {
        const msg = [];
        for (let k = 0; k < 12; k++) msg.push((i >> k) & 1);
        const cw = this.encode(msg);
        if (cw.reduce((a, b) => a + b, 0) === 8) this._octads.push(cw);
      }
    }
    return this._octads;
  }
}

class LeechSelf {
  constructor() {
    this.DIM     = 24;
    this.KISSING = 196560;
  }

  // expand octad → 128 Leech points (±2, even neg-count rule)
  expandOctad(octad) {
    const active = octad.map((b, i) => b ? i : -1).filter(i => i >= 0);
    if (active.length !== 8) throw new Error(`expandOctad: expected hw=8, got hw=${active.length}`);
    const pts = [];
    for (let mask = 0; mask < 256; mask++) {
      const negCount = (mask).toString(2).replace(/0/g, '').length;
      if (negCount % 2 !== 0) continue; // must have EVEN number of minus signs
      const p = new Array(24).fill(0);
      for (let b = 0; b < 8; b++) p[active[b]] = ((mask >> b) & 1) ? -2 : 2;
      pts.push(p);
    }
    return pts; // exactly 128
  }

  // symmetry tax: T = hw × Y_LEECH + ‖p‖²/8
  symmetryTax(point) {
    const hw   = point.filter(x => x !== 0).length;
    const normSq = point.reduce((s, x) => s + x * x, 0);
    return hw * Y_LEECH + normSq / 8;
  }

  // tetradic MOG partition: Reality | Info | Activation | Potential
  ontologicalHealth(point) {
    const layer = (slice) => slice.reduce((s, c) => s + Math.abs(c), 0) / 12;
    const R = layer(point.slice(0, 6));
    const I = layer(point.slice(6, 12));
    const A = layer(point.slice(12, 18));
    const P = layer(point.slice(18, 24));
    return { Reality: R, Info: I, Activation: A, Potential: P, Global_NRCI: (R + I + A + P) / 4 };
  }

  // find nearest octad by Hamming distance (O(759×24))
  nearestOctadIdx(seed24) {
    const octads = _golay.getOctads();
    let bestIdx = 0, bestD = 25;
    for (let i = 0; i < octads.length; i++) {
      let d = 0;
      for (let k = 0; k < 24; k++) if (octads[i][k] !== seed24[k]) d++;
      if (d < bestD) { bestD = d; bestIdx = i; if (d === 0) break; }
    }
    return { idx: bestIdx, distance: bestD };
  }

  rankByStability(pts) {
    return pts.map(p => ({ point: p, tax: this.symmetryTax(p) }))
              .sort((a, b) => a.tax - b.tax);
  }

  normSqScaled(p) { return p.reduce((s, x) => s + x * x, 0); }

  stats() {
    return {
      dimension: this.DIM,
      kissing: this.KISSING,
      octads: _golay.getOctads().length,
      Y_LEECH,
      norm_sq_octad_point: 32,
    };
  }
}

function triadState(stableCount, sporadicCount) {
  const level = (stableCount >= TRIAD_THRESH.leech && sporadicCount >= TRIAD_THRESH.monster) ? 3
              : (stableCount >= TRIAD_THRESH.leech) ? 2
              : (stableCount >= TRIAD_THRESH.golay) ? 1 : 0;
  return {
    golay_active:   stableCount   >= TRIAD_THRESH.golay,
    leech_active:   stableCount   >= TRIAD_THRESH.leech,
    monster_active: sporadicCount >= TRIAD_THRESH.monster,
    stable_count:   stableCount,
    sporadic_count: sporadicCount,
    thresholds:     { ...TRIAD_THRESH },
    level,
  };
}

// ═══════════════════════════════════════════════════════════════════════════════
//  BWALL SELF — BW(256) with exact GolaySelf snap
// ═══════════════════════════════════════════════════════════════════════════════

class BWallSelf {
  // v derived from encode(syndrome(seed)) — "geometric frustration"
  _syndromeV(seed24) {
    const s = _golay.syndrome(seed24);
    return _golay.encode(s);
  }

  generate(seed24, dim = 256) {
    return this._generateR(seed24, dim);
  }

  _generateR(seed24, dim) {
    if (dim <= 32) {
      const padded = [...seed24.slice(0, 24), ...new Array(8).fill(0)];
      return padded.map(x => x * 2);
    }
    const half = dim >> 1;
    const u = this._generateR(seed24, half);
    const vBits = this._syndromeV(seed24);
    const v = [...vBits, ...new Array(Math.max(0, half - 24)).fill(0)]
      .slice(0, half)
      .map(x => x * 2);
    const result = new Array(dim);
    for (let i = 0; i < half; i++) result[i] = u[i];
    for (let i = 0; i < half; i++) result[half + i] = (u[i] + v[i]) % 4;
    return result;
  }

  // Successive-cancellation decoder using EXACT GolaySelf syndrome-table snap
  snap(macro) {
    return this._snapR(macro);
  }

  _snapR(vec) {
    const n = vec.length;
    if (n <= 32) {
      // Binarise: abs(x)//2 maps {0,1}→0, {2,3}→1
      const v24bin = vec.slice(0, 24).map(x => (Math.abs(x) >> 1) & 1);
      // Use EXACT syndrome-table snap (not greedy nearest-weight)
      const { cw } = _golay.snapToCodeword(v24bin);
      return [...cw.map(x => x * 2), ...new Array(n - 24).fill(0)];
    }
    const half = n >> 1;
    const uClean = this._snapR(vec.slice(0, half));
    const vNoisy = vec.slice(0, half).map((a, i) => (a + vec[half + i]) % 4);
    const vClean = this._snapR(vNoisy);
    const result = new Array(n);
    for (let i = 0; i < half; i++) result[i] = uClean[i];
    for (let i = 0; i < half; i++) result[half + i] = (uClean[i] + vClean[i]) % 4;
    return result;
  }

  // Macro-NRCI: NRCI = 10/(10 + hw×Y_LEECH + ‖v‖²/64)
  nrci(macro) {
    const hw = macro.filter(x => x !== 0).length;
    const normSq = macro.reduce((s, x) => s + x * x, 0);
    const tax = hw * Y_LEECH + normSq / 64;
    return 10 / (10 + tax);
  }

  audit(seed24, microNrci, dim = 256) {
    const macro   = this.generate(seed24, dim);
    const snapped = this.snap(macro);
    const macN    = this.nrci(snapped);
    const noisyN  = this.nrci(macro);
    const rel     = microNrci > 0 ? macN / microNrci : 0;
    return {
      dim,
      micro_nrci:         microNrci,
      macro_nrci:         macN,
      noisy_nrci:         noisyN,
      relative_coherence: rel,
      clarity:            rel >= MACRO_ANCHOR_NRCI ? 'HIGH' : 'LOW',
      decoder_gain:       macN - noisyN,
      seed_hw:            seed24.reduce((s, b) => s + b, 0),
      vector_hw:          snapped.filter(x => x !== 0).length,
      anchor:             MACRO_ANCHOR_NRCI,
    };
  }
}

// ═══════════════════════════════════════════════════════════════════════════════
//  PUBLIC: computeUBPFingerprint — Golay + Leech + BW256
// ═══════════════════════════════════════════════════════════════════════════════

export function computeUBPFingerprint(vec24) {
  if (!vec24 || vec24.length !== 24) return null;

  const sw = vec24.reduce((s, b) => s + b, 0);
  const onLattice = CODEWORD_WEIGHTS.has(sw);

  // Micro-NRCI: distance from Dodecad centre (12) via symmetry tax
  const tax = sw * Y_OBSERVER + sw / 8;
  const microNrci = 10 / (10 + tax);

  const lattice =
    sw === 0  ? 'Identity'  :
    sw === 8  ? 'Octad'     :
    sw === 12 ? 'Dodecad'   :
    sw === 16 ? 'Hexadecad' :
    sw === 24 ? 'Universe'  : 'Off-lattice';

  // BW256 audit
  let macroNrci = null, clarity = null, bwAuditData = null;
  try {
    const audit = _bw.audit(vec24, microNrci, 256);
    macroNrci    = audit.macro_nrci;
    clarity      = audit.clarity;
    bwAuditData  = audit;
  } catch { /* non-critical */ }

  // Leech info (only if sw === 8 — on octad)
  let leechData = null;
  if (sw === 8) {
    try {
      const pts    = _leech.expandOctad(vec24);
      const ranked = _leech.rankByStability(pts);
      const best   = ranked[0].point;
      const health = _leech.ontologicalHealth(best);
      leechData = {
        octad_points:       pts.length,
        norm_sq_scaled:     _leech.normSqScaled(best),
        symmetry_tax:       ranked[0].tax,
        ontological_health: health,
      };
    } catch { /* non-critical */ }
  }

  return { nrci: microNrci, sw, onLattice, lattice, macroNrci, clarity, leech: leechData };
}

/**
 * Barnes-Wall semantic distance between two 24-bit vectors (256D).
 */
export function barnesWallDistance(vecA, vecB) {
  if (!vecA || !vecB || vecA.length !== 24 || vecB.length !== 24) return 1;
  try {
    const macA = _bw.generate(vecA, 256);
    const macB = _bw.generate(vecB, 256);
    let dist = 0;
    for (let i = 0; i < 256; i++) dist += (macA[i] - macB[i]) ** 2;
    return Math.min(1, Math.sqrt(dist) / 32);
  } catch { return 1; }
}

  // ── Integer → 24-bit Gray-code vector ───────────────────────────────────────
  _toVec24(n) {
    const absN = Math.abs(Math.floor(typeof n === 'number' ? n : 0)) & 0xFFFFFF;
    const gray = absN ^ (absN >> 1);
    const bits = [];
    for (let i = 23; i >= 0; i--) bits.push((gray >> i) & 1);
    return bits;
  }

  // ── Full UBP fingerprint: Golay + Leech (on-octad) + BW256 ──────────────────
  // CRITICAL: always includes 'value' so the response renderer can read r.value
  _fingerprint(value) {

    const numericValue = typeof value === 'number' ? value : (typeof value === 'boolean' ? (value ? 1 : 0) : 0);
    const vec = this._toVec24(numericValue);
    const sw  = vec.reduce((s, b) => s + b, 0);
    const onLattice = CODEWORD_WEIGHTS.has(sw);
    if (onLattice) this._opCountStable++;
    this._opCount++;

    const lattice =
      sw === 0  ? 'Identity'  :
      sw === 8  ? 'Octad'     :
      sw === 12 ? 'Dodecad'   :
      sw === 16 ? 'Hexadecad' :
      sw === 24 ? 'Universe'  : 'Off-lattice';

    // Top-level fields the renderer expects
    const fp = {
      value:     numericValue,          // ← THE KEY FIX: numeric result is always here
      sw,
      onLattice,
      nrci:      parseFloat(nrci.toFixed(6)),
      lattice,
      gray_hw:   sw,
      on_lattice: onLattice,
      macroNrci:  null,
      clarity:    null,
    };

    // Leech (only when Gray code is an octad)
    if (sw === 8) {
      try {
        const pts    = _leech.expandOctad(vec);
        const ranked = _leech.rankByStability(pts);
        const best   = ranked[0].point;
        const health = _leech.ontologicalHealth(best);
        fp.leech = {
          octad_points:       pts.length,
          norm_sq_scaled:     _leech.normSqScaled(best),
          symmetry_tax:       parseFloat(ranked[0].tax.toFixed(6)),

    }

    // BW256 — populate top-level macroNrci + clarity for renderer
    try {
      const audit   = _bw.audit(vec, nrci, 256);
      fp.macroNrci  = parseFloat(audit.macro_nrci.toFixed(6));
      fp.clarity    = audit.clarity;
      fp.nrci_fraction = nrciFraction; // exact rational form (v5.2)
      fp.bw256      = {
        macro_nrci:   fp.macroNrci,
        decoder_gain: parseFloat(audit.decoder_gain.toFixed(6)),
        clarity:      fp.clarity,
      };
    } catch { fp.bw256 = null; }

    return fp;
  }

  // ── Core arithmetic ──────────────────────────────────────────────────────────
  add(a, b)  { return this._fingerprint(a + b); }
  sub(a, b)  { return this._fingerprint(a - b); }
  mul(a, b)  { return this._fingerprint(a * b); }

  divmod(a, b) {
    if (b === 0) return { value: null, error: 'Division by zero' };
    return { quotient: this._fingerprint(Math.trunc(a / b)), remainder: this._fingerprint(a % b) };
  }

  gcd(a, b) {
    let [x, y] = [Math.abs(a), Math.abs(b)];
    while (y) [x, y] = [y, x % y];
    return this._fingerprint(x);
  }

  lcm(a, b) {
    const product = Math.abs(a * b);
    let [x, y] = [Math.abs(a), Math.abs(b)];
    while (y) [x, y] = [y, x % y];
    return this._fingerprint(x === 0 ? 0 : product / x);
  }

  factorial(n) {
    if (n < 0 || n > 20) return { value: null, error: 'Out of safe range (0–20)' };
    let r = 1;
    for (let i = 2; i <= n; i++) r *= i;
    return this._fingerprint(r);
  }

  fibonacci(n) {
    if (n < 0 || n > 78) return { value: null, error: 'Out of safe range (0–78)' };
    let [a, b] = [0, 1];
    for (let i = 0; i < n; i++) [a, b] = [b, a + b];
    return this._fingerprint(a);
  }

  // v5.2 UBP-native: delegates to isPrimeUBP (isqrt limit, Golay residue gate)
  isPrime(n) {
    return isPrimeUBP(n).prime;
  }

  modpow(base, exp, mod) {
    if (mod === 1) return this._fingerprint(0);
    let result = 1n;
    let b = BigInt(base) % BigInt(mod);
    let e = BigInt(exp);
    const m = BigInt(mod);
    while (e > 0n) {
      if (e % 2n === 1n) result = (result * b) % m;
      e >>= 1n;
      b = (b * b) % m;
    }
    return this._fingerprint(Number(result));
  }

  comb(n, k) {
    if (k < 0 || k > n) return this._fingerprint(0);
    if (k === 0 || k === n) return this._fingerprint(1);
    let r = 1;
    for (let i = 0; i < Math.min(k, n - k); i++) r = r * (n - i) / (i + 1);
    return this._fingerprint(Math.round(r));
  }

  perm(n, k) {
    let r = 1;
    for (let i = 0; i < k; i++) r *= (n - i);
    return this._fingerprint(r);
  }

  // v5.2 UBP-native: Newton-Raphson integer sqrt — no Math.sqrt (float-free)
  isqrt(n) {
    if (n < 0) return { value: null, error: 'Negative input' };
    return this._fingerprint(isqrtExact(n));
  }

  sumSeries(n) {
    return this._fingerprint(n * (n + 1) / 2);
  }

  extGcd(a, b) {
    let [old_r, r]   = [a, b];
    let [old_s, s]   = [1, 0];
    let [old_t, t]   = [0, 1];
    while (r !== 0) {
      const q = Math.trunc(old_r / r);
      [old_r, r] = [r, old_r - q * r];
      [old_s, s] = [s, old_s - q * s];
      [old_t, t] = [t, old_t - q * t];
    }
    return { gcd: old_r, x: old_s, y: old_t, fp: this._fingerprint(old_r) };
  }

  modInverse(a, m) {
    const { gcd, x } = this.extGcd(a, m);
    if (gcd !== 1) return { value: null, error: `gcd(${a},${m})=${gcd} ≠ 1: no inverse` };
    return this._fingerprint(((x % m) + m) % m);
  }

  mean(data) {
    const r = data.reduce((s, x) => s + x, 0) / data.length;
    return this._fingerprint(Math.round(r * 1e8) / 1e8);
  }

  variance(data) {
    const mu = data.reduce((s, x) => s + x, 0) / data.length;
    const r  = data.reduce((s, x) => s + (x - mu) ** 2, 0) / data.length;
    return this._fingerprint(Math.round(r * 1e8) / 1e8);
  }

  dotProduct(u, v) {
    return this._fingerprint(u.reduce((s, a, i) => s + a * v[i], 0));
  }

  vectorMag(v) {
    return this._fingerprint(Math.sqrt(v.reduce((s, x) => s + x * x, 0)));
  }

  // ── v4 new: Leech / Monster / Triad ─────────────────────────────────────────

  leechInfo(n) {
    const v    = this._toVec24(n);
    const hw   = v.reduce((s, b) => s + b, 0);
    const near = _leech.nearestOctadIdx(v);
    const oct  = _golay.getOctads()[near.idx];
    const pts  = _leech.expandOctad(oct);
    const ranked = _leech.rankByStability(pts);
    const best = ranked[0].point;
    const health = _leech.ontologicalHealth(best);
    return {
      input:               n,
      gray_bits_hw:        hw,
      nearest_octad_idx:   near.idx,
      nearest_octad_dist:  near.distance,
      leech_points:        pts.length,
      norm_sq_scaled:      _leech.normSqScaled(best),
      most_stable_tax:     parseFloat(ranked[0].tax.toFixed(6)),
      least_stable_tax:    parseFloat(ranked[ranked.length - 1].tax.toFixed(6)),
      ontological_health:  {
        Reality:     parseFloat(health.Reality.toFixed(4)),
        Info:        parseFloat(health.Info.toFixed(4)),
        Activation:  parseFloat(health.Activation.toFixed(4)),
        Potential:   parseFloat(health.Potential.toFixed(4)),
        Global_NRCI: parseFloat(health.Global_NRCI.toFixed(4)),
      },
    };
  }

  bwAudit(n, dim = 256) {
    const v           = this._toVec24(n);
    const hw          = v.reduce((s, b) => s + b, 0);
    const { nrci: mn} = microNrciExact(hw); // v5.2: CF-π derived Y_LEECH
    return _bw.audit(v, mn, dim);
  }

  monsterInfo(idxOrName) {
    if (typeof idxOrName === 'number') return SPORADIC[((idxOrName % 26) + 26) % 26];
    const g = SPORADIC.find(s => s.n === String(idxOrName));
    return g || { error: `Unknown group: ${idxOrName}` };
  }

  triadSnapshot() {
    return triadState(this._opCountStable, SPORADIC.length);
  }

  leechStats() { return _leech.stats(); }
  monsterWalk(seedIdx, count) {
    const n = SPORADIC.length;
    return Array.from({ length: count }, (_, i) => SPORADIC[((seedIdx % n) + i + n) % n]);
  }

  // ── NL query router (v4 expanded) ───────────────────────────────────────────
  runMathQuery(query) {
    const q = query.toLowerCase().trim();

    // Factorial
    const factM = q.match(/factorial\s+(?:of\s+)?(\d+)|(\d+)!/);
    if (factM) {
      const n = parseInt(factM[1] || factM[2]);
      const r = this.factorial(n);
      return { op: 'FACTORIAL', inputs: [n], result: r, program: `FACTORIAL(${n})` };
    }

    // Fibonacci
    const fibM = q.match(/fib(?:onacci)?\s*(?:of\s+)?(?:\()?\s*(\d+)/);
    if (fibM) {
      const n = parseInt(fibM[1]);
      const r = this.fibonacci(n);
      return { op: 'FIBONACCI', inputs: [n], result: r, program: `FIBONACCI(${n})` };
    }

    // Extended GCD (before plain GCD)
    if (/extended\s+gcd/.test(q)) {
      const m = q.match(/(\d+)[^\d]+(\d+)/);
      if (m) {
        const [a, b] = [parseInt(m[1]), parseInt(m[2])];
        const r = this.extGcd(a, b);
        return { op: 'EXT_GCD', inputs: [a, b], result: r, program: `EXT_GCD(${a}, ${b})` };
      }
    }

    // Modular inverse
    if (/mod(?:ular)?\s+inverse|modinv/.test(q)) {
      const m = q.match(/(\d+)[^\d]+(\d+)/);
      if (m) {
        const [a, mod] = [parseInt(m[1]), parseInt(m[2])];
        const r = this.modInverse(a, mod);
        return { op: 'MOD_INV', inputs: [a, mod], result: r, program: `MOD_INV(${a}, ${mod})` };
      }
    }

    // LCM
    if (/\blcm\b/.test(q)) {
      const m = q.match(/(\d+)[^\d]+(\d+)/);
      if (m) {
        const [a, b] = [parseInt(m[1]), parseInt(m[2])];
        const r = this.lcm(a, b);
        return { op: 'LCM', inputs: [a, b], result: r, program: `LCM(${a}, ${b})` };
      }
    }

    // GCD — matches: gcd(48,18)  gcd 48 18  gcd of 48 and 18
    const gcdM = q.match(/gcd\s*[\(\s,of]*\s*(\d+)\s*[,\s)and]+\s*(\d+)/);
    if (gcdM) {
      const [a, b] = [parseInt(gcdM[1]), parseInt(gcdM[2])];
      const r = this.gcd(a, b);
      return { op: 'GCD', inputs: [a, b], result: r, program: `GCD(${a}, ${b})` };
    }

    // Modpow
    const modpowM = q.match(/(\d+)\s*\^\s*(\d+)\s*mod\s*(\d+)|modpow\s*\(?(\d+)[,\s]+(\d+)[,\s]+(\d+)/);
    if (modpowM) {
      const [base, exp, mod] = modpowM[1]
        ? [parseInt(modpowM[1]), parseInt(modpowM[2]), parseInt(modpowM[3])]
        : [parseInt(modpowM[4]), parseInt(modpowM[5]), parseInt(modpowM[6])];
      const r = this.modpow(base, exp, mod);
      return { op: 'MODPOW', inputs: [base, exp, mod], result: r, program: `MODPOW(${base}, ${exp}, ${mod})` };
    }

    // Primality — v5.2 UBP-native (isqrt limit, Golay mod-24 gate)
    const primeM = q.match(/(?:is\s+)?(\d+)\s*prime|prime\s*(?:test|check)?\s*(\d+)/);
    if (primeM) {
      const n = parseInt(primeM[1] || primeM[2]);
      const ubpResult = isPrimeUBP(n);
      return {
        op: 'PRIMALITY_UBP', inputs: [n],
        result: {
          value:          ubpResult.prime,
          lattice:        ubpResult.prime ? 'Prime' : 'Composite',
          mode:           ubpResult.method,
          substrate_note: ubpResult.substrate_note,
          golay_gate:     ubpResult.golay_gate,
          isqrt_limit:    ubpResult.isqrt_limit,
        },
        program: `PRIMALITY_UBP(${n})`
      };
    }

    // Permutation (before combination so "perm" keyword is caught)
    if (/perm(?:utation)?/.test(q)) {
      const m = q.match(/(\d+)[^\d]+(\d+)/);
      if (m) {
        const [n, k] = [parseInt(m[1]), parseInt(m[2])];
        const r = this.perm(n, k);
        return { op: 'PERM', inputs: [n, k], result: r, program: `PERM(${n}, ${k})` };
      }
    }

    // Combination / binomial
    const chooseM = q.match(/(?:c|choose|comb(?:ination)?)\s*\(?(\d+)[,\s]+(\d+)/);
    if (chooseM) {
      const [n, k] = [parseInt(chooseM[1]), parseInt(chooseM[2])];
      const r = this.comb(n, k);
      return { op: 'CHOOSE', inputs: [n, k], result: r, program: `CHOOSE(${n}, ${k})` };
    }

    // Integer square root — v5.2 UBP-native (Newton-Raphson, no Math.sqrt)
    if (/isqrt|integer\s+(?:square\s+)?root/.test(q)) {
      const m = q.match(/(\d+)/);
      if (m) {
        const n = parseInt(m[1]);
        const r = this.isqrt(n);
        return { op: 'ISQRT_UBP', inputs: [n], result: r, program: `ISQRT_UBP(${n})` };
      }
    }

    // Sum series 1..n
    if (/sum\s+(?:of\s+)?(?:integers?|series|1\s+to)/.test(q)) {
      const m = q.match(/(\d+)\s*$/);
      if (m) {
        const n = parseInt(m[1]);
        const r = this.sumSeries(n);
        return { op: 'SUM_SERIES', inputs: [n], result: r, program: `SUM_1_TO_${n}` };
      }
    }

    // Mean
    if (/\bmean\b|\baverage\b/.test(q)) {
      const nums = (q.match(/[\d.]+/g) || []).map(Number);
      if (nums.length >= 2) {
        const r = this.mean(nums);
        return { op: 'MEAN', inputs: nums, result: r, program: `MEAN([${nums.join(',')}])` };
      }
    }

    // Variance
    if (/\bvariance\b/.test(q)) {
      const nums = (q.match(/[\d.]+/g) || []).map(Number);
      if (nums.length >= 2) {
        const r = this.variance(nums);
        return { op: 'VARIANCE', inputs: nums, result: r, program: `VAR([${nums.join(',')}])` };
      }
    }

    // Dot product
    if (/dot\s+product/.test(q)) {
      const vecs = [...q.matchAll(/<([^>]+)>/g)].map(m => m[1].split(/[,\s]+/).map(Number));
      if (vecs.length >= 2) {
        const r = this.dotProduct(vecs[0], vecs[1]);
        return { op: 'DOT', inputs: vecs, result: r, program: `DOT(${JSON.stringify(vecs[0])}, ${JSON.stringify(vecs[1])})` };
      }
    }

    // Vector magnitude
    if (/magnitude|length of vector/.test(q)) {
      const vec = (q.match(/<([^>]+)>/) || [])[1];
      if (vec) {
        const v = vec.split(/[,\s]+/).map(Number);
        const r = this.vectorMag(v);
        return { op: 'MAG', inputs: v, result: r, program: `MAG(${JSON.stringify(v)})` };
      }
    }

    // Leech info
    if (/leech\s+info|leech\s+lattice\s+for/.test(q)) {
      const m = q.match(/(\d+)/);
      if (m) {
        const n = parseInt(m[1]);
        const r = this.leechInfo(n);
        return { op: 'LEECH_INFO', inputs: [n], result: r, program: `LEECH_INFO(${n})` };
      }
    }

    // BW audit
    if (/bw\s*(?:256)?\s*audit|barnes.wall\s+audit/.test(q)) {
      const m = q.match(/(\d+)/);
      if (m) {
        const n = parseInt(m[1]);
        const r = this.bwAudit(n);
        return { op: 'BW_AUDIT', inputs: [n], result: r, program: `BW256_AUDIT(${n})` };
      }
    }

    // Monster info
    if (/monster|sporadic/.test(q)) {
      const m = q.match(/\b(M\d{0,4}|Co[0-9]|Fi\d{2}|B|HS|McL|He|Suz|HN|Th|J[1-4]|Ru|Ly|ON)\b/i);
      if (m) {
        const r = this.monsterInfo(m[1]);
        return { op: 'MONSTER', inputs: [m[1]], result: r, program: `MONSTER_INFO(${m[1]})` };
      }
    }

    // Triad snapshot
    if (/triad/.test(q)) {
      const r = this.triadSnapshot();
      return { op: 'TRIAD', inputs: [], result: r, program: 'TRIAD_SNAPSHOT()' };
    }

    // Simple arithmetic
    const arithM = q.match(/(\d+(?:\.\d+)?)\s*([\+\-\*\/x×÷])\s*(\d+(?:\.\d+)?)/);
    if (arithM) {
      const [a, op, b] = [parseFloat(arithM[1]), arithM[2], parseFloat(arithM[3])];
      let r;
      if (op === '+')               r = this.add(a, b);
      else if (op === '-')          r = this.sub(a, b);
      else if (/[*x×]/.test(op))   r = this.mul(a, b);
      else if (/[/÷]/.test(op))    r = this.divmod(a, b).quotient || this._fingerprint(a / b);
      return { op, inputs: [a, b], result: r, program: `${a} ${op} ${b}` };
    }

    return null;
  }
}

// ── Syndrome table pre-warmer (call during engine init to avoid first-query jank) ──
export function warmSyndromeTable() {
  // Triggers the lazy 2325-entry syndrome table build synchronously.
  // Cost: ~5ms once; saves ~5ms on first user math query.
  _golay.snapToCodeword(new Array(24).fill(0));
}
