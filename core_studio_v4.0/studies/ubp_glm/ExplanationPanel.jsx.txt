function CodeBlock({ item }) {
  const [copied, setCopied] = useState(false);
  const copy = () => {
    navigator.clipboard.writeText(item.code);
    setCopied(true);
    setTimeout(() => setCopied(false), 1500);
  };
  return (
    <div className="mb-3 last:mb-0">
      <div className="flex items-center justify-between mb-1">
        <span className="text-xs font-medium text-slate-400">{item.label}</span>
        <button onClick={copy} className="text-slate-500 hover:text-slate-300 transition-colors">
          {copied ? <Check className="w-3 h-3 text-green-400" /> : <Copy className="w-3 h-3" />}
        </button>
      </div>
      <pre className="text-xs text-violet-300 bg-slate-950/80 rounded-lg p-3 overflow-x-auto whitespace-pre font-mono leading-relaxed border border-slate-700/40">{item.code}</pre>
    </div>
  );
