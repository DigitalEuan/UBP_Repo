function encodeGolay(msg12) {
  const parity = new Array(12).fill(0);
  for (let i = 0; i < 12; i++)
    for (let j = 0; j < 12; j++)
      parity[i] = (parity[i] + msg12[j] * GOLAY_B[j][i]) % 2;
  return [...msg12, ...parity];
}

// ── Helpers ───────────────────────────────────────────────────────────────────
function extractVector(entry) {
  // v9.9: vector is a direct field (no atlas wrapper)
  if (Array.isArray(entry.vector) && entry.vector.length === 24) return entry.vector;
  // legacy fallback
  const a = entry.atlas;
  if (a && Array.isArray(a.vector) && a.vector.length === 24) return a.vector;
  return null;
}

function extractNrci(entry) {
  // v9.9: nrci_val is a direct float field
  if (typeof entry.nrci_val === 'number') return Math.min(1, Math.max(0, entry.nrci_val));
  // legacy fallback
  const a = entry.atlas || {};
  if (typeof a.nrci_score === 'number') return a.nrci_score;
  if (typeof a.nrci === 'string' && a.nrci.includes('/')) {
    const [n, d] = a.nrci.split('/').map(Number);
    return d > 0 ? Math.min(1, n / d) : 0.5;
  }
  return 0.5;
}

export function extractName(entry) {
  const lex = entry.lexicon || '';
  const m = lex.match(/^\[\s*([^\]]+?)\s*\]/);
  if (m) {
    let c = m[1].trim();
    if (c.includes(':')) c = c.split(':', 2)[1].trim();
    return c.replace(/\s*\([^)]*\)\s*$/, '').trim() || (entry.ubp_id || '?').replace(/_/g, ' ');
  }
  return (entry.ubp_id || '?').replace(/_/g, ' ');
}

function extractDescription(entry) {
  const lex = entry.lexicon || '';
  const parts = [...lex.matchAll(/\[([^\]]+)\]/g)].map(m => m[1].trim());
  if (parts.length >= 2) return parts[1];
  return lex;
}

function extractMathData(entry) {
  // v9.9: no math string — properties are in mog_tensor (complex sparse array).
  // NL-to-math extraction handles query-side; entry-side math display is deprecated.
  if (entry.math) {
    const d = {};
    entry.math.split('|').forEach(p => {
      const [k, v] = p.split('=');
      if (k && v) d[k.trim()] = v.trim();
    });
    return d;
  }
  return {};
}

function inferDomain(entry) {
  const uid = entry.ubp_id || '';
  if (/^ELEM_|^MOLECULE_|^CRYSTAL_/.test(uid)) return 'SUBSTANCE';
  if (/^PARTICLE_|^REACTION_/.test(uid)) return 'MECHANISM';
  if (/^MATH_|^ALGO_/.test(uid)) return 'ALGORITHM';
  if (/^LAW_|^BELIEF_|^AXIOM_|^IMPERATIVE_/.test(uid)) return 'IMPERATIVE';
  if (/^OP_/.test(uid)) return 'MEANING';
  const tags = (entry.tags || []).join(' ').toUpperCase();
  if (/SUBSTANCE|ELEMENT|MOLECULE/.test(tags)) return 'SUBSTANCE';
  if (/MECHANISM|PHYSICS|QUANTUM|PARTICLE/.test(tags)) return 'MECHANISM';
  if (/ORGANISM|BIO|LIFE/.test(tags)) return 'ORGANISM';
  if (/ALGORITHM|COMPUTE/.test(tags)) return 'ALGORITHM';
  if (/QUANTITY|MATH|CONSTANT/.test(tags)) return 'QUANTITY';
  if (/IMPERATIVE|LAW/.test(tags)) return 'IMPERATIVE';
  if (/ENTROPY|RANDOM/.test(tags)) return 'ENTROPY';
  return 'MEANING';
}

// ── Gray-code fallback for unknown words (v7.3 — topological continuity) ────────
// Replaces DJB2 hash which had avalanche effect (similar words → distant vectors).
// UBP seed schema: [Domain:3][Magnitude:5][State:4] = 12-bit message
//
// Topological continuity guarantee:
//   • Magnitude bits (Gray-encoded word-length) → length-similar words stay adjacent
//   • State bits (Gray-encoded char-density sum) → lexically similar words stay adjacent
//   • No avalanche: single-char difference → at most 1-2 bit flip in the 24D manifold
function inferDomainBits(word) {
  if (/force$|energy$|wave$|field$|gravity|quantum|photon|boson|kinetic/.test(word)) return [0,0,1];
  if (/sis$|biology|cell|organism|life|neural|protein/.test(word)) return [0,1,0];
  if (/algorithm|code|compute|function|logic|binary/.test(word)) return [0,1,1];
  if (/math|number|ratio|constant|sqrt/.test(word)) return [1,0,0];
  if (/law$|rule$|axiom|principle/.test(word)) return [1,0,1];
  if (/entropy|chaos|decay|random/.test(word)) return [1,1,0];
  if (/ium$|ogen$|ine$/.test(word) && word.length <= 8) return [0,0,0];
  return [1,1,1];
}

function wordToVector(word) {
  const w = word.toLowerCase();
  const dom = inferDomainBits(w); // [D0, D1, D2] — ontological domain

  // Magnitude: 5 bits — Gray-encode word length (0-31)
  // Lexically similar words (same length class) share Magnitude bits
  const lenVal = Math.min(w.length, 31);
  const grayLen = lenVal ^ (lenVal >> 1);
  const mag = [];
  for (let i = 4; i >= 0; i--) mag.push((grayLen >> i) & 1);

  // State: 4 bits — Gray-encode character density (sum of char indices mod 16)
  // Similar prefix/suffix patterns produce nearby State values
  let charSum = 0;
  for (let i = 0; i < Math.min(w.length, 8); i++) {
    const c = w.charCodeAt(i);
    if (c >= 97 && c <= 122) charSum += (c - 96); // a=1..z=26, continuous
  }
  const stateVal = charSum % 16;
  const grayState = stateVal ^ (stateVal >> 1);
  const state = [];
  for (let i = 3; i >= 0; i--) state.push((grayState >> i) & 1);

  // Compose [Domain:3][Magnitude:5][State:4] = 12-bit message → 24-bit Golay codeword
  return encodeGolay([...dom, ...mag, ...state]);
}

// ── Cosine similarity (v8.0 — replaces bipolar dot for semantic queries) ─────
function cosineSim(a, b) {
  if (!a || !b || a.length !== b.length) return 0;
  let dot = 0, magA = 0, magB = 0;
  for (let i = 0; i < a.length; i++) {
    const ba = 2 * a[i] - 1, bb = 2 * b[i] - 1;
    dot += ba * bb; magA += ba * ba; magB += bb * bb;
  }
  if (magA === 0 || magB === 0) return 0;
  return dot / (Math.sqrt(magA) * Math.sqrt(magB));
}

// Kept for Hamming distance reporting only
function bipolarDot(a, b) {
  if (!a || !b || a.length !== b.length) return 0;
  let s = 0;
  for (let i = 0; i < a.length; i++) s += (2 * a[i] - 1) * (2 * b[i] - 1);
  return s / a.length;
}

function hammingDistance(a, b) {
  let d = 0;
  for (let i = 0; i < a.length; i++) if (a[i] !== b[i]) d++;
  return d;
}

// ── KB indexes ────────────────────────────────────────────────────────────────
function buildIndexes(kbData) {
  const shortNameIndex = new Map();
  const bigramIndex    = new Map();
  const trigramIndex   = new Map();
  const lexiconIndex   = new Map();
  const tagIndex       = new Map(); // tag -> [uid, ...]
  const uidToEntry     = new Map(); // O(1) lookup by ubp_id
  const KEEP_SHORT = new Set(['w', 'z', 'h', 'e', 'p', 'n']);

  for (const [fp, entry] of Object.entries(kbData)) {
    const vec = extractVector(entry);
    if (!vec) continue;
    const uid = entry.ubp_id;
    if (!uid) continue;
    entry.fingerprint = fp;
    uidToEntry.set(uid, entry);

    const lex = (entry.lexicon || '').toLowerCase();
    const chunks = lex.split('[').filter(c => c.includes(']')).map(c => c.split(']')[0]);
    for (const chunk of chunks) {
      let name = chunk.split(':').slice(-1)[0].trim().split('(')[0].trim();
      if (!name) continue;
      for (const v of new Set([name, name.replace(/-/g, ' ')])) {
        if (!shortNameIndex.has(v)) shortNameIndex.set(v, uid);
        const words = v.split(/\s+/).filter(Boolean);
        if (words.length >= 3 && !trigramIndex.has(words.slice(0,3).join(' ')))
          trigramIndex.set(words.slice(0,3).join(' '), uid);
        if (words.length >= 2 && !bigramIndex.has(words.slice(0,2).join(' ')))
          bigramIndex.set(words.slice(0,2).join(' '), uid);
        for (const w of words) {
          if (w.length > 2 || KEEP_SHORT.has(w)) {
            if (!lexiconIndex.has(w)) lexiconIndex.set(w, []);
            lexiconIndex.get(w).push(uid);
          }
        }
      }
    }
    for (const tag of (entry.tags || [])) {
      const tl = tag.toLowerCase();
      if (!shortNameIndex.has(tl)) shortNameIndex.set(tl, uid);
      // Tag index: tag -> [uid] for 5+ char tag matching
      if (!tagIndex.has(tl)) tagIndex.set(tl, []);
      tagIndex.get(tl).push(uid);
    }
  }
  return { shortNameIndex, bigramIndex, trigramIndex, lexiconIndex, tagIndex, uidToEntry };
}

// ── Reflexive auto-trigger scan ────────────────────────────────────────────────
// Law/belief/axiom UID prefixes — these get reduced reflexive boost to avoid flooding
const LAW_PREFIXES = /^(LAW_|BELIEF_|AXIOM_|IMPERATIVE_)/;
// Keywords that suggest the user IS asking about laws/principles
const LAW_KEYWORDS = /\b(law|axiom|principle|theorem|imperative|belief|conservation|rule|postulate|conjecture)\b/i;

function reflexiveScan(query, kbData, shortNameIndex, tagIndex, uidToEntry) {
  const memories = new Map();
  const il = query.toLowerCase();
  const isLawQuery = LAW_KEYWORDS.test(query);
  const getE = uid => uidToEntry ? uidToEntry.get(uid) : Object.values(kbData).find(e => e.ubp_id === uid);

  for (const [phrase, uid] of shortNameIndex) {
    if (phrase.length > 2 && il.includes(phrase) && !memories.has(uid)) {
      const entry = getE(uid);
      if (entry) {
        const isLawEntry = LAW_PREFIXES.test(uid);
        const boost = (isLawEntry && !isLawQuery) ? 0.3 : 2.0;
        memories.set(uid, { entry, boost });
      }
    }
  }

  // Tag index recall: 5+ char query words matched against tag index
  const queryWords = (il.match(/\b\w{5,}\b/g) || []);
  for (const word of queryWords) {
    const tagUids = tagIndex ? tagIndex.get(word) : null;
    if (tagUids) {
      for (const uid of tagUids) {
        if (!memories.has(uid)) {
          const entry = getE(uid);
          if (entry) memories.set(uid, { entry, boost: 1.2 });
        }
      }
    }
  }

  const idMatches = query.match(/\b[A-Z]+_[A-Z0-9_]+_\d+\b/g) || [];
  for (const uid of idMatches) {
    if (!memories.has(uid)) {
      const entry = getE(uid);
      if (entry) memories.set(uid, { entry, boost: 1.5 });
    }
  }
  if (memories.size < 5) {
    const words = (il.match(/\b\w{4,}\b/g) || []);
    const allEntries = uidToEntry ? [...uidToEntry.values()] : Object.values(kbData);
    outer: for (const word of words) {
      for (const entry of allEntries) {
        const uid = entry.ubp_id;
        if (uid && !memories.has(uid) &&
            ((entry.lexicon || '').toLowerCase().includes(word) ||
             (uid || '').toLowerCase().includes(word))) {
          memories.set(uid, { entry, boost: 1.0 });
          if (memories.size > 12) break outer;
        }
      }
    }
  }
  return memories;
}

// ── N-gram tokenisation ───────────────────────────────────────────────────────
function tokenise(query, kbData, shortNameIndex, bigramIndex, trigramIndex, lexiconIndex, uidToEntry) {
  const words = query.toLowerCase().replace(/[^a-z0-9\s\-]/g, ' ')
    .split(/\s+/).filter(w => w && !STOP_WORDS.has(w));
  const consumed = new Set();
  const matchedVectors = [];
  const corroboration = new Map();
  // Use O(1) uidToEntry map instead of O(N) scan
  const getEntry = uid => uidToEntry ? uidToEntry.get(uid) : Object.values(kbData).find(e => e.ubp_id === uid);

  // Trigrams (weight 9x — brain v7.2)
  for (let i = 0; i <= words.length - 3; i++) {
    const key = words.slice(i, i+3).join(' ');
    const uid = trigramIndex.get(key);
    if (uid) { const e = getEntry(uid); const v = e && extractVector(e); if (v) { matchedVectors.push({ vec: v, weight: 9, word: key, uid, type: 'trigram' }); corroboration.set(uid, (corroboration.get(uid)||0)+3); [i,i+1,i+2].forEach(j=>consumed.add(j)); } }
  }
  // Bigrams
  for (let i = 0; i <= words.length - 2; i++) {
    if (consumed.has(i)) continue;
    const key = words.slice(i, i+2).join(' ');
    const uid = bigramIndex.get(key);
    if (uid) { const e = getEntry(uid); const v = e && extractVector(e); if (v) { matchedVectors.push({ vec: v, weight: 2, word: key, uid, type: 'bigram' }); corroboration.set(uid, (corroboration.get(uid)||0)+2); [i,i+1].forEach(j=>consumed.add(j)); } }
  }
  // Unigrams
  for (let i = 0; i < words.length; i++) {
    if (consumed.has(i)) continue;
    const w = words[i];
    const uid = shortNameIndex.get(w);
    if (uid) { const e = getEntry(uid); const v = e && extractVector(e); if (v) { matchedVectors.push({ vec: v, weight: 1, word: w, uid, type: 'unigram-kb' }); corroboration.set(uid, (corroboration.get(uid)||0)+1); consumed.add(i); continue; } }
    const uidList = lexiconIndex.get(w);
    if (uidList?.length) { const lu = uidList[0]; const e = getEntry(lu); const v = e && extractVector(e); if (v) { matchedVectors.push({ vec: v, weight: 0.8, word: w, uid: lu, type: 'lexicon' }); corroboration.set(lu, (corroboration.get(lu)||0)+1); consumed.add(i); continue; } }
    if (w.length > 2) matchedVectors.push({ vec: wordToVector(w), weight: 0.5, word: w, uid: null, type: 'geometric-hash' });
  }
  return { matchedVectors, corroboration, words };
}

function buildQueryVector(matchedVectors) {
  if (!matchedVectors.length) return null;
  const totalWeight = matchedVectors.reduce((s, t) => s + t.weight, 0);
  if (!totalWeight) return null;
  const sum = new Array(24).fill(0);
  for (const { vec, weight } of matchedVectors)
    for (let i = 0; i < 24; i++) sum[i] += vec[i] * weight;
  return sum.map(s => s / totalWeight >= 0.5 ? 1 : 0);
}

// ── Domain intent detection (v7.0 — adds particle, scripting, math domains) ──
function detectDomainIntent(query) {
  const q = query.toLowerCase();
  const physical  = /\b(speed|velocity|mass|weight|boiling|melting|density|radius|energy|temperature|pressure|charge|frequency|wavelength|force|momentum|element|molecule|atom|compound|substance|liquid|solid|gas|metal|crystal)\b/.test(q);
  const symbolic  = /\b(what|why|explain|define|how does|meaning|concept|theory|principle|golay|nrci|ubp|leech|observer|lattice|code|axiom|law|belief)\b/.test(q);
  const particle  = /\b(quark|boson|lepton|neutrino|particle|higgs|graviton|fermion|photon|gluon|meson|hadron|baryon|electron|proton|neutron)\b/.test(q);
  const scripting = /\b(python|script|function|def |class |import|loop|list|dict|algorithm|code|program|syntax|variable|iterate|recursion|lambda|async|await)\b/.test(q);
  const mathDeep  = /\b(prime|factorial|fibonacci|modular|matrix|vector|integral|derivative|eigenvalue|polynomial|sequence|series|permutation|combinat)\b/.test(q);
  return { physical, symbolic, particle, scripting, mathDeep };
}

// ── Identity Lock (from ubp_brain_consolidated.py v7.2) ──────────────────────
// Trigram → Bigram → Unigram priority cascade with stop-word gate.
// Returns the locked UID immediately when a high-confidence n-gram match is found
// and the UID prefix validates against detected domain intent.
function identityLock(words, trigramIndex, bigramIndex, shortNameIndex, uidToEntry, domainIntent) {
  // n-gram priority: 3 → 2 → 1
  const levels = [
    { n: 3, index: trigramIndex, confidence: 1.0 },
    { n: 2, index: bigramIndex,  confidence: 0.95 },
    { n: 1, index: shortNameIndex, confidence: 0.90 },
  ];

  for (const { n, index, confidence } of levels) {
    for (let i = 0; i <= words.length - n; i++) {
      const phrase = words.slice(i, i + n).join(' ');
      // Stop-word gate: single-word match on a stop word is not a valid lock
      if (n === 1 && STOP_WORDS.has(phrase)) continue;

      const uid = index.get ? index.get(phrase) : index[phrase];
      if (!uid) continue;

      // Domain validation (precision gating from ubp_brain_consolidated.py)
      if (domainIntent.particle  && !/^PARTICLE_/.test(uid)) continue;
      if (domainIntent.scripting && !/^(ALGO_|MATH_)/.test(uid) && !domainIntent.symbolic) continue;
      // Physical queries: don't let OP_ entries steal the lock
      if (domainIntent.physical && !domainIntent.symbolic && /^OP_/.test(uid)) continue;

      const entry = uidToEntry ? uidToEntry.get(uid) : null;
      if (!entry) continue;

      return { uid, confidence, method: `Identity Lock (${n}-gram: "${phrase}")`, entry };
    }
  }
  return null;
}

// ── Dual-channel scoring (v7.0 — strict domain gating + semantic reflection) ──
function scoreAll(queryVec, queryNumericVec, queryPresenceMask, kbData, corroboration, fomManager, alpha, isLawQuery, domainIntent) {
  const physicalQuery  = domainIntent?.physical  && !domainIntent?.symbolic;
  const particleQuery  = domainIntent?.particle;
  const scriptingQuery = domainIntent?.scripting && !domainIntent?.symbolic;
  const results = [];
  for (const [fp, entry] of Object.entries(kbData)) {
    const golayVec = extractVector(entry);
    if (!golayVec) continue;

    const uid = entry.ubp_id || '';

    // ── Strict Domain Gating (ubp_brain_consolidated.py precision gating) ────
    // Particle queries: only PARTICLE_ entries pass
    if (particleQuery && !/^PARTICLE_/.test(uid) && !/^(ELEM_|MOLECULE_)/.test(uid)) continue;
    // Physical queries: OP_ entries are suppressed
    if (physicalQuery && /^OP_/.test(uid)) continue;
    // Scripting/code queries: boost ALGO_/MATH_, suppress IMPERATIVE_ and OP_
    if (scriptingQuery && /^(LAW_|BELIEF_|AXIOM_|IMPERATIVE_|OP_)/.test(uid)) continue;

    // Channel 1: Shadow-weighted cosine (v9.0 — noumenal/phenomenal split)
    // For symbolic queries (alpha→1): noumenal bits weighted higher (structural meaning)
    // For property queries (alpha→0): phenomenal bits weighted higher (observable props)
    const semSim = queryVec ? ShadowProcessor.shadowCosineSim(queryVec, golayVec, alpha) : 0;
    if (alpha > 0 && semSim < 0.05) continue;

    // Channel 2: Numeric (Gray-code masked Hamming)
    let numericSim = 0;
    if (alpha < 1.0 && queryNumericVec) {
      const { vector: entryNumVec } = getEntryNumericVector(entry);
      const d = maskedHammingDistance(queryNumericVec, entryNumVec, queryPresenceMask);
      numericSim = 1 - d;
    }

    // Early reject: both channels weak
    const raw = alpha * Math.max(semSim, 0) + (1 - alpha) * numericSim;
    if (raw < 0.05 && !(corroboration.get(uid) > 0)) continue;

    const domain = inferDomain(entry);
    const nrci = extractNrci(entry);
    const fomWeight = fomManager ? fomManager.getWeight(uid, domain) : 0.5;
    const nrciWeighted = nrci * (0.5 + 0.5 * fomWeight);
    const corrobBoost = corroboration.get(uid) || 0;

    // Penalise IMPERATIVE/LAW/BELIEF entries strongly when query is not about laws
    // This prevents "Law of Glycemic Parity" etc. from dominating general queries
    const isLawEntry = /^(LAW_|BELIEF_|AXIOM_|IMPERATIVE_)/.test(uid);
    const domainPenalty = (!isLawQuery && isLawEntry) ? 0.10 : 1.0;
    // Matter dominance (v7.0 — aligned with ubp_semantic_engine.py weighted probes)
    // ELEM_/MOLECULE_/PARTICLE_ = 10x, ALGO_/MATH_ = 3x on scripting queries, OP_ = 0.5x
    const matterBoost =
      /^(ELEM_|MOLECULE_|PARTICLE_)/.test(uid) ? (physicalQuery || particleQuery ? 2.5 : 1.4) :
      /^(ALGO_|MATH_)/.test(uid)               ? (scriptingQuery ? 3.0 : 1.2) :
      /^OP_/.test(uid)                          ? 0.5 : 1.0;

    // TGIC 3-6-9 stability bonus: measures geometric harmony of the entry's vector
    // ortho (3-axis), face (6-face coherence), no manifold pressure at score time
    const tgicStability = TGICConstraintSystem.auditStability(golayVec, [], nrciWeighted);
    // DQI: combines NRCI + semantic match + gap coverage (gap=1 when corroborated)
    const dqi = DQIMetrics.calculateDQI(nrciWeighted, Math.max(semSim, numericSim), corrobBoost > 0 ? 1.0 : 0.5);
    const score = (corrobBoost * 10 + raw * (0.5 + 0.5 * dqi)) * domainPenalty * matterBoost * (1 + 0.1 * tgicStability);

    // Barnes-Wall fingerprint (precomputed lazily — only for top candidates)
    results.push({ entry, fingerprint: fp, uid, semSim, numericSim, raw, score, nrci, corrobBoost, domain, alpha });
  }
  results.sort((a, b) => b.score - a.score);

  // ── Channel 3: Barnes-Wall 256D refinement for top-5 candidates ──────────
  // Re-score top candidates using Macro-NRCI from the 256D manifold
  // This resolves ties that 24D Golay cosine similarity cannot distinguish.
  if (queryVec && results.length > 1) {
    const TOP_N = Math.min(5, results.length);
    for (let i = 0; i < TOP_N; i++) {
      const r = results[i];
      const v = extractVector(r.entry);
      if (!v) continue;
      try {
        const bwFp = computeUBPFingerprint(v);
        if (bwFp) {
          // Clarity bonus: HIGH clarity = entry sits cleanly in its 256D Voronoi cell
          const clarityBonus = bwFp.clarity === 'HIGH' ? 0.08 : 0;
          // Lattice type bonus: on-lattice entries (Octad/Dodecad etc) are more stable
          const latticeBonus = bwFp.onLattice ? 0.05 : 0;
          r.score *= (1 + clarityBonus + latticeBonus);
          r.bwFp = bwFp;
        }
      } catch { /* non-critical */ }
    }
    results.sort((a, b) => b.score - a.score);
  }

  return results;
}

function computeConfidence(results) {
  if (!results.length) return 0;
  if (results.length === 1) return Math.min(1, results[0].raw);
  const corrobBonus = Math.min(0.4, results[0].corrobBoost * 0.1);
  return Math.min(1, (results[0].score - results[1].score) + corrobBonus);
}

// ── Response synthesis ────────────────────────────────────────────────────────
function synthesise(results, tokenLog, confidence, queryType, mathDict, linguistElaboration, langVectors, phaseResult) {
  if (!results.length) return "No matching KB entries found. Try a specific element, molecule, particle, or a property query like \"liquid with mass 18 g/mol\".";

  const top = results[0];
  const name = extractName(top.entry);
  const desc = extractDescription(top.entry);
  const math = extractMathData(top.entry);
  const lines = [];

  const LABELS = { Z:'Atomic №', M:'Mass (u)', EN:'Electronegativity', Ion:'Ionisation (kJ/mol)',
    BP:'Boiling Pt (K)', MP:'Melting Pt (K)', Valence_e:'Valence e⁻', Phase_STP:'Phase @STP',
    Crystal:'Crystal', Dipole:'Dipole (D)', Density:'Density (g/cm³)', Phase:'Phase', pKa1:'pKa',
    Rad:'Radius (pm)', Rho:'Density (g/cm³)' };

  // Gray-code hex color fingerprint + Barnes-Wall lattice type
  const vec = extractVector(top.entry);
  const hexColor = vec ? grayCodeToHex(vec) : null;
  const bwFp = top.bwFp || (vec ? computeUBPFingerprint(vec) : null);
  const lattice = bwFp ? bwFp.lattice : null;

  // ── 1. Primary match header ───────────────────────────────────────────────
  lines.push(`**${name}**${hexColor ? ` \`${hexColor}\`` : ''}${lattice ? ` · *${lattice}*` : ''}`);
  if (desc) { lines.push(''); lines.push(desc); }

  // Math properties — rich display (up to 15 properties)
  const mathKeys = Object.keys(math);
  if (mathKeys.length) {
    lines.push('');
    lines.push(mathKeys.slice(0, 15).map(k => `**${LABELS[k] || k}**: ${math[k]}`).join(' · '));
  }

  // For property queries, show which properties matched the query
  if (queryType === 'property' && Object.keys(mathDict).length > 0) {
    lines.push('');
    const matched = Object.entries(mathDict).map(([k, v]) => `${LABELS[k] || k}=${typeof v === 'number' ? v.toFixed(3) : v}`);
    lines.push(`*Query extracted: ${matched.join(', ')}*`);
  }

  const hier = top.entry.atlas?.hierarchy;
  if (hier) { lines.push(''); lines.push(`*Composition: ${hier}*`); }

  // Tags as context
  const tags = (top.entry.tags || []).filter(t => !['SOP_002','HARDENED'].includes(t));
  if (tags.length) { lines.push(''); lines.push(`*Tags: ${tags.join(' · ')}*`); }

  // MOG active categories (from auto_trigger.py synth_context)
  const mogCats = getActiveMogCategories(top.entry.mog_tensor);
  if (mogCats.length) {
    lines.push('');
    lines.push(`**MOG active:** ${mogCats.slice(0, 12).join(' · ')}`);
  }

  // Semantic Reflection (from ubp_semantic_engine.py _reflect_meaning)
  if (langVectors && vec && queryType !== 'symbolic') {
    const reflection = reflectMeaning(vec, langVectors);
    if (reflection && !reflection.isLexicalGap) {
      lines.push('');
      lines.push(`**Semantic bridge:** This entity resonates with the geometric operator **${reflection.word}** (${(reflection.score * 100).toFixed(1)}% cosine bridge)`);
    } else if (reflection?.isLexicalGap) {
      lines.push('');
      lines.push(`**Semantic bridge:** ⚠ *Undefined Geometric State* — no OP_ operator within 0.45 cosine threshold (gap score ${(reflection.score * 100).toFixed(1)}%)`);
    }
  }

  // ── 2. Related entries — TGIC Relational Pull ranking (v9.0) ────────────
  // Problem 2 fix: instead of slicing top-N by score, we rank ALL remaining
  // candidates by their Hamming-weighted Relational Pull to the top result.
  // This surfaces the "correct but lower-scored" memory that was previously hidden.
  const topVec = extractVector(top.entry);
  const rawRelated = results.slice(1).filter(r => {
    if (/^(LAW_|BELIEF_|AXIOM_|IMPERATIVE_)/.test(r.uid)) return false;
    return true; // pass all non-law candidates to pull ranking
  });
  // Rank by relational pull to top result
  const pullRanked = TGICRelationalGravity.rankByPull(
    topVec,
    rawRelated.map(r => r.entry),
    extractVector,
    queryType === 'symbolic' ? 12 : 8   // wider pool for symbolic queries
  );
  // Re-attach scoring data and filter low-pull entries
  const pullMap = new Map(rawRelated.map(r => [r.uid, r]));
  const related = pullRanked
    .filter(({ pull }) => pull > Y_LEECH * 0.1)  // minimum pull threshold
    .map(({ entry, pull, dist }) => ({
      ...(pullMap.get(entry.ubp_id) || { entry, uid: entry.ubp_id, raw: 0, corrobBoost: 0, nrci: 0, domain: 'MEANING', alpha: 0.25 }),
      entry,
      relationalPull: pull,
      hammingDist: dist,
    }))
    .slice(0, queryType === 'symbolic' ? 8 : 5);

  if (related.length) {
    lines.push('');
    lines.push(queryType === 'symbolic' ? '**Related concepts** *(TGIC Relational Pull)*:' : '**Geometrically adjacent** *(TGIC Relational Pull)*:');
    for (const r of related) {
      const rVec = extractVector(r.entry);
      const rHex = rVec ? grayCodeToHex(rVec) : null;
      const rDesc = extractDescription(r.entry).replace(/\([^)]*\)/g, '').substring(0, 300).trim();
      const rMath = extractMathData(r.entry);
      const rMathStr = Object.keys(rMath).slice(0, 5).map(k => `${LABELS[k]||k}=${rMath[k]}`).join(', ');
      // Shadow split for related entry
      const rShadow = rVec ? ShadowProcessor.split(rVec) : null;
      const rTgic = rVec ? TGICConstraintSystem.check3AxisOrthogonality(rVec) : null;
      const pullStr = r.relationalPull ? `pull=${r.relationalPull.toFixed(3)} d=${r.hammingDist}` : '';
      lines.push('');
      lines.push(`• **${extractName(r.entry)}**${rHex ? ` \`${rHex}\`` : ''}${pullStr ? ` *[${pullStr}]*` : ''}`);
      lines.push(`  ${rDesc}`);
      if (rMathStr) lines.push(`  *${rMathStr}*`);
      if (rShadow && rTgic !== null) {
        lines.push(`  *Shadow: N=${rShadow.noumenal_hw}/12 · P=${rShadow.phenomenal_hw}/12 · 3-axis harmony=${rTgic.toFixed(3)}*`);
      }
    }
  }

  // ── 3. MoE Linguist elaboration ───────────────────────────────────────────
  if (linguistElaboration) {
    lines.push('');
    lines.push(`**Geometric elaboration:** *${linguistElaboration}*`);
  }

  // ── 4. Collected Thoughts — final synthesised reply ───────────────────────
  {
    const thoughtLines = [];
    thoughtLines.push('---');
    thoughtLines.push('**Collected Thoughts:**');

    const channelDesc = queryType === 'property'
      ? `Gray-code numeric channel (α=0.00) matched on ${Object.keys(mathDict).map(k => LABELS[k]||k).join(', ') || 'numeric properties'}`
      : queryType === 'symbolic'
      ? `Golay symbolic channel (α=1.00) with BW-256 tiebreaker`
      : `Dual-channel (α=0.25): Golay semantic + Gray-code numeric`;

    thoughtLines.push(`The geometric substrate identified **${name}** as the primary resonance via ${channelDesc}.`);

    // UBP-grounded semantic reasoning
    const nrciVal = top.nrci;
    if (nrciVal > 0.7) {
      thoughtLines.push(`With NRCI ${nrciVal.toFixed(3)}, this entry is geometrically stable — it occupies a well-defined Golay codeword position with low symmetry tax.`);
    } else if (nrciVal > 0.4) {
      thoughtLines.push(`NRCI ${nrciVal.toFixed(3)} indicates moderate geometric stability — this concept sits in a mid-weight codeword region of the 24D manifold.`);
    } else {
      thoughtLines.push(`NRCI ${nrciVal.toFixed(3)} is relatively low — the entry sits in a high-weight, high-entropy region of the Golay sphere.`);
    }

    // Confidence assessment
    if (confidence > 0.6) {
      thoughtLines.push(`Confidence is high (${(confidence*100).toFixed(0)}%) — the Hamming landscape shows a clear spike; the nearest competitor is well-separated in the 24D manifold.`);
    } else if (confidence > 0.25) {
      thoughtLines.push(`Confidence is moderate (${(confidence*100).toFixed(0)}%) — adjacent entries exist at similar distances. The top match is geometrically preferred but not uniquely isolated.`);
    } else {
      thoughtLines.push(`Confidence is low (${(confidence*100).toFixed(0)}%) — the query lands in a dense region of the Golay space. Consider providing more specific property values or a longer phrase.`);
    }

    // Lattice geometry note
    if (lattice && lattice !== 'Off-lattice') {
      thoughtLines.push(`The entry sits on a **${lattice}** lattice node (syndrome weight ${bwFp?.sw}/24), indicating ${
        lattice === 'Dodecad' ? 'maximum geometric stability in the Golay sphere-packing' :
        lattice === 'Octad'   ? 'a well-structured 8-weight codeword — high information density' :
        lattice === 'Universe'? 'a maximally complex 24-weight codeword — universal coverage' :
        'a recognised lattice position with defined Hamming geometry'
      }.`);
    }

    // Shadow Processor report (from core.txt)
    if (vec) {
      const shadow = ShadowProcessor.split(vec);
      const tgicScore = TGICConstraintSystem.auditStability(vec, [], nrciVal);
      thoughtLines.push(`**Shadow Processor:** Noumenal (structural) = ${shadow.noumenal_hw}/12 bits · Phenomenal (observable) = ${shadow.phenomenal_hw}/12 bits · Balance = ${(shadow.balance * 100).toFixed(0)}% phenomenal.`);
      thoughtLines.push(`**TGIC 3-6-9 Audit:** 3-axis orthogonality=${TGICConstraintSystem.check3AxisOrthogonality(vec).toFixed(3)} · 6-face coherence=${TGICConstraintSystem.check6FaceCoherence(vec).toFixed(3)} · Combined stability=${tgicScore.toFixed(3)}.`);
    }

    // Relational manifold coherence (from TGICRelationalGravity)
    if (related.length >= 2) {
      const relVecs = related.map(r => extractVector(r.entry)).filter(Boolean);
      const manifoldCoh = TGICRelationalGravity.manifoldCoherence(relVecs);
      thoughtLines.push(`**Relational manifold coherence** (${related.length} entries): ${manifoldCoh.toFixed(4)} — ${manifoldCoh > 0.05 ? 'high geometric clustering (related entries share a dense Golay neighbourhood)' : 'sparse neighbourhood (related entries are geometrically distributed)'}.`);
    }

    // Semantic bridge summary
    if (langVectors && vec && queryType !== 'symbolic') {
      const refl = reflectMeaning(vec, langVectors);
      if (refl && !refl.isLexicalGap) thoughtLines.push(`Through Semantic Reflection, this physical entity maps back to the language operator **${refl.word}** — meaning the substrate associates its geometry with that conceptual primitive.`);
      else if (refl?.isLexicalGap) thoughtLines.push(`Semantic Reflection returned an **Undefined Geometric State** (gap score ${(refl.score * 100).toFixed(1)}%) — this entity's geometry falls between known OP_ operators, suggesting a lexical gap in the language KB.`);
    }

    // MOG synthesis note
    if (mogCats.length > 0) {
      thoughtLines.push(`Active MOG tensor categories (${mogCats.slice(0,4).join(', ')}) indicate the entry engages ${mogCats.length} ontological dimensions simultaneously.`);
    }

    // Related summary
    if (related.length > 0) {
      const relNames = related.slice(0, 3).map(r => extractName(r.entry)).join(', ');
      thoughtLines.push(`Geometrically adjacent entries include ${relNames} — these share overlapping Golay codeword regions and may be relevant to the broader query context.`);
    }

    // Phase-Lock status (from Triadic Verification Manifold — paper §4.1)
    if (phaseResult) {
      thoughtLines.push(`**Triadic Phase-Lock:** ${phaseResult.statusEmoji} **${phaseResult.status}** — ${phaseResult.message}`);
      if (phaseResult.status === 'PHASE_LOCKED') {
        thoughtLines.push(`All three conditions satisfied: semantic (${phaseResult.checks.semantic.value}≥${phaseResult.checks.semantic.threshold}), substrate NRCI (${phaseResult.checks.substrate.value}≥${phaseResult.checks.substrate.threshold}), logic (${phaseResult.checks.logic.value}).`);
      }
    }

    lines.push('');
    lines.push(thoughtLines.join('\n'));
  }

  // ── 5. Provenance footer ──────────────────────────────────────────────────
  if (confidence < 0.15 && results.length > 1) {
    lines.push('');
    lines.push(`*Low confidence — consider rephrasing with more specific property values or entity names.*`);
  }
  if (bwFp && bwFp.macroNrci !== null) {
    lines.push('');
    lines.push(`*BW-256: macroNRCI ${bwFp.macroNrci.toFixed(4)} · clarity ${bwFp.clarity}*`);
  }
  const channel = queryType === 'property' ? 'Gray-code numeric' :
                  queryType === 'symbolic'  ? 'Golay symbolic + BW-256' : 'dual-channel + BW-256';
  const prov = [channel, `α=${top.alpha?.toFixed(2) ?? '?'}`, `sim ${(top.raw * 100).toFixed(1)}%`, `NRCI ${top.nrci.toFixed(3)}`, `conf ${(confidence * 100).toFixed(1)}%`];
  lines.push(''); lines.push(`*${prov.join(' · ')}*`);

  return lines.join('\n');
}

// ── Context & Learning ────────────────────────────────────────────────────────
class ContextWindow {
  constructor(max = 8) { this.max = max; this.turns = []; }
  add(q, r) { this.turns.push({ q, r, t: Date.now() }); if (this.turns.length > this.max) this.turns = this.turns.slice(-this.max); }
  recentWords() { return this.turns.slice(-2).flatMap(t => t.q.toLowerCase().split(/\s+/)).filter(w => w.length > 3 && !STOP_WORDS.has(w)); }
}

class HebbianTrainer {
  constructor() { this.overrides = new Map(); this.log = []; }
  reinforce(uids, positive, uidToEntry) {
    uids.forEach((uid, i) => {
      const entry = uidToEntry ? uidToEntry.get(uid) : null;
      if (!entry) return;
      const base = extractNrci(entry);
      const cur = this.overrides.get(uid) ?? base;
      const delta = 0.08 * ((uids.length - i) / uids.length);
      this.overrides.set(uid, Math.max(0.01, Math.min(1, positive ? cur + delta * (1 - cur) : cur - delta)));
    });
    this.log.push({ positive, t: Date.now() });
  }
  getNrci(uid, base) { return this.overrides.get(uid) ?? base; }
}

            if (li.nearest_octad_dist <= 4) {
              lines.push('');
              lines.push(`**Leech Λ₂₄:** nearest octad dist=${li.nearest_octad_dist} · ${li.leech_points} lattice points · norm²_scaled=${li.norm_sq_scaled}`);
              const oh = li.ontological_health;
              lines.push(`  Reality=${oh.Reality.toFixed(3)} · Info=${oh.Info.toFixed(3)} · Activation=${oh.Activation.toFixed(3)} · Potential=${oh.Potential.toFixed(3)} · Global_NRCI=${oh.Global_NRCI.toFixed(3)}`);
            }
          } catch { /* non-critical */ }
        }

        // ── Triad state ───────────────────────────────────────────────────────
        try {
          const alu   = getNoiseCoreALU();
          const triad = alu.triadSnapshot();
          lines.push('');
          lines.push(`**Triad state:** Golay ${triad.golay_active ? '✓' : '○'} Leech ${triad.leech_active ? '✓' : '○'} Monster ${triad.monster_active ? '✓' : '○'} · Level **${triad.level}/3** · Stable ops: ${triad.stable_count}`);
        } catch { /* non-critical */ }

      } else if (rNorm && rNorm.error) {
        lines.push(`⚠ ${rNorm.error}`);
      }
      lines.push('');
      lines.push(`*NoiseCore v4.0 · Substrate-Verified (SV) mode · Golay [24,12,8] + Leech Λ₂₄ + BW-256*`);
      const response = lines.join('\n');
      this.context.add(query, response);
      this.linguist.learnFrom(query + ' ' + response, 80, `NoiseCore: "${aluResult.program}"`);
      return {
        query, domain: 'ALGORITHM', response, confidence: 1.0,
        queryType: 'noisecore', mathDict: {}, alpha: 0,
        reflexiveHits: 0,
        steps: [{ content: `NoiseCore ALU: ${aluResult.program} = ${rNorm?.value ?? '?'}`, domain: 'ALGORITHM', source: 'noisecore-alu', coherence: 1.0, vectorHex: null, distance: null, nrci: rNorm?.nrci ?? 0.5, semSim: 1, numericSim: 1, uid: null, vector: null, ontologicalHealth: null, fold3: null }],
        tokens: [],
        queryVector: { bits: null, numericBits: 0, propertiesExtracted: 0, anchors: 0, lexicalGaps: [] },
        stats: { spikesFound: 1, anchors: 0, reflexiveHits: 0, queryType: 'noisecore', alpha: '0.00', geometricHashes: 0, lexicalGaps: 0, chainLength: 1, memoriesSearched: 0, confidence: '100%', propertiesExtracted: 0 },
        aluResult,
      };
    }

    // ── Domain intent detection ───────────────────────────────────────────────
    const domainIntent = detectDomainIntent(query);
    const isLawQuery   = LAW_KEYWORDS.test(query);

    // ── Identity Lock (ubp_brain_consolidated.py v7.2) ───────────────────────
    // Fast-path: if a high-confidence n-gram lock fires, skip full scoring.
    const queryWords = query.toLowerCase().replace(/[^a-z0-9\s]/g, ' ').split(/\s+/).filter(w => w && !STOP_WORDS.has(w));
    const lock = identityLock(queryWords, this.trigramIndex, this.bigramIndex, this.shortNameIndex, this.uidToEntry, domainIntent);

    // ── Reflexive scan (phrase-lock + tag index recall) ──────────────────────
    const reflexive = reflexiveScan(query, this.kbData, this.shortNameIndex, this.tagIndex, this.uidToEntry);

    // ── NL-to-Math extraction (Channel 2) ─────────────────────────────────────
    const { mathDict, queryType } = extractMathFromNL(query);
    let queryNumericVec = null, queryPresenceMask = null;
    if (Object.keys(mathDict).length > 0) {
      const encoded = encodeNumericVector(mathDict);
      queryNumericVec = encoded.vector;
      queryPresenceMask = encoded.presenceMask;
    }

    // Alpha: how much to weight the Golay symbolic channel
    const alpha = queryType === 'property' ? 0.0 :
                  queryType === 'symbolic'  ? 1.0 : 0.25;

    // ── Symbolic tokenisation (Channel 1) ─────────────────────────────────────
    const ctxQuery = this.context.recentWords().slice(0, 3).reduce((q, w) => q + ' ' + w, query);
    const { matchedVectors, corroboration } = tokenise(
      ctxQuery, this.kbData, this.shortNameIndex,
      this.bigramIndex, this.trigramIndex, this.lexiconIndex, this.uidToEntry
    );
    // v7.0 weighted probe assembly (from ubp_semantic_engine.py):
    // ELEM_/MOLECULE_ = 10x, PARTICLE_ = 8x, OP_ = 0.5x on physical queries
    for (const t of matchedVectors) {
      if (t.uid && /^(ELEM_|MOLECULE_)/.test(t.uid))       t.weight *= (domainIntent.physical ? 10 : 3);
      else if (t.uid && /^PARTICLE_/.test(t.uid))           t.weight *= (domainIntent.particle ? 8 : 3);
      else if (t.uid && /^(ALGO_|MATH_)/.test(t.uid))       t.weight *= (domainIntent.scripting ? 5 : 1.5);
      else if (t.uid && /^OP_/.test(t.uid))                 t.weight *= (domainIntent.physical && !domainIntent.symbolic ? 0.1 : 0.5);
    }

    // If Identity Lock fired, inject it with maximum corroboration boost
    if (lock) {
      corroboration.set(lock.uid, (corroboration.get(lock.uid) || 0) + 50);
    }

    // Lexical gap detection: words with no KB anchor
    const lexicalGaps = matchedVectors
      .filter(t => t.type === 'geometric-hash')
      .map(t => t.word);

    // Inject reflexive boosts into corroboration
    // Cap law/belief entries to a small reflexive boost regardless of phrase matches
    for (const [uid, { boost }] of reflexive) {
      const cappedBoost = (!isLawQuery && /^(LAW_|BELIEF_|AXIOM_|IMPERATIVE_)/.test(uid)) ? Math.min(boost, 0.2) : boost;
      corroboration.set(uid, (corroboration.get(uid) || 0) + cappedBoost);
    }

    const queryVec = buildQueryVector(matchedVectors);
    if (!queryVec && !queryNumericVec) return this._emptyResult(query);

    // ── Dual-channel scoring ──────────────────────────────────────────────────
    const results = scoreAll(queryVec, queryNumericVec, queryPresenceMask,
                              this.kbData, corroboration, this.fomManager, alpha, isLawQuery, domainIntent);
    if (!results.length) return this._emptyResult(query);

    const confidence = computeConfidence(results);

    // MoE elaboration via Linguist
    const topVec = results[0] ? extractVector(results[0].entry) : null;
    const linguistElaboration = this.linguist.trained
      ? this.linguist.elaborate(topVec, query, this.shortNameIndex, this.uidToEntry)
      : null;

    // MoE Researcher: XOR bridge ranking for related entries (from ubp_moe_cortex_v2.py)
    // Re-ranks related entries by their XOR bridge stability to the top result's vector
    if (topVec && results.length > 1) {
      const relatedCandidates = results.slice(1, 10).map(r => r.entry);
      const moeRanked = MoEResearcher.rankByGoalProximity(topVec, relatedCandidates, extractVector, extractName);
      // Boost corroboration of MoE-ranked entries
      for (const { uid, score } of moeRanked.slice(0, 3)) {
        if (uid) {
          const idx = results.findIndex(r => r.uid === uid);
          if (idx > 0) results[idx].score *= (1 + score * 0.15); // up to +15% boost
        }
      }
      results.sort((a, b) => b.score - a.score);
    }

    // Phase-Lock Verification on top result
    const topNrci = results[0]?.nrci ?? 0;
    const phaseVerifier = getPhaseLockVerifier();
    const phaseResult = phaseVerifier.verify(confidence, topNrci, results[0]?.entry, false);

    const response = synthesise(results, matchedVectors, confidence, queryType, mathDict, linguistElaboration, this.langVectors, phaseResult);
    this.context.add(query, response);
    this.lastUids = results.slice(0, 5).map(r => r.uid).filter(Boolean);

    // Incremental learning: teach linguist from this interaction
    this.linguist.learnFrom(query + ' ' + response, 150, `Query: "${query.slice(0, 60)}…"`);
    // Track lexical gaps
    for (const w of lexicalGaps) this.linguist.trackGap(w);

    return {
      query, domain: results[0].domain, response, confidence,
      queryType, mathDict, alpha,
      reflexiveHits: reflexive.size,
      steps: results.slice(0, 5).map(r => {
        const rVec = extractVector(r.entry);
        const bwFp = r.bwFp || (rVec ? (() => { try { return computeUBPFingerprint(rVec); } catch { return null; } })() : null);
        const shadow = rVec ? ShadowProcessor.split(rVec) : null;
        const tgicOrtho = rVec ? TGICConstraintSystem.check3AxisOrthogonality(rVec) : null;
        return {
          content: `${extractName(r.entry)}. ${extractDescription(r.entry)}`.substring(0, 300),
          domain: r.domain,
          source: r.corrobBoost > 0 ? 'anchor' : (r.alpha < 0.5 ? 'numeric' : 'symbolic'),
          coherence: (r.raw + 1) / 2,
          vectorHex: r.fingerprint?.substring(0, 6) || null,
          distance: queryVec ? hammingDistance(queryVec, rVec || []) : null,
          nrci: this.trainer.getNrci(r.uid, r.nrci),
          semSim: r.semSim, numericSim: r.numericSim,
          uid: r.uid,
          vector: rVec,
          ontologicalHealth: rVec ? getOntologicalHealth(rVec) : null,
          fold3: rVec ? fold24to3(rVec) : null,
          lattice: bwFp ? bwFp.lattice : null,
          macroNrci: bwFp ? bwFp.macroNrci : null,
          clarity: bwFp ? bwFp.clarity : null,
          sw: bwFp ? bwFp.sw : null,
          // TGIC + Shadow (new in v9.0)
          shadow: shadow ? { noumenal_hw: shadow.noumenal_hw, phenomenal_hw: shadow.phenomenal_hw, balance: shadow.balance } : null,
          tgicOrtho,
        };
      }),
      tokens: matchedVectors.slice(0, 10).map(t => ({
        word: t.word,
        domain: t.uid ? inferDomain(this.uidToEntry?.get(t.uid) || {}) : '?',
        definition: t.uid
          ? extractDescription(this.uidToEntry?.get(t.uid) || {}).substring(0, 80)
          : '[geometric hash → Golay 24D]',
        type: t.type,
      })),
      queryVector: {
        bits: queryVec,
        numericBits: queryNumericVec ? queryNumericVec.length : 0,
        propertiesExtracted: Object.keys(mathDict).length,
        anchors: corroboration.size,
        lexicalGaps,
      },
      stats: {
        spikesFound: results.length,
        anchors: corroboration.size,
        reflexiveHits: reflexive.size,
        queryType,
        alpha: alpha.toFixed(2),
        geometricHashes: matchedVectors.filter(t => t.type === 'geometric-hash').length,
        lexicalGaps: lexicalGaps.length,
        tagHits: [...reflexive.values()].filter(v => v.boost === 1.2).length,
        domainIntent: [
          domainIntent.physical  ? 'physical'  : '',
          domainIntent.symbolic  ? 'symbolic'  : '',
          domainIntent.particle  ? 'particle'  : '',
          domainIntent.scripting ? 'scripting' : '',
          domainIntent.mathDeep  ? 'math'      : '',
        ].filter(Boolean).join('+') || 'general',
        identityLock: lock ? lock.method : null,
        chainLength: Math.min(results.length, 5),
        memoriesSearched: Object.keys(this.kbData).length,
        confidence: (confidence * 100).toFixed(1) + '%',
        propertiesExtracted: Object.keys(mathDict).length,
      }
    };
  }

  // ── Swarm result renderer (v28 NativeMath operations) ────────────────────
  _renderSwarmResult(query, swarmResult) {
    const { op, inputs, result: r, program, mode } = swarmResult;
    const lines = [];
    lines.push(`**Swarm Agent [${op}]** · *${program}*`);
    lines.push(`*Mode: ${mode}*`);
    lines.push('');

    if (!r || r.error || r.value === null || r.value === undefined) {
      lines.push(`⚠ ${r?.error || 'No result'}`);
    } else {
      // Format value
      const displayVal = typeof r.value === 'number'
        ? (Number.isInteger(r.value) ? r.value.toLocaleString() : r.value.toFixed(8))
        : String(r.value);
      lines.push(`**Result:** \`${displayVal}\``);

      // Special handling for structured results
      if (r.report) {
        // Lattice attraction report
        const rep = r.report;
        lines.push('');
        lines.push(`**Lattice Attraction Analysis:**`);
        lines.push(`Total ops: ${rep.total} · On-lattice: ${rep.onLattice} (${rep.rate})`);
        lines.push(`Random baseline: ${rep.randomBaseline} · **Attraction ratio: ${rep.attractionRatio}**`);
        lines.push(`Top lattice type: ${rep.topLattice} (${rep.topLatticeCount} hits)`);
        lines.push(`*${rep.hypothesis}*`);
      } else if (r.factors) {
        // Factorisation
        lines.push('');
        lines.push(`**Prime factors:** ${Object.entries(r.factors).map(([p,e]) => e > 1 ? `${p}^${e}` : p).join(' × ')}`);
      } else if (r.type) {
        // Quadratic
        lines.push(`**Discriminant type:** ${r.type}`);
      }

      // Substrate fingerprint (if available from NativeMath+Substrate mode)
      if (r.lattice && r.lattice !== 'N/A') {
        lines.push('');
        lines.push(`**Lattice:** ${r.lattice} · **NRCI:** ${r.nrci ? r.nrci.toFixed(4) : '—'}`);
        if (r.sw !== undefined) lines.push(`**Syndrome weight:** ${r.sw}/24 · **On lattice:** ${r.onLattice ? 'Yes' : 'No'}`);
        if (r.macroNrci) lines.push(`**BW-256:** macroNRCI ${r.macroNrci.toFixed(4)} · clarity **${r.clarity}**`);

        // Phase-lock verification
        const phaseVerifier = getPhaseLockVerifier();
        const nrciVal = r.nrci || 0;
        const phaseResult = phaseVerifier.verify(1.0, nrciVal, r.value, false);
        lines.push('');
        lines.push(`**Phase-Lock:** ${phaseResult.statusEmoji} ${phaseResult.status} — ${phaseResult.message}`);

        // Golay trace
        if (typeof r.value === 'number') {
          const absV = Math.abs(Math.floor(r.value)) & 0xFFFFFF;
          const gray = absV ^ (absV >> 1);
          const bits24 = [];
          for (let i = 23; i >= 0; i--) bits24.push((gray >> i) & 1);
          const msg12 = bits24.slice(0, 12);
          const computedParity = new Array(12).fill(0);
          for (let i = 0; i < 12; i++)
            for (let j = 0; j < 12; j++)
              computedParity[i] = (computedParity[i] + msg12[j] * GOLAY_B[j][i]) % 2;
          const sw = computedParity.reduce((s, b) => s + b, 0);
          lines.push('');
          lines.push(`**Golay Fingerprint:** \`[${bits24.join('')}]\` → sw=${sw}/12 → ${r.lattice}`);
        }
      }
    }

    lines.push('');
    lines.push(`*Swarm v1.0 · NativeMath (v28 Oracle port) · ${mode}*`);
    const response = lines.join('\n');
    this.context.add(query, response);
    this.linguist.learnFrom(query + ' ' + response, 80, `Swarm: "${program}"`);
    const displayVal = r?.value !== null && r?.value !== undefined ? r.value : '?';
    return {
      query, domain: 'ALGORITHM', response, confidence: 1.0,
      queryType: 'noisecore', mathDict: {}, alpha: 0,
      reflexiveHits: 0,
      steps: [{ content: `Swarm [${op}]: ${program} = ${displayVal}`, domain: 'ALGORITHM', source: 'swarm-router', coherence: 1.0, vectorHex: null, distance: null, nrci: r?.nrci ?? 0.5, semSim: 1, numericSim: 1, uid: null, vector: null, ontologicalHealth: null, fold3: null }],
      tokens: [],
      queryVector: { bits: null, numericBits: 0, propertiesExtracted: 0, anchors: 0, lexicalGaps: [] },
      stats: { spikesFound: 1, anchors: 0, reflexiveHits: 0, queryType: 'noisecore', alpha: '0.00', geometricHashes: 0, lexicalGaps: 0, chainLength: 1, memoriesSearched: 0, confidence: '100%', propertiesExtracted: 0 },
    };
  }

  _emptyResult(query) {
    return {
      query, domain: 'MEANING',
      response: "No KB matches found. Try a specific element, molecule, particle, or UBP concept.",
      confidence: 0, steps: [], tokens: [], mathDict: {}, queryType: 'symbolic', alpha: 1,
      queryVector: { bits: null, numericBits: 0, propertiesExtracted: 0, anchors: 0 },
      stats: { spikesFound: 0, anchors: 0, reflexiveHits: 0, queryType: 'symbolic', alpha: '1.00',
               geometricHashes: 0, chainLength: 0, memoriesSearched: Object.keys(this.kbData).length,
               confidence: '0%', propertiesExtracted: 0 }
    };
  }

  feedback(positive) {
    this.trainer.reinforce(this.lastUids, positive, this.uidToEntry);
    // Linguist reinforcement: boost patterns from good responses
    this.linguist.reinforceFromFeedback(this.lastUids, this.uidToEntry, positive);
  }
  switchFOM(id) { return this.fomManager.switchFrame(id); }
  listFOMs() { return this.fomManager.listFrames(); }

  stats() {
    const domains = {};
    for (const e of Object.values(this.kbData)) { const d = inferDomain(e); domains[d] = (domains[d] || 0) + 1; }
    return {
      totalMemories: Object.keys(this.kbData).length,
      lexiconClusters: this.shortNameIndex.size,
      bigramCount: this.bigramIndex.size,
      trigramCount: this.trigramIndex.size,
      domains,
      contextTurns: this.context.turns.length,
      feedbackCount: this.trainer.log.length,
      activeFOM: this.fomManager.activeFrameId,
      initialized: this.initialized,
      linguist: this.linguist.stats(),
    };
  }

  lexiconStats() {
    const N = Object.keys(this.kbData).length;
    const t = this.shortNameIndex.size + this.bigramIndex.size + this.trigramIndex.size;
    return {
      loaded: this.shortNameIndex.size > 0,
      totalWords: t,
      clusterCount: this.shortNameIndex.size,
      bigramCount: this.bigramIndex.size,
      trigramCount: this.trigramIndex.size,
      avgClusterSize: N > 0 ? parseFloat((t / N).toFixed(2)) : 1,
    };
  }

  clearConversation() { this.context.turns = []; }

  langVectorCount() { return this.langVectors ? Object.keys(this.langVectors).length : 0; }
}

let _engine = null;
