    return 1 - np.std(abs_corr) / np.mean(abs_corr) if np.mean(abs_corr) > 0 else 0

import os
import numpy as np
import pandas as pd
from scipy.spatial.distance import cosine


# Config (tune as needed)
DATA_DIR = '/content/nist_data/'
ALICE_FILE = '21_15_CH_pockel_100kHz.run.200nsadditiondelay_lightconeshift.alice.dat'
BOB_FILE = '21_15_CH_pockel_100kHz.run.200nsadditiondelay_lightconeshift.bob.dat'
SAMPLE_SIZE = 1000000  # Subsample for speed; set None for full
W_MIN, W_MAX, W_STEPS = 1.7, 2.2, 100
W_TETRA = np.pi / ((1 + np.sqrt(5)) / 2)  # ~1.9416


def extract_binary_outcomes(df, plus_col=1, minus_col=2):
    """Binary: 1 (+ click), 0 (−), -1 (no-click/ambig)."""
    plus = df.iloc[:, plus_col].values.astype(int)
    minus = df.iloc[:, minus_col].values.astype(int)
    return np.where(plus > 0, 1, np.where(minus > 0, 0, -1))


def calculate_nrci(a_bits, b_bits, w):
    valid = (a_bits != -1) & (b_bits != -1)
    if np.sum(valid) == 0: return 0.0
    a, b = a_bits[valid], b_bits[valid]
    xor = np.bitwise_xor(a, b).astype(float)
    if np.sum(np.abs(xor)) == 0: return 0.0
    dist = cosine(xor, np.ones(len(a)))
    nrci = 1.0 - dist
    anticorr = np.mean(xor)
    penalty = np.abs(anticorr - 1.0)
    return nrci * np.exp(-w * penalty)


def estimate_coherence_pressure(a_bits, b_bits, w_opt):
    valid = (a_bits != -1) & (b_bits != -1)
    if np.sum(valid) == 0: return 0.0
    xor = np.bitwise_xor(a_bits[valid], b_bits[valid])
    anticorr = np.mean(xor)
    return (1.0 - anticorr) / w_opt


# Load (chunk if huge; here assumes fits in mem)
df_a = pd.read_csv(os.path.join(DATA_DIR, ALICE_FILE), sep=r'\s+', header=None)
df_b = pd.read_csv(os.path.join(DATA_DIR, BOB_FILE), sep=r'\s+', header=None)
print(f"Loaded: Alice {df_a.shape}, Bob {df_b.shape}")


# Extract (align rows; subsample if needed)
a_bits = extract_binary_outcomes(df_a)
b_bits = extract_binary_outcomes(df_b)
if SAMPLE_SIZE:
    idx = np.random.choice(len(a_bits), SAMPLE_SIZE, replace=False)
    a_bits, b_bits = a_bits[idx], b_bits[idx]
n_valid = np.sum((a_bits != -1) & (b_bits != -1))
print(f"Valid pairs: {n_valid}")


# Scan
weights = np.linspace(W_MIN, W_MAX, W_STEPS)
nrci_scores = np.array([calculate_nrci(a_bits, b_bits, w) for w in weights])
best_idx = np.argmax(nrci_scores)
best_w, max_nrci = weights[best_idx], nrci_scores[best_idx]
psi_p = estimate_coherence_pressure(a_bits, b_bits, best_w)


# Results
print("\n=== NIST Inverse BitGrok Results ===")
print(f"Best w: {best_w:.6f} (Tetra: {W_TETRA:.6f}, Dev: {abs(best_w - W_TETRA):.6f})")
print(f"Max NRCI: {max_nrci:.6f}")
print(f"Est Ψ_p: {psi_p:.2e}")
print("Aligns? " + ("✅ Tetra peak!" if abs(best_w - W_TETRA) < 0.05 else "⚠️ Deviation—noise?"))


# Plot
import matplotlib.pyplot as plt
plt.plot(weights, nrci_scores); plt.axvline(W_TETRA, color='r', ls='--', label='Tetra')
plt.xlabel('w'); plt.ylabel('NRCI'); plt.legend(); plt.show()


# Save scan
np.savetxt('nist_nrci_scan.txt', np.column_stack([weights, nrci_scores]), header='w nrci')
