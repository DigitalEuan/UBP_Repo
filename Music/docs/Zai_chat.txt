def to_gray_code(n: int, bits: int = 12) -> list:
    """Convert integer to n-bit binary-reflected Gray code."""
    gray = n ^ (n >> 1)
    return [(gray >> i) & 1 for i in range(bits - 1, -1, -1)]

def vector_to_hex_color(vec: list) -> str:
    """Convert 24-bit vector to #RRGGBB hex color."""
    r = sum((1 << (7 - i)) for i in range(8) if vec[i])
    g = sum((1 << (7 - i)) for i in range(8) if vec[8 + i])
    b = sum((1 << (7 - i)) for i in range(8) if vec[16 + i])
